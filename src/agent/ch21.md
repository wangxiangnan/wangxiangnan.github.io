# 第21章 自定义Tool开发 —— 打造Agent的专属工具箱

## 21.1 开篇：好的Tool像好的API

如果你写过 REST API，你一定深有体会：一个好的 API 是什么样的？——简洁、可预测、有清晰的文档（Schema）。你传入什么参数，得到什么返回，一目了然。调用者不需要看实现代码就能正确使用它。

**好的 Tool 也是一样的。** 对于 AI Agent 来说，Tool 就是它和外部世界交互的 API。Agent 通过 Tool 来读取文件、搜索网页、调用外部服务、执行计算。如果 Tool 设计得不好，Agent 就会像一个拿着一堆烂工具的工程师——再聪明也干不好活。

在第12章中，我们已经学习了 LangChain Tools 的基础用法。这一章，我们要从"会用Tool"进阶到"会造Tool"。

## 21.2 Tool设计原则

### 21.2.1 单一职责：一个Tool做一件事

```python
# 反面例子：一个Tool干了太多事
@tool
def do_everything(query: str, action: str) -> str:
    """做所有事情"""
    if action == "search":
        return search_something(query)
    elif action == "analyze":
        return analyze_something(query)
    elif action == "translate":
        return translate_something(query)

# 正面例子：每个Tool只做一件事
@tool
def search_documents(query: str) -> str:
    """从文档库中搜索相关信息"""
    return search_something(query)

@tool
def analyze_text(text: str) -> str:
    """对文本进行深度分析"""
    return analyze_something(text)

@tool
def translate_text(text: str, target_language: str) -> str:
    """将文本翻译为目标语言"""
    return translate_something(text, target_language)
```

为什么要这样？因为 LLM 在选择 Tool 时，需要根据 Tool 的名称和描述来判断该用哪个。如果一个 Tool 什么都能干，LLM 就很难做出正确选择——就像面前有一把瑞士军刀，但你需要的是精确的手术刀。

### 21.2.3 输出可预测：一致的成功/失败格式

```python
import json


def make_success(data: dict) -> str:
    """构建成功响应"""
    return json.dumps({
        "status": "success",
        "data": data,
        "message": "操作成功"
    }, ensure_ascii=False)


def make_error(error_type: str, message: str, details: str = "") -> str:
    """构建错误响应"""
    return json.dumps({
        "status": "error",
        "error_type": error_type,
        "message": message,
        "details": details
    }, ensure_ascii=False)
```

Agent 看到一致的输出格式，就能更容易理解结果并决定下一步行动。

## 21.3 @tool装饰器深入

### 21.3.1 参数类型标注和Pydantic验证

```python
from langchain_core.tools import tool
from pydantic import BaseModel, Field, field_validator
from typing import Optional


class PDFParserInput(BaseModel):
    file_path: str = Field(description="PDF文件的绝对路径")
    pages: Optional[str] = Field(default=None, description="页码范围，如'1-5'")
    extract_tables: bool = Field(default=False, description="是否提取表格数据")
    output_format: str = Field(default="text", description="输出格式：text/markdown/json")
    
    @field_validator("file_path")
    @classmethod
    def validate_path(cls, v):
        if not v.endswith(".pdf"):
            raise ValueError("文件必须是PDF格式")
        return v
    
    @field_validator("output_format")
    @classmethod
    def validate_format(cls, v):
        if v not in ("text", "markdown", "json"):
            raise ValueError("output_format必须是text/markdown/json之一")
        return v


@tool(args_schema=PDFParserInput)
def pdf_parser(file_path: str, pages: str = None, 
               extract_tables: bool = False, output_format: str = "text") -> str:
    """解析PDF文档，提取文本内容和表格数据。支持指定页码范围和多种输出格式。"""
    import fitz  # PyMuPDF
    
    try:
        doc = fitz.open(file_path)
        page_indices = _parse_page_range(pages, len(doc)) if pages else list(range(len(doc)))
        
        results = []
        for idx in page_indices:
            page = doc[idx]
            text = page.get_text()
            if output_format == "markdown":
                text = f"## 第{idx + 1}页\n\n{text}"
            elif output_format == "json":
                text = json.dumps({"page": idx + 1, "content": text}, ensure_ascii=False)
            results.append(text)
        
        doc.close()
        output = "\n\n".join(results)
        return make_success({"content": output, "page_count": len(page_indices)})
    except FileNotFoundError:
        return make_error("FILE_NOT_FOUND", f"文件不存在: {file_path}")
    except Exception as e:
        return make_error("PARSE_ERROR", f"PDF解析失败: {str(e)}")


def _parse_page_range(pages_str: str, total: int) -> list:
    """解析页码范围字符串，如'1-5,8,10-12'"""
    indices = set()
    for part in pages_str.split(","):
        part = part.strip()
        if "-" in part:
            start, end = part.split("-", 1)
            start = max(1, int(start.strip()))
            end = min(total, int(end.strip()))
            indices.update(range(start - 1, end))
        else:
            idx = int(part) - 1
            if 0 <= idx < total:
                indices.add(idx)
    return sorted(indices)
```

### 21.3.2 异步Tool支持

对于 I/O 密集型操作（网络请求、文件读写），异步 Tool 可以显著提升性能：

```python
import aiohttp
import asyncio


@tool
async def web_scraper(url: str, selector: str = "body") -> str:
    """异步爬取网页内容。

    Args:
        url: 要爬取的网页URL
        selector: CSS选择器，默认提取body内容
    """
    try:
        async with aiohttp.ClientSession() as session:
            headers = {"User-Agent": "DocMind-Bot/1.0"}
            timeout = aiohttp.ClientTimeout(total=30)
            async with session.get(url, headers=headers, timeout=timeout) as resp:
                if resp.status != 200:
                    return make_error("HTTP_ERROR", f"HTTP {resp.status}")
                
                html = await resp.text()
                from bs4 import BeautifulSoup
                soup = BeautifulSoup(html, "html.parser")
                
                # 移除不需要的标签
                for tag in soup(["script", "style", "nav", "footer"]):
                    tag.decompose()
                
                elements = soup.select(selector)
                text = "\n".join([el.get_text(strip=True) for el in elements])
                text = "\n".join([l.strip() for l in text.splitlines() if l.strip()])
                
                return make_success({
                    "url": url,
                    "title": soup.title.string if soup.title else "",
                    "content": text[:5000],
                    "content_length": len(text)
                })
    except asyncio.TimeoutError:
        return make_error("TIMEOUT", f"请求超时: {url}")
    except Exception as e:
        return make_error("SCRAPE_ERROR", f"爬取失败: {str(e)}")
```

## 21.4 StructuredTool.from_function()详解

当你需要更精细地控制 Tool 行为时：

```python
from langchain_core.tools import StructuredTool
from pydantic import BaseModel, Field
from typing import Optional
import os


class GitHubSearchInput(BaseModel):
    query: str = Field(description="搜索关键词")
    repo: str = Field(description="仓库名，格式：owner/repo")
    search_type: str = Field(default="issues", description="类型：issues/pulls/code")
    state: Optional[str] = Field(default="open", description="状态：open/closed/all")
    limit: int = Field(default=10, ge=1, le=50)


def github_search(query: str, repo: str, search_type: str = "issues",
                  state: str = "open", limit: int = 10) -> str:
    """搜索GitHub仓库"""
    import requests
    
    base_url = f"https://api.github.com/search/{search_type}"
    params = {"q": f"{query} repo:{repo}", "per_page": limit, "sort": "updated"}
    if state and state != "all":
        params["q"] += f" state:{state}"
    
    headers = {"Accept": "application/vnd.github.v3+json"}
    token = os.getenv("GITHUB_TOKEN")
    if token:
        headers["Authorization"] = f"token {token}"
    
    try:
        resp = requests.get(base_url, params=params, headers=headers, timeout=15)
        resp.raise_for_status()
        data = resp.json()
        
        items = [{
            "number": item.get("number"), "title": item.get("title"),
            "state": item.get("state"), "url": item.get("html_url"),
            "labels": [l["name"] for l in item.get("labels", [])]
        } for item in data.get("items", [])[:limit]]
        
        return make_success({"total_count": data.get("total_count", 0), "items": items})
    except requests.RequestException as e:
        return make_error("API_ERROR", f"GitHub API请求失败: {str(e)}")


github_tool = StructuredTool.from_function(
    func=github_search,
    name="github_search",
    description="搜索GitHub仓库中的Issues、PR或代码。支持按状态过滤。",
    args_schema=GitHubSearchInput
)
```

## 21.5 BaseTool类：完全自定义

```python
from langchain_core.tools import BaseTool
from pydantic import BaseModel, Field
from typing import Optional, Type


class APIIntegratorInput(BaseModel):
    base_url: str = Field(description="API基础URL")
    endpoint: str = Field(description="API端点路径")
    method: str = Field(default="GET", description="HTTP方法")
    headers: Optional[dict] = Field(default=None, description="请求头")
    params: Optional[dict] = Field(default=None, description="查询参数")
    body: Optional[dict] = Field(default=None, description="请求体")
    timeout: int = Field(default=30, description="超时秒数")


class APIIntegrator(BaseTool):
    """通用API集成工具 — 可调用任何REST API"""
    
    name: str = "api_integrator"
    description: str = "通用REST API调用工具。支持GET/POST/PUT/DELETE。有域名白名单。"
    args_schema: Type[BaseModel] = APIIntegratorInput
    allowed_domains: list = ["api.github.com", "api.openai.com", "jsonplaceholder.typicode.com"]
    max_response_size: int = 100_000  # 100KB
    
    def _run(self, base_url: str, endpoint: str, method: str = "GET",
             headers: dict = None, params: dict = None, body: dict = None,
             timeout: int = 30) -> str:
        import requests
        from urllib.parse import urlparse
        
        # 安全检查：域名白名单
        domain = urlparse(base_url).netloc
        if domain not in self.allowed_domains:
            return make_error("FORBIDDEN_DOMAIN", f"域名 {domain} 不在白名单中")
        
        url = f"{base_url.rstrip('/')}/{endpoint.lstrip('/')}"
        headers = headers or {}
        headers.setdefault("User-Agent", "DocMind-Agent/1.0")
        
        try:
            resp = requests.request(method, url, headers=headers,
                                    params=params, json=body, timeout=timeout)
            if len(resp.content) > self.max_response_size:
                return make_error("RESPONSE_TOO_LARGE", "响应超过大小限制")
            
            try:
                data = resp.json()
            except ValueError:
                data = resp.text[:2000]
            
            return make_success({"status_code": resp.status_code, "data": data})
        except requests.Timeout:
            return make_error("TIMEOUT", f"请求超时: {url}")
        except requests.ConnectionError:
            return make_error("CONNECTION_ERROR", f"连接失败: {url}")
        except Exception as e:
            return make_error("UNKNOWN_ERROR", str(e))
    
    async def _arun(self, **kwargs) -> str:
        return self._run(**kwargs)


api_tool = APIIntegrator()
```

```python
from datetime import datetime


@tool
def write_report(content: str, filename: str, output_dir: str = "./reports") -> str:
    """将分析报告写入文件。"""
    try:
        # 安全检查：防止路径遍历攻击
        safe_filename = os.path.basename(filename)
        if not safe_filename.endswith(".md"):
            safe_filename += ".md"
        
        os.makedirs(output_dir, exist_ok=True)
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        name, ext = os.path.splitext(safe_filename)
        final_name = f"{name}_{timestamp}{ext}"
        file_path = os.path.join(output_dir, final_name)
        
        with open(file_path, "w", encoding="utf-8") as f:
            f.write(content)
        
        return make_success({
            "file_path": file_path, "file_size": len(content),
            "message": f"报告已保存: {file_path}"
        })
    except PermissionError:
        return make_error("PERMISSION_DENIED", f"无权限写入: {output_dir}")
    except Exception as e:
        return make_error("WRITE_ERROR", f"写入失败: {str(e)}")


@tool
def send_notification(recipient: str, subject: str, body: str,
                     channel: str = "email") -> str:
    """发送通知消息（email/slack/webhook）。"""
    if channel == "email" and "@" not in recipient:
        return make_error("INVALID_INPUT", "邮箱格式不正确")
    
    notification = {
        "id": f"notif_{datetime.now().timestamp():.0f}",
        "recipient": recipient, "subject": subject,
        "body": body[:200], "channel": channel,
        "sent_at": datetime.now().isoformat(), "status": "sent"
    }
    _log_notification(notification)
    return make_success({
        "notification_id": notification["id"],
        "message": f"通知已通过{channel}发送给{recipient}"
    })


def _log_notification(notification: dict):
    """记录通知审计日志"""
    log_dir = "./logs"
    os.makedirs(log_dir, exist_ok=True)
    with open(os.path.join(log_dir, "notifications.jsonl"), "a", encoding="utf-8") as f:
        f.write(json.dumps(notification, ensure_ascii=False) + "\n")
```

## 21.7 Tool的错误处理与降级策略

```python
from functools import wraps
import time
import logging

logger = logging.getLogger(__name__)


def with_retry(max_retries: int = 3, delay: float = 1.0):
    """重试装饰器 — 指数退避"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_error = None
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_error = e
                    logger.warning(f"Tool {func.__name__} 第{attempt+1}次失败: {e}")
                    if attempt < max_retries - 1:
                        time.sleep(delay * (2 ** attempt))  # 指数退避
            return make_error("MAX_RETRIES", f"重试{max_retries}次仍失败", str(last_error))
        return wrapper
    return decorator


def with_fallback(fallback_func):
    """降级策略装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                logger.warning(f"Tool失败，启用降级: {e}")
                return fallback_func(*args, **kwargs)
        return wrapper
    return decorator


def search_fallback(query: str, **kwargs) -> str:
    """搜索降级方案：返回缓存结果"""
    return make_success({"source": "cache", "message": "在线搜索不可用，返回缓存", "data": []})


@tool
@with_retry(max_retries=3, delay=1.0)
@with_fallback(search_fallback)
def web_search(query: str, max_results: int = 5) -> str:
    """在线搜索。失败时自动降级到缓存。"""
    import requests
    resp = requests.get("https://api.search.example.com/search",
                        params={"q": query, "limit": max_results}, timeout=10)
    resp.raise_for_status()
    return make_success({"results": resp.json()})
```

## 21.9.1 Tool注册与管理

当Tool数量增多时，需要一个统一的管理器：

```python
class ToolRegistry:
    """Tool注册表 — 统一管理所有Tool"""
    
    def __init__(self):
        self.tools = {}
        self.categories = {}
    
    def register(self, tool, category: str = "general", 
                 requires_auth: bool = False):
        """注册Tool"""
        self.tools[tool.name] = {
            "tool": tool,
            "category": category,
            "requires_auth": requires_auth,
            "usage_count": 0,
            "error_count": 0,
            "avg_latency_ms": 0
        }
        if category not in self.categories:
            self.categories[category] = []
        self.categories[category].append(tool.name)
    
    def get_tools(self, category: str = None) -> list:
        """获取Tool列表"""
        if category:
            names = self.categories.get(category, [])
            return [self.tools[n]["tool"] for n in names if n in self.tools]
        return [t["tool"] for t in self.tools.values()]
    
    def get_tools_for_agent(self, role: str = "viewer") -> list:
        """根据角色获取可用Tool"""
        permission_map = {
            "viewer": ["general", "search", "read"],
            "editor": ["general", "search", "read", "write"],
            "admin": ["general", "search", "read", "write", "delete", "admin"]
        }
        allowed_cats = permission_map.get(role, ["general"])
        return [t["tool"] for name, t in self.tools.items()
                if t["category"] in allowed_cats]
    
    def stats(self) -> dict:
        """使用统计"""
        return {
            "total_tools": len(self.tools),
            "categories": {cat: len(names) for cat, names in self.categories.items()},
            "most_used": sorted(self.tools.items(), 
                               key=lambda x: x[1]["usage_count"], reverse=True)[:5]
        }
```

## 21.10 本章小结

| 概念 | 要点 |
| ------ | ------ |
| 单一职责 | 一个Tool做一件事 |
| 输入明确 | Pydantic Schema，类型清晰，有默认值 |
| 输出一致 | 统一的成功/失败格式 |
| 幂等性 | 多次调用不产生副作用 |
| @tool | 最简洁的定义方式 |
| StructuredTool | 更精细的控制 |
| BaseTool | 最大灵活性 |
| 错误处理 | 重试 + 降级 + 友好消息 |
| 测试 | 独立单元测试，Mock外部依赖 |
