# 第 19 章 设计模式精选

你在装修房子时，不会从零发明螺丝钉、水泥配方和电线标准——你会使用行业已经验证的成熟方案。软件开发也一样，前辈们总结了大量"设计模式"（Design Pattern），它们是解决常见软件设计问题的通用模板。

Java 世界把设计模式搞得非常隆重——23 种 GoF 模式、各种 UML 图、层层抽象的接口。但 Python 不一样：Python 的函数是一等公民、类定义是动态的、鸭子类型无处不在，很多在 Java 中需要大费周章的模式，在 Python 中只需要几行代码。本章精选最实用的 4 种模式，用 Python 的方式来实现——简洁、实用、不端着。

## 19.1 单例模式：全局只有一个实例

有些东西天然就应该是唯一的：数据库连接池、日志记录器、应用配置、线程池。单例模式确保一个类只有一个实例，所有人拿到的都是同一个。

### 方式 1：模块级变量（最 Pythonic）

```python
# config.py —— 整个文件就是一个天然的单例
_database_url = "postgresql://localhost/mydb"
_debug = False

def get_config():
    return {"database_url": _database_url, "debug": _debug}

def set_debug(value: bool):
    global _debug
    _debug = value
```

Python 的模块导入机制保证了模块只会被加载一次。所以模块级的变量天然就是单例。这是最简单、最 Pythonic 的方式。

### 方式 2：__new__ 方法

```python
class DatabasePool:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance

    def __init__(self):
        if self._initialized:
            return  # 避免重复初始化
        self._initialized = True
        self.connections = []
        print("数据库连接池初始化完成")


# 验证单例
pool1 = DatabasePool()  # 数据库连接池初始化完成
pool2 = DatabasePool()  # 不会再打印（不会重复初始化）
print(pool1 is pool2)   # True ← 同一个对象
```

### 方式 3：元类

```python
class SingletonMeta(type):
    _instances: dict[type, object] = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]


class Logger(metaclass=SingletonMeta):
    def __init__(self, level="INFO"):
        self.level = level
        print(f"Logger 初始化，级别: {level}")


log1 = Logger("DEBUG")   # Logger 初始化，级别: DEBUG
log2 = Logger("ERROR")   # 不会再初始化
print(log1 is log2)      # True
print(log2.level)        # DEBUG ← 第一次初始化后就不变了
```

小贴士：实际项目中，单例模式最简洁的实现就是模块级变量。`__new__` 和元类方式适合需要"类"的形式（比如需要继承或需要类型检查）的场景。

## 19.2 工厂模式：让专门的代码负责创建对象

当你创建对象的逻辑比较复杂（需要根据条件创建不同类型的对象），把创建逻辑集中到一个"工厂"里是好主意。

### 简单工厂

```python
from typing import Protocol


class Notification(Protocol):
    def send(self, message: str) -> None: ...


class EmailNotification:
    def __init__(self, to: str):
        self.to = to

    def send(self, message: str) -> None:
        print(f"发送邮件到 {self.to}: {message}")


class SMSNotification:
    def __init__(self, phone: str):
        self.phone = phone

    def send(self, message: str) -> None:
        print(f"发送短信到 {self.phone}: {message}")


class PushNotification:
    def __init__(self, device_token: str):
        self.device_token = device_token

    def send(self, message: str) -> None:
        print(f"推送通知到设备 {self.device_token}: {message}")


# 工厂函数
def create_notification(channel: str, **kwargs) -> Notification:
    factories = {
        "email": EmailNotification,
        "sms": SMSNotification,
        "push": PushNotification,
    }
    factory = factories.get(channel)
    if factory is None:
        raise ValueError(f"不支持的通知渠道: {channel}")
    return factory(**kwargs)


# 使用
notifier = create_notification("email", to="alice@example.com")
notifier.send("你的订单已发货")
# 发送邮件到 alice@example.com: 你的订单已发货
```

### 建造者模式（Builder）

当对象的构造参数很多、或者有可选配置时，建造者模式让创建过程更清晰：

```python
class HttpRequest:
    def __init__(self):
        self.method = "GET"
        self.url = ""
        self.headers = {}
        self.body = None
        self.timeout = 30

    def __repr__(self):
        return (f"HttpRequest(method={self.method}, url={self.url!r}, "
                f"headers={self.headers}, timeout={self.timeout})")


class HttpRequestBuilder:
    def __init__(self):
        self._request = HttpRequest()

    def method(self, method: str) -> "HttpRequestBuilder":
        self._request.method = method
        return self  # 返回自身以支持链式调用

    def url(self, url: str) -> "HttpRequestBuilder":
        self._request.url = url
        return self

    def header(self, key: str, value: str) -> "HttpRequestBuilder":
        self._request.headers[key] = value
        return self

    def body(self, body: str) -> "HttpRequestBuilder":
        self._request.body = body
        return self

    def timeout(self, seconds: int) -> "HttpRequestBuilder":
        self._request.timeout = seconds
        return self

    def build(self) -> HttpRequest:
        if not self._request.url:
            raise ValueError("URL 不能为空")
        return self._request


# 链式调用，清晰明了
request = (
    HttpRequestBuilder()
    .method("POST")
    .url("https://api.example.com/users")
    .header("Content-Type", "application/json")
    .header("Authorization", "Bearer token123")
    .body('{"name": "Alice"}')
    .timeout(10)
    .build()
)
print(request)
```

## 19.3 观察者模式：事件驱动的消息传递

观察者模式定义了一种一对多的依赖关系：当一个对象（主题）的状态改变时，所有依赖它的对象（观察者）都会收到通知。就像微信公众号——你关注了它，它发文章时你会收到推送。

### 事件总线实现

```python
from typing import Callable, Any
from collections import defaultdict


class EventBus:
    """简单的事件总线"""

    def __init__(self):
        self._listeners: dict[str, list[Callable]] = defaultdict(list)

    def on(self, event: str, callback: Callable) -> None:
        """注册事件监听器"""
        self._listeners[event].append(callback)

    def off(self, event: str, callback: Callable) -> None:
        """移除事件监听器"""
        self._listeners[event].remove(callback)

    def emit(self, event: str, *args, **kwargs) -> None:
        """触发事件"""
        for callback in self._listeners.get(event, []):
            callback(*args, **kwargs)


# 使用示例
bus = EventBus()

# 注册监听器
def on_user_created(username: str):
    print(f"[邮件] 欢迎新用户: {username}")

def on_user_created_log(username: str):
    print(f"[日志] 用户创建: {username}")

def on_user_deleted(username: str):
    print(f"[邮件] 再见, {username}")

bus.on("user.created", on_user_created)
bus.on("user.created", on_user_created_log)
bus.on("user.deleted", on_user_deleted)

# 触发事件
bus.emit("user.created", "Alice")
# [邮件] 欢迎新用户: Alice
# [日志] 用户创建: Alice

bus.emit("user.deleted", "Bob")
# [邮件] 再见, Bob
```

### 用装饰器简化注册

```python
class EventBus:
    def __init__(self):
        self._listeners = defaultdict(list)

    def on(self, event: str):
        """装饰器形式注册监听器"""
        def decorator(func):
            self._listeners[event].append(func)
            return func
        return decorator

    def emit(self, event: str, *args, **kwargs):
        for callback in self._listeners.get(event, []):
            callback(*args, **kwargs)


bus = EventBus()

@bus.on("order.placed")
def send_confirmation(order_id: int):
    print(f"发送订单确认: #{order_id}")

@bus.on("order.placed")
def update_inventory(order_id: int):
    print(f"更新库存: 订单 #{order_id}")

bus.emit("order.placed", 1001)
# 发送订单确认: #1001
# 更新库存: 订单 #1001
```

## 19.4 策略模式：用函数替代类

策略模式让你定义一系列算法，把它们封装起来，并使它们可以互相替换。在 Java 中需要一堆接口和实现类，在 Python 中——函数就够了。

### 传统方式（类实现）

```python
from typing import Protocol


class SortStrategy(Protocol):
    def sort(self, data: list) -> list: ...


class QuickSort:
    def sort(self, data: list) -> list:
        return sorted(data)  # 简化实现


class BubbleSort:
    def sort(self, data: list) -> list:
        arr = data.copy()
        n = len(arr)
        for i in range(n):
            for j in range(0, n - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
        return arr


class Sorter:
    def __init__(self, strategy: SortStrategy):
        self.strategy = strategy

    def sort(self, data: list) -> list:
        return self.strategy.sort(data)


# 使用
sorter = Sorter(QuickSort())
print(sorter.sort([3, 1, 4, 1, 5]))  # [1, 1, 3, 4, 5]

sorter.strategy = BubbleSort()
print(sorter.sort([3, 1, 4, 1, 5]))  # [1, 1, 3, 4, 5]
```

### Python 方式（函数即策略）

```python
# 策略就是函数，不需要类
def quick_sort(data: list) -> list:
    return sorted(data)

def bubble_sort(data: list) -> list:
    arr = data.copy()
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

def reverse_sort(data: list) -> list:
    return sorted(data, reverse=True)


# 使用：直接传函数
from typing import Callable

def sort_data(data: list, strategy: Callable[[list], list] = quick_sort) -> list:
    return strategy(data)

data = [3, 1, 4, 1, 5, 9]

print(sort_data(data))                    # [1, 1, 3, 4, 5, 9]
print(sort_data(data, bubble_sort))       # [1, 1, 3, 4, 5, 9]
print(sort_data(data, reverse_sort))      # [9, 5, 4, 3, 1, 1]
print(sort_data(data, lambda d: d[::-1])) # [9, 5, 1, 4, 1, 3] ← 甚至用 lambda
```

小贴士：Python 中函数是一等公民，很多在 Java 中需要"策略接口 + 多个实现类"的场景，在 Python 中直接传函数就行。简单就是最好的。

## 19.5 组合多种模式：实战案例

让我们设计一个简单的任务调度系统，综合运用多种设计模式：

```python
from typing import Callable
from collections import defaultdict
import time


# ---- 事件总线（观察者模式）----
class EventBus:
    def __init__(self):
        self._listeners = defaultdict(list)

    def on(self, event: str):
        def decorator(func):
            self._listeners[event].append(func)
            return func
        return decorator

    def emit(self, event: str, **kwargs):
        for callback in self._listeners.get(event, []):
            callback(**kwargs)


# ---- 单例的事件总线 ----
_bus = EventBus()


# ---- 任务类 ----
class Task:
    def __init__(self, name: str, action: Callable):
        self.name = name
        self.action = action

    def execute(self):
        _bus.emit("task.started", task_name=self.name)
        start = time.perf_counter()
        result = self.action()
        elapsed = time.perf_counter() - start
        _bus.emit("task.completed", task_name=self.name, elapsed=elapsed, result=result)
        return result


# ---- 调度器（工厂 + 策略）----
class Scheduler:
    _registry: dict[str, Callable] = {}

    @classmethod
    def register(cls, name: str):
        """注册任务工厂"""
        def decorator(func):
            cls._registry[name] = func
            return func
        return decorator

    @classmethod
    def create_task(cls, name: str, **kwargs) -> Task:
        factory = cls._registry.get(name)
        if factory is None:
            raise ValueError(f"未知任务: {name}")
        return factory(**kwargs)


# ---- 注册任务 ----
@Scheduler.register("backup")
def _create_backup_task(target_dir: str = "/tmp", **_):
    def action():
        print(f"备份到 {target_dir}...")
        return f"已备份到 {target_dir}"
    return Task("backup", action)


@Scheduler.register("report")
def _create_report_task(report_type: str = "daily", **_):
    def action():
        print(f"生成 {report_type} 报告...")
        return f"{report_type} 报告已生成"
    return Task("report", action)


# ---- 注册事件监听 ----
@_bus.on("task.started")
def _log_start(task_name: str):
    print(f"[开始] {task_name}")

@_bus.on("task.completed")
def _log_complete(task_name: str, elapsed: float, result: str):
    print(f"[完成] {task_name} 耗时 {elapsed:.4f}s，结果: {result}")


# ---- 使用 ----
task1 = Scheduler.create_task("backup", target_dir="/data/backup")
task1.execute()
# [开始] backup
# 备份到 /data/backup...
# [完成] backup 耗时 0.0001s，结果: 已备份到 /data/backup

task2 = Scheduler.create_task("report", report_type="weekly")
task2.execute()
# [开始] report
# 生成 weekly 报告...
# [完成] report 耗时 0.0001s，结果: weekly 报告已生成
```

## 19.7 小结

Python 风格的设计模式核心理念：__简单优于复杂__。

- __单例__：模块级变量是最简方案，`__new__` 和元类方式适合需要"类"形式的场景
- __工厂__：用字典映射 + 函数/类实现，比 Java 的工厂接口简洁得多
- __建造者__：链式调用让复杂对象的构建过程清晰可读
- __观察者__：事件总线实现解耦，用装饰器注册监听器
- __策略__：Python 函数是一等公民，直接传函数即可替代"策略接口 + 实现类"

```python
import json
import csv
import io

def to_json(data: list[dict]) -> str:
    return json.dumps(data, ensure_ascii=False, indent=2)

def to_csv(data: list[dict]) -> str:
    if not data:
        return ""
    output = io.StringIO()
    writer = csv.DictWriter(output, fieldnames=data[0].keys())
    writer.writeheader()
    writer.writerows(data)
    return output.getvalue()

def to_xml(data: list[dict]) -> str:
    lines = ["<?xml version='1.0'?>", "<records>"]
    for record in data:
        lines.append("  <record>")
        for key, value in record.items():
            lines.append(f"    <{key}>{value}</{key}>")
        lines.append("  </record>")
    lines.append("</records>")
    return "\n".join(lines)

def format_data(data, fmt="json"):
    formatters = {"json": to_json, "csv": to_csv, "xml": to_xml}
    return formatters[fmt](data)
```

```python
class QueryBuilder:
    def __init__(self):
        self._select = "*"
        self._from = ""
        self._where_clauses = []
        self._order = ""
        self._limit = None

    def select(self, *columns):
        self._select = ", ".join(columns)
        return self

    def from_table(self, table: str):
        self._from = table
        return self

    def where(self, condition: str):
        self._where_clauses.append(condition)
        return self

    def order_by(self, column: str, direction: str = "ASC"):
        self._order = f"{column} {direction}"
        return self

    def limit(self, n: int):
        self._limit = n
        return self

    def build(self) -> str:
        query = f"SELECT {self._select} FROM {self._from}"
        if self._where_clauses:
            query += " WHERE " + " AND ".join(self._where_clauses)
        if self._order:
            query += f" ORDER BY {self._order}"
        if self._limit:
            query += f" LIMIT {self._limit}"
        return query

sql = (
    QueryBuilder()
    .select("name", "age")
    .from_table("users")
    .where("age > 18")
    .where("status = 'active'")
    .order_by("name")
    .limit(10)
    .build()
)
print(sql)
# SELECT name, age FROM users WHERE age > 18 AND status = 'active' ORDER BY name ASC LIMIT 10
```

## 19.10 常见错误与避坑

__1. 单例在多线程中不安全__

```python
# 两个线程同时创建实例，可能产生两个不同的对象
import threading

class UnsafeSingleton:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            # 这里可能被两个线程同时进入
            cls._instance = super().__new__(cls)
        return cls._instance

# 解决：加锁
class ThreadSafeSingleton:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls):
        with cls._lock:
            if cls._instance is None:
                cls._instance = super().__new__(cls)
        return cls._instance
```

__2. 观察者模式中忘记取消注册__

```python
# 观察者被销毁了但没有取消注册 → 内存泄漏
class ShortLivedObserver:
    def on_event(self):
        print("处理事件")

observer = ShortLivedObserver()
bus.on("event", observer.on_event)
# ... 后来 observer 不再需要了
del observer  # 但 bus 还持有对 on_event 的引用！

# 解决：记得取消注册
bus.off("event", observer.on_event)
```
