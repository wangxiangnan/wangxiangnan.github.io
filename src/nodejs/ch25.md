```typescript
            type: 'http',

            bearerFormat: 'JWT',
          },
        },
      },
    },
  });
```

## 25.3 认证系统：JWT双Token + RBAC

### 25.3.1 JWT服务

```typescript
// src/modules/auth/auth.service.ts
import * as jose from 'jose';
import { env } from '../../config/env.js';
import { prisma } from '../../shared/database.js';
import { redis } from '../../shared/cache.js';
import { AppError } from '../../shared/errors.js';
import bcrypt from 'bcrypt';

interface TokenPayload {
  userId: number;
  email: string;
  role: string;
  teamId?: number;
}

interface TokenPair {
  accessToken: string;
  refreshToken: string;
  expiresIn: number;
}

class AuthService {
  private accessSecret: Uint8Array;
  private refreshSecret: Uint8Array;

  constructor() {
    this.accessSecret = new TextEncoder().encode(env.JWT_ACCESS_SECRET);
    this.refreshSecret = new TextEncoder().encode(env.JWT_REFRESH_SECRET);
  }

  async register(data: { email: string; password: string; name: string }): Promise<TokenPair> {
    // 检查邮箱是否已注册
    const existing = await prisma.user.findUnique({ where: { email: data.email } });
    if (existing) {
      throw new AppError('Email already registered', 409);
    }

    // 密码哈希
    const hashedPassword = await bcrypt.hash(data.password, 12);
    
    // 创建用户
    const user = await prisma.user.create({
      data: {
        email: data.email,
        password: hashedPassword,
        name: data.name,
        role: 'USER',
      },
    });

    // 生成Token对
    return this.generateTokenPair({
      userId: user.id,
      email: user.email,
      role: user.role,
    });
  }

  async login(email: string, password: string): Promise<TokenPair> {
    const user = await prisma.user.findUnique({ 
      where: { email },
      select: { id: true, email: true, password: true, role: true, isActive: true },
    });

    if (!user || !user.isActive) {
      throw new AppError('Invalid credentials', 401);
    }

    const isPasswordValid = await bcrypt.compare(password, user.password);
    if (!isPasswordValid) {
      throw new AppError('Invalid credentials', 401);
    }

    // 更新最后登录时间
    await prisma.user.update({
      where: { id: user.id },
      data: { lastLoginAt: new Date() },
    });

    return this.generateTokenPair({
      userId: user.id,
      email: user.email,
      role: user.role,
    });
  }

  async refreshTokens(refreshToken: string): Promise<TokenPair> {
    // 验证refresh token
    const payload = await this.verifyRefreshToken(refreshToken);
    
    // 检查token是否在黑名单中（已注销）
    const isBlacklisted = await redis.get(`blacklist:${refreshToken}`);
    if (isBlacklisted) {
      throw new AppError('Token has been revoked', 401);
    }

    // 将旧的refresh token加入黑名单
    await redis.set(`blacklist:${refreshToken}`, '1', 'EX', 7 * 24 * 3600);

    // 确认用户仍然存在且活跃
    const user = await prisma.user.findUnique({
      where: { id: payload.userId },
      select: { id: true, email: true, role: true, isActive: true },
    });

    if (!user || !user.isActive) {
      throw new AppError('User not found or inactive', 401);
    }

    // 生成新的Token对
    return this.generateTokenPair({
      userId: user.id,
      email: user.email,
      role: user.role,
    });
  }

  async logout(refreshToken: string): Promise<void> {
    // 将refresh token加入黑名单
    await redis.set(`blacklist:${refreshToken}`, '1', 'EX', 7 * 24 * 3600);
  }

  // 生成Token对
  private async generateTokenPair(payload: TokenPayload): Promise<TokenPair> {
    const accessToken = await new jose.SignJWT({ ...payload })
      .setProtectedHeader({ alg: 'HS256' })
      .setIssuedAt()
      .setExpirationTime(env.JWT_ACCESS_EXPIRES)
      .setSubject(String(payload.userId))
      .sign(this.accessSecret);

    const refreshToken = await new jose.SignJWT({ userId: payload.userId })
      .setProtectedHeader({ alg: 'HS256' })
      .setIssuedAt()
      .setExpirationTime(env.JWT_REFRESH_EXPIRES)
      .setSubject(String(payload.userId))
      .sign(this.refreshSecret);

    return {
      accessToken,
      refreshToken,
      expiresIn: 900, // 15分钟(秒)
    };
  }

  async verifyAccessToken(token: string): Promise<TokenPayload> {
    try {
      const { payload } = await jose.jwtVerify(token, this.accessSecret);
      return payload as unknown as TokenPayload;
    } catch (error) {
      throw new AppError('Invalid or expired access token', 401);
    }
  }

  private async verifyRefreshToken(token: string): Promise<{ userId: number }> {
    try {
      const { payload } = await jose.jwtVerify(token, this.refreshSecret);
      return { userId: Number(payload.sub) };
    } catch (error) {
      throw new AppError('Invalid or expired refresh token', 401);
    }
  }
}

export const authService = new AuthService();
```

### 25.3.2 RBAC权限控制

```typescript
// src/middleware/rbac.middleware.ts
import { FastifyRequest, FastifyReply } from 'fastify';
import { AppError } from '../shared/errors.js';

// 权限定义
type Permission = 
  | 'user:read' | 'user:write' | 'user:delete'
  | 'team:read' | 'team:write' | 'team:delete' | 'team:manage'
  | 'project:read' | 'project:write' | 'project:delete'
  | 'task:read' | 'task:write' | 'task:delete'
  | 'admin:access' | 'admin:users' | 'admin:settings';

type Role = 'USER' | 'ADMIN' | 'TEAM_OWNER' | 'TEAM_MEMBER' | 'VIEWER';

// 角色-权限映射
const rolePermissions: Record<Role, Permission[]> = {
  ADMIN: [
    'user:read', 'user:write', 'user:delete',
    'team:read', 'team:write', 'team:delete', 'team:manage',
    'project:read', 'project:write', 'project:delete',
    'task:read', 'task:write', 'task:delete',
    'admin:access', 'admin:users', 'admin:settings',
  ],
  TEAM_OWNER: [
    'user:read',
    'team:read', 'team:write', 'team:manage',
    'project:read', 'project:write', 'project:delete',
    'task:read', 'task:write', 'task:delete',
  ],
  TEAM_MEMBER: [
    'user:read',
    'team:read',
    'project:read', 'project:write',
    'task:read', 'task:write',
  ],
  VIEWER: [
    'user:read',
    'team:read',
    'project:read',
    'task:read',
  ],
  USER: [
    'user:read', 'user:write',
    'team:read',
    'project:read',
    'task:read',
  ],
};

function hasPermission(role: Role, permission: Permission): boolean {
  return rolePermissions[role]?.includes(permission) ?? false;
}

// Fastify装饰器：权限检查
function requirePermission(...permissions: Permission[]) {
  return async (request: FastifyRequest, reply: FastifyReply) => {
    const user = (request as any).user;
    
    if (!user) {
      throw new AppError('Authentication required', 401);
    }

    const userRole = user.role as Role;
    const hasAllPermissions = permissions.every(p => hasPermission(userRole, p));
    
    if (!hasAllPermissions) {
      throw new AppError(
        `Insufficient permissions. Required: ${permissions.join(', ')}`,
        403
      );
    }
  };
}

export { requirePermission, hasPermission, Permission, Role };
```

## 25.4 多租户数据隔离

```typescript
// src/middleware/tenant.middleware.ts
import { FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../shared/database.js';
import { AppError } from '../shared/errors.js';

// 多租户策略：通过team_id实现行级数据隔离
// 所有查询都会自动注入tenant条件

interface TenantContext {
  teamId: number;
  userId: number;
  role: string;
}

// 中间件：提取和验证租户上下文
async function tenantMiddleware(request: FastifyRequest, reply: FastifyReply): Promise<void> {
  const user = (request as any).user;
  if (!user) return; // 公开路由不需要租户上下文
  
  // 从header或query中获取team_id
  const teamId = request.headers['x-team-id'] || (request.query as any)?.teamId;
  
  if (!teamId) return; // 有些路由不需要团队上下文

  // 验证用户确实属于这个团队
  const membership = await prisma.teamMember.findUnique({
    where: {
      userId_teamId: { userId: user.userId, teamId: Number(teamId) },
    },
  });

  if (!membership) {
    throw new AppError('You are not a member of this team', 403);
  }

  // 注入租户上下文
  (request as any).tenant = {
    teamId: Number(teamId),
    userId: user.userId,
    role: membership.role,
  } satisfies TenantContext;
}

// Prisma中间件：自动注入租户过滤
// 使用Prisma的$extends实现
function createTenantPrisma(teamId: number) {
  return prisma.$extends({
    query: {
      project: {
        async findMany({ args, query }) {
          args.where = { ...args.where, teamId };
          return query(args);
        },
        async findFirst({ args, query }) {
          args.where = { ...args.where, teamId };
          return query(args);
        },
        async create({ args, query }) {
          args.data = { ...args.data, teamId } as any;
          return query(args);
        },
      },
      task: {
        async findMany({ args, query }) {
          // 通过关联的project来隔离
          args.where = { 
            ...args.where,
            project: { teamId },
          };
          return query(args);
        },
      },
    },
  });
}

export { tenantMiddleware, createTenantPrisma, TenantContext };
```

```typescript
  constructor() {
    this.client = new S3Client({
      endpoint: env.S3_ENDPOINT,
      region: 'auto',
      credentials: {
        accessKeyId: env.S3_ACCESS_KEY || '',
        secretAccessKey: env.S3_SECRET_KEY || '',
      },
    });
    this.bucket = env.S3_BUCKET || 'uploads';
  }

  async upload(file: NodeJS.ReadableStream, metadata: FileMetadata): Promise<UploadResult> {
    const id = randomUUID();
    const ext = extname(metadata.originalName);
    const key = `uploads/${id}${ext}`;

    // 收集stream到Buffer
    const chunks: Buffer[] = [];
    for await (const chunk of file) {
      chunks.push(Buffer.from(chunk));
    }
    const buffer = Buffer.concat(chunks);

    await this.client.send(new PutObjectCommand({
      Bucket: this.bucket,
      Key: key,
      Body: buffer,
      ContentType: metadata.mimeType,
      Metadata: {
        originalName: metadata.originalName,
      },
    }));

    return {
      id,
      filename: key,
      originalName: metadata.originalName,
      mimeType: metadata.mimeType,
      size: buffer.length,
      url: await this.getUrl(key),
      storage: 's3',
    };
  }

  async delete(key: string): Promise<void> {
    await this.client.send(new DeleteObjectCommand({
      Bucket: this.bucket,
      Key: key,
    }));
  }

  async getUrl(key: string): Promise<string> {
    // 生成预签名URL（1小时有效）
    const command = new GetObjectCommand({ Bucket: this.bucket, Key: key });
    return getSignedUrl(this.client, command, { expiresIn: 3600 });
  }
}

// 工厂函数：根据配置选择存储方式
function createStorageProvider(): StorageProvider {
  if (env.S3_ENDPOINT && env.S3_ACCESS_KEY) {
    return new S3Storage();
  }
  return new LocalStorage(env.UPLOAD_DIR);
}

export const storage = createStorageProvider();
export { UploadResult, StorageProvider };
```

```typescript
// src/modules/notification/notification.gateway.ts
import { FastifyInstance } from 'fastify';
import { WebSocket } from 'ws';
import { redis } from '../../shared/cache.js';
import { authService } from '../auth/auth.service.js';
import { logger } from '../../shared/logger.js';

interface ConnectedClient {
  ws: WebSocket;
  userId: number;
  teamId?: number;
  connectedAt: Date;
}

class NotificationGateway {
  private clients: Map<number, Set<ConnectedClient>> = new Map();

  register(app: FastifyInstance): void {
    app.get('/ws/notifications', { websocket: true }, (socket, request) => {
      this.handleConnection(socket, request);
    });

    // SSE替代方案（对不支持WebSocket的环境）
    app.get('/api/notifications/stream', async (request, reply) => {
      const user = (request as any).user;
      if (!user) {
        return reply.status(401).send({ error: 'Unauthorized' });
      }

      reply.raw.writeHead(200, {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        'Connection': 'keep-alive',
        'X-Accel-Buffering': 'no',
      });

      // 发送心跳
      const heartbeat = setInterval(() => {
        reply.raw.write(': heartbeat\n\n');
      }, 30000);

      // 订阅Redis通道接收通知
      const subscriber = redis.duplicate();
      await subscriber.subscribe(`notifications:${user.userId}`);
      
      subscriber.on('message', (channel, message) => {
        reply.raw.write(`data: ${message}\n\n`);
      });

      // 连接关闭时清理
      request.raw.on('close', () => {
        clearInterval(heartbeat);
        subscriber.unsubscribe();
        subscriber.quit();
      });
    });
  }

  private async handleConnection(socket: WebSocket, request: any): Promise<void> {
    // 从query参数获取token进行认证
    const url = new URL(request.url, 'http://localhost');
    const token = url.searchParams.get('token');
    
    if (!token) {
      socket.close(4001, 'Authentication required');
      return;
    }

    try {
      const payload = await authService.verifyAccessToken(token);
      const client: ConnectedClient = {
        ws: socket,
        userId: payload.userId,
        connectedAt: new Date(),
      };

      // 注册客户端
      if (!this.clients.has(payload.userId)) {
        this.clients.set(payload.userId, new Set());
      }
      this.clients.get(payload.userId)!.add(client);

      logger.info(`WebSocket connected: user ${payload.userId}`);

      // 消息处理
      socket.on('message', (data) => {
        try {
          const message = JSON.parse(data.toString());
          this.handleMessage(client, message);
        } catch (err) {
          socket.send(JSON.stringify({ type: 'error', message: 'Invalid message format' }));
        }
      });

      // 断开连接
      socket.on('close', () => {
        this.clients.get(payload.userId)?.delete(client);
        if (this.clients.get(payload.userId)?.size === 0) {
          this.clients.delete(payload.userId);
        }
        logger.info(`WebSocket disconnected: user ${payload.userId}`);
      });

      // 发送连接成功消息
      socket.send(JSON.stringify({ type: 'connected', userId: payload.userId }));
    } catch (error) {
      socket.close(4001, 'Invalid token');
    }
  }

  private handleMessage(client: ConnectedClient, message: any): void {
    switch (message.type) {
      case 'ping':
        client.ws.send(JSON.stringify({ type: 'pong' }));
        break;
      case 'subscribe':
        // 订阅特定频道
        client.teamId = message.teamId;
        break;
    }
  }

  // 发送通知给特定用户
  async sendToUser(userId: number, notification: object): Promise<void> {
    const clients = this.clients.get(userId);
    const message = JSON.stringify(notification);
    
    if (clients && clients.size > 0) {
      // 用户在线，直接推送
      for (const client of clients) {
        if (client.ws.readyState === WebSocket.OPEN) {
          client.ws.send(message);
        }
      }
    }
    
    // 同时发布到Redis（支持多实例部署）
    await redis.publish(`notifications:${userId}`, message);
  }

  // 广播给团队所有在线成员
  async broadcastToTeam(teamId: number, notification: object, excludeUserId?: number): Promise<void> {
    const message = JSON.stringify(notification);
    
    for (const [userId, clients] of this.clients) {
      if (userId === excludeUserId) continue;
      
      for (const client of clients) {
        if (client.teamId === teamId && client.ws.readyState === WebSocket.OPEN) {
          client.ws.send(message);
        }
      }
    }
  }

  getOnlineUsers(): number[] {
    return [...this.clients.keys()];
  }
}

export const notificationGateway = new NotificationGateway();
```

```typescript
    }),

      subject: d.title as string,
      html: `<p>${d.body}</p>`,
    }),
  };

  const tmpl = templates[template];
  if (!tmpl) throw new Error(`Unknown template: ${template}`);
```

```typescript
// src/middleware/error-handler.ts
import { FastifyError, FastifyReply, FastifyRequest } from 'fastify';
import { Sentry } from '../shared/sentry.js';
import { logger } from '../shared/logger.js';
import { ZodError } from 'zod';

// 自定义错误基类
export class AppError extends Error {
  statusCode: number;
  isOperational: boolean;

  constructor(message: string, statusCode: number = 500, isOperational: boolean = true) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = isOperational;
    Error.captureStackTrace(this, this.constructor);
  }
}

// 全局错误处理器
export function errorHandler(
  error: FastifyError | AppError | ZodError | Error,
  request: FastifyRequest,
  reply: FastifyReply,
): void {
  // Zod验证错误
  if (error instanceof ZodError) {
    reply.status(400).send({
      error: 'Validation Error',
      details: error.errors.map(e => ({
        field: e.path.join('.'),
        message: e.message,
      })),
    });
    return;
  }

  // 自定义业务错误
  if (error instanceof AppError) {
    reply.status(error.statusCode).send({
      error: error.message,
    });
    return;
  }

  // Fastify验证错误
  if ('validation' in error && error.validation) {
    reply.status(400).send({
      error: 'Validation Error',
      details: error.validation,
    });
    return;
  }

  // 未知错误 - 记录日志并上报Sentry
  logger.error({ err: error, requestId: request.id }, 'Unhandled error');
  Sentry.captureException(error, {
    extra: {
      requestId: request.id,
      url: request.url,
      method: request.method,
    },
  });

  // 生产环境不暴露内部错误详情
  const message = process.env.NODE_ENV === 'production' 
    ? 'Internal Server Error' 
    : error.message;

  reply.status(500).send({ error: message });
}
```
