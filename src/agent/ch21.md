# 第21章 RAG进阶 — 从能用用到用得好

## 21.1 开篇：基础RAG的常见痛点

在前面的章节中，你已经实现了基础 RAG 管线：文档 → 分割 → 向量化 → 存储 → 检索 → 生成。它能跑，但你会发现几个问题：

1. **检索不准**：问"Python装饰器怎么用"，返回的是"Python设计模式"——语义近但不精确
2. **信息丢失**：文档被切成小块后，跨段落上下文丢失
3. **回答不全**：有些问题需要综合多文档信息，top-K只返回部分

这一章，我们把 RAG 从"能用"升级到"好用"。就像数据库优化——不只是查到数据，还要查得准、查得全、查得快。

## 21.2 高质量文本分割策略

### 21.2.1 语义感知分割

基础分割像用尺子量布——机械地每 N 厘米切一刀。语义感知分割像裁剪衣服——沿着自然"接缝"（段落、章节、函数）来切。

```python
from langchain_text_splitters import (
    RecursiveCharacterTextSplitter,
    MarkdownHeaderTextSplitter,
    PythonCodeTextSplitter
)


# 1. Markdown文档：按标题层级分割
def split_markdown_by_headers(text: str) -> list:
    """按Markdown标题分割，保留层级结构"""
    splitter = MarkdownHeaderTextSplitter(
        headers_to_split_on=[("#", "H1"), ("##", "H2"), ("##", "H3")],
        strip_headers=False
    )
    return splitter.split_text(text)


# 2. Python代码：按函数/类分割
def split_python_code(text: str) -> list:
    """按代码结构分割"""
    return PythonCodeTextSplitter(chunk_size=1000, chunk_overlap=100).split_text(text)


# 3. 通用文档：递归分割 + 语义边界
def split_semantic(text: str) -> list:
    """语义感知的递归分割"""
    return RecursiveCharacterTextSplitter(
        chunk_size=500, chunk_overlap=50,
        separators=["\n\n", "\n", "。", ". ", "！", "？", "；", "，", " ", ""]
    ).split_text(text)


# 4. 段落聚合：小块合并为有意义的单元
def aggregate_paragraphs(chunks: list, max_length: int = 1000) -> list:
    """将过小的chunk合并为更大的语义单元"""
    aggregated, current = [], ""
    for chunk in chunks:
        if len(current) + len(chunk) < max_length:
            current += "\n" + chunk if current else chunk
        else:
            if current:
                aggregated.append(current)
            current = chunk
    if current:
        aggregated.append(current)
    return aggregated
```

### 21.2.2 父子文档策略

核心思想：小块（200字）做精细检索，大块（1000字）给LLM完整上下文。

```python
from langchain_core.documents import Document
from langchain_text_splitters import RecursiveCharacterTextSplitter
import uuid


class ParentChildRetriever:
    """父子文档检索器：小块索引，大块检索"""
    
    def __init__(self, vectorstore, parent_size=1000, child_size=200):
        self.vectorstore = vectorstore
        self.parent_splitter = RecursiveCharacterTextSplitter(
            chunk_size=parent_size, chunk_overlap=100)
        self.child_splitter = RecursiveCharacterTextSplitter(
            chunk_size=child_size, chunk_overlap=30)
        self.parent_store = {}  # parent_id -> parent_doc
    
    def add_documents(self, documents: list):
        """添加文档：创建父子关系"""
        for doc in documents:
            for parent in self.parent_splitter.split_documents([doc]):
                parent_id = str(uuid.uuid4())
                parent.metadata["doc_id"] = parent_id
                self.parent_store[parent_id] = parent
                
                children = self.child_splitter.split_documents([parent])
                for child in children:
                    child.metadata["parent_id"] = parent_id
                self.vectorstore.add_documents(children)
        print(f"已索引 {len(self.parent_store)} 个父文档")
    
    def retrieve(self, query: str, k: int = 5) -> list:
        """小块搜索，返回对应大块"""
        child_results = self.vectorstore.similarity_search(query, k=k * 2)
        seen, parent_docs = set(), []
        for child in child_results:
            pid = child.metadata.get("parent_id")
            if pid and pid not in seen:
                seen.add(pid)
                parent = self.parent_store.get(pid)
                if parent:
                    parent_docs.append(parent)
            if len(parent_docs) >= k:
                break
        return parent_docs
```

        })
    return chunks


def _detect_section(text: str) -> str:
    for line in text.split("\n")[:5]:
        if line.startswith("#"):
            return line.strip("# ").strip()
    return "unknown"

## 21.3 混合检索（Hybrid Search）

向量搜索擅长语义匹配，BM25擅长精确匹配。**混合检索 = 各取所长。**

```python
from rank_bm25 import BM25Okapi
from langchain_openai import OpenAIEmbeddings
import numpy as np


class HybridSearchRetriever:
    """混合检索器：向量 + BM25 + RRF融合"""
    
    def __init__(self, documents: list, embeddings_model=None):
        self.documents = documents
        self.embeddings_model = embeddings_model or OpenAIEmbeddings()
        self.bm25 = BM25Okapi([d.page_content.lower().split() for d in documents])
        self.doc_embeddings = self.embeddings_model.embed_documents(
            [d.page_content for d in documents])
    
    def search(self, query: str, k: int = 5) -> list:
        vector_results = self._vector_search(query, k*2)
        bm25_results = self._bm25_search(query, k*2)
        return self._rrf_fusion(vector_results, bm25_results, k)
    
    def _vector_search(self, query: str, k: int):
        q_emb = self.embeddings_model.embed_query(query)
        sims = [(i, np.dot(q_emb, d) / (np.linalg.norm(q_emb) * np.linalg.norm(d)))
                for i, d in enumerate(self.doc_embeddings)]
        return sorted(sims, key=lambda x: x[1], reverse=True)[:k]
    
    def _bm25_search(self, query: str, k: int):
        scores = self.bm25.get_scores(query.lower().split())
        return sorted(enumerate(scores), key=lambda x: x[1], reverse=True)[:k]
    
    def _rrf_fusion(self, *lists, k=60, top_k=5):
        """Reciprocal Rank Fusion: score(d) = sum(1/(k+rank))"""
        scores = {}
        for results in lists:
            for rank, (idx, _) in enumerate(results):
                scores[idx] = scores.get(idx, 0) + 1.0 / (k + rank + 1)
        sorted_docs = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [{"document": self.documents[idx], "rrf_score": s, "rank": r+1}
                for r, (idx, s) in enumerate(sorted_docs[:top_k])]
```

> 💡 **RRF直觉**：一个文档如果在多个排序列表中都靠前，大概率最相关。就像多个搜索引擎都排第一页的文章。

## 21.4 Reranking检索重排序

```python
class CrossEncoderReranker:
    """Cross-Encoder重排序：同时编码query和document，更精确"""
    
    def __init__(self, model_name="cross-encoder/ms-marco-MiniLM-L-6-v2"):
        try:
            from sentence_transformers import CrossEncoder
            self.model = CrossEncoder(model_name)
            self.available = True
        except ImportError:
            self.available = False
    
    def rerank(self, query: str, documents: list, top_k: int = 5) -> list:
        if not self.available:
            return documents[:top_k]
        pairs = [(query, doc.page_content) for doc in documents]
        scores = self.model.predict(pairs)
        scored = sorted(zip(documents, scores), key=lambda x: x[1], reverse=True)
        results = []
        for doc, score in scored[:top_k]:
            doc.metadata["rerank_score"] = float(score)
            results.append(doc)
        return results


def llm_rerank(query: str, documents: list, llm=None, top_k: int = 5) -> list:
    """用LLM重排序（无需额外模型）"""
    from langchain_openai import ChatOpenAI
    llm = llm or ChatOpenAI(model="gpt-4o-mini", temperature=0)
    scored = []
    for doc in documents:
        resp = llm.invoke(f"评估相关性(0-10):\n查询:{query}\n文档:{doc.page_content[:500]}\n分数:")
        try:
            score = max(0, min(10, int(resp.content.strip())))
        except ValueError:
            score = 5
        doc.metadata["rerank_score"] = score
        scored.append(doc)
    scored.sort(key=lambda d: d.metadata["rerank_score"], reverse=True)
    return scored[:top_k]
```

```python
class MultiQueryRetriever:
    """生成多个不同角度的查询，取并集"""
    
    def __init__(self, retriever, llm=None):
        from langchain_openai import ChatOpenAI
        self.retriever = retriever
        self.llm = llm or ChatOpenAI(model="gpt-4o", temperature=0.7)
    
    def retrieve(self, query: str, k: int = 5) -> list:
        variants = self._generate_variants(query)
        print(f"生成{len(variants)}个变体查询")
        
        all_docs, seen = [], set()
        for q in variants:
            for doc in self.retriever.invoke(q):
                h = hash(doc.page_content[:200])
                if h not in seen:
                    seen.add(h)
                    doc.metadata["matched_query"] = q
                    all_docs.append(doc)
        return all_docs[:k]
    
    def _generate_variants(self, query: str) -> list:
        resp = self.llm.invoke(f"将查询改写为3个不同角度的搜索查询(每行一个):\n{query}")
        variants = [q.strip() for q in resp.content.split("\n") if q.strip()]
        return [query] + variants[:3]
```

## 21.6 Self-RAG：Agent自主决定是否需要检索

```python
class SelfRAGAgent:
    """Self-RAG：自主决定是否检索"""
    
    def __init__(self, retriever, llm=None):
        from langchain_openai import ChatOpenAI
        self.retriever = retriever
        self.llm = llm or ChatOpenAI(model="gpt-4o", temperature=0)
    
    def answer(self, query: str) -> dict:
        if not self._should_retrieve(query):
            return {"answer": self.llm.invoke(query).content, "used_retrieval": False}
        
        docs = self.retriever.invoke(query)
        context = "\n\n".join([d.page_content for d in docs])
        
        from langchain_core.prompts import ChatPromptTemplate
        prompt = ChatPromptTemplate.from_messages([
            ("system", "基于上下文回答问题。不足则说明。\n上下文：{context}"),
            ("human", "{query}")
        ])
        answer = (prompt | self.llm).invoke({"query": query, "context": context}).content
        return {"answer": answer, "used_retrieval": True,
                "sources": [d.metadata.get("source","") for d in docs]}
    
    def _should_retrieve(self, query: str) -> bool:
        resp = self.llm.invoke(f"是否需要检索文档来回答:{query}\n只回答yes或no:")
        return "yes" in resp.content.lower()
```

## 21.6.1 Self-RAG完整实现

让我们深入Self-RAG的完整流程：

```python
from pydantic import BaseModel, Field
from enum import Enum


class RetrievalDecision(str, Enum):
    RETRIEVE = "retrieve"
    NO_RETRIEVE = "no_retrieve"


class AdvancedSelfRAG:
    """高级Self-RAG"""
    
    def __init__(self, retriever, llm=None):
        from langchain_openai import ChatOpenAI
        self.retriever = retriever
        self.llm = llm or ChatOpenAI(model="gpt-4o", temperature=0)
    
    def answer(self, query: str) -> dict:
        """完整Self-RAG流程"""
        # 1. 评估是否需要检索
        need_retrieval = self._assess_need(query)
        if not need_retrieval:
            return {"answer": self.llm.invoke(query).content, "retrieved": False}
        
        # 2. 检索
        docs = self.retriever.invoke(query)
        context = "\n\n".join([d.page_content for d in docs])
        
        # 3. 生成回答
        from langchain_core.prompts import ChatPromptTemplate
        prompt = ChatPromptTemplate.from_messages([
            ("system", "基于上下文回答。不足则说明。\n上下文：{context}"),
            ("human", "{query}")
        ])
        answer = (prompt | self.llm).invoke({"query": query, "context": context}).content
        
        # 4. 质量评估
        quality = self._assess_quality(query, answer, context)
        
        # 5. 质量不够则改进
        if quality < 0.6:
            answer = self._improve(query, answer, context)
        
        return {
            "answer": answer, "retrieved": True,
            "quality": quality, "docs_used": len(docs)
        }
    
    def _assess_need(self, query):
        resp = self.llm.invoke(f"是否需要检索文档回答：{query}\n(yes/no):")
        return "yes" in resp.content.lower()
    
    def _assess_quality(self, query, answer, context):
        resp = self.llm.invoke(
            f"评估回答质量(0-10)：\nQ:{query}\nA:{answer}\nCtx:{context[:1000]}\n分数："
        )
        try:
            return max(0, min(10, int(resp.content.strip()))) / 10
        except:
            return 0.5
    
    def _improve(self, query, answer, context):
        resp = self.llm.invoke(
            f"改进以下回答：\nQ:{query}\n原答:{answer}\n上下文:{context[:1500]}\n改进："
        )
        return resp.content
```

## 21.6.2 RAG评估指标

量化RAG管线表现的关键指标：

```python
class RAGMetrics:
    """RAG评估指标"""
    
    def __init__(self):
        self.results = []
    
    def add_result(self, query, retrieved_docs, answer, ground_truth=None):
        self.results.append({
            "query": query, "retrieved": retrieved_docs,
            "answer": answer, "ground_truth": ground_truth
        })
    
    def precision_at_k(self, k=5) -> float:
        """Precision@K: 前K个结果中相关的比例"""
        if not self.results:
            return 0
        precisions = []
        for r in self.results:
            if r["ground_truth"]:
                relevant = sum(1 for d in r["retrieved"][:k]
                             if self._is_relevant(d, r["ground_truth"]))
                precisions.append(relevant / k)
        return sum(precisions) / max(len(precisions), 1)
    
    def recall_at_k(self, k=5) -> float:
        """Recall@K: 检索到的相关文档占总相关文档的比例"""
        if not self.results:
            return 0
        recalls = []
        for r in self.results:
            if r["ground_truth"]:
                total = len(r["ground_truth"])
                found = sum(1 for d in r["retrieved"][:k]
                          if self._is_relevant(d, r["ground_truth"]))
                recalls.append(found / max(total, 1))
        return sum(recalls) / max(len(recalls), 1)
    
    def mrr(self) -> float:
        """MRR: 第一个相关结果排名的倒数"""
        rr_list = []
        for r in self.results:
            if r["ground_truth"]:
                for i, doc in enumerate(r["retrieved"]):
                    if self._is_relevant(doc, r["ground_truth"]):
                        rr_list.append(1.0 / (i + 1))
                        break
                else:
                    rr_list.append(0)
        return sum(rr_list) / max(len(rr_list), 1)
    
    def _is_relevant(self, doc, ground_truth_docs) -> bool:
        for gt in ground_truth_docs:
            if doc.page_content[:100] in gt or gt in doc.page_content:
                return True
        return False
    
    def report(self) -> dict:
        return {
            "queries": len(self.results),
            "precision@5": round(self.precision_at_k(5), 3),
            "recall@5": round(self.recall_at_k(5), 3),
            "mrr": round(self.mrr(), 3)
        }
```

```python
    """RAG策略对比实验"""
    
    def __init__(self, test_cases: list, evaluator):
        self.test_cases = test_cases
        self.evaluator = evaluator
    
    def compare_strategies(self, strategies: dict) -> dict:
        """对比多个检索策略"""
        results = {}
        
        for name, strategy_func in strategies.items():
            scores = []
            for case in self.test_cases:
                query = case["input"]["question"]
                reference = case.get("expected", "")
                
                # 使用策略检索并回答
                answer = strategy_func(query)
                
                # 评估

                "avg_score": sum(scores) / len(scores),
                "min_score": min(scores),
                "max_score": max(scores),
                "scores": scores
            }
        
        # 排序
        ranked = sorted(results.items(), key=lambda x: x[1]["avg_score"], reverse=True)
        print("\n策略对比结果：")
        for i, (name, r) in enumerate(ranked):
            print(f"  #{i+1} {name}: avg={r['avg_score']:.3f}")
        
        return {"rankings": ranked, "results": results}
```

## 21.7 实操：优化DocMind的RAG管线

```python
"""DocMind 高级RAG管线"""

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from rank_bm25 import BM25Okapi


class AdvancedRAGPipeline:
    def __init__(self):
        self.embeddings = OpenAIEmbeddings()
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0)
        self.vectorstore = None
        self.bm25 = None
        self.documents = []
    
    def ingest_documents(self, file_paths: list):
        all_docs = []
        for path in file_paths:
            if path.endswith(".pdf"):
                from langchain_community.document_loaders import PyPDFLoader
                loader = PyPDFLoader(path)
            else:
                from langchain_community.document_loaders import TextLoader
                loader = TextLoader(path, encoding="utf-8")
            docs = loader.load()
            for d in docs:
                d.metadata["source_file"] = path
            all_docs.extend(docs)
        
        chunks = RecursiveCharacterTextSplitter(
            chunk_size=500, chunk_overlap=50,
            separators=["\n\n", "\n", "。", ". ", " "]
        ).split_documents(all_docs)
        
        for i, c in enumerate(chunks):
            c.metadata["chunk_index"] = i
        self.documents = chunks
        self.vectorstore = Chroma.from_documents(chunks, self.embeddings)
        self.bm25 = BM25Okapi([d.page_content.lower().split() for d in chunks])
        print(f"索引完成: {len(chunks)}个片段")
    
    def hybrid_search(self, query: str, k: int = 5) -> list:
        vec_docs = self.vectorstore.similarity_search_with_relevance_scores(query, k=k*2)
        bm25_scores = self.bm25.get_scores(query.lower().split())
        bm25_ranked = sorted(enumerate(bm25_scores), key=lambda x: x[1], reverse=True)[:k*2]
        
        rrf = {}
        for rank, (doc, _) in enumerate(vec_docs):
            did = hash(doc.page_content[:200])
            rrf[did] = rrf.get(did, 0) + 0.6 / (60 + rank + 1)
        for rank, (idx, _) in enumerate(bm25_ranked):
            did = hash(self.documents[idx].page_content[:200])
            rrf[did] = rrf.get(did, 0) + 0.4 / (60 + rank + 1)
        
        top_ids = sorted(rrf, key=rrf.get, reverse=True)[:k]
        results, seen = [], set()
        for doc, _ in vec_docs:
            did = hash(doc.page_content[:200])
            if did in top_ids and did not in seen:
                seen.add(did)
                results.append(doc)
        for idx, _ in bm25_ranked:
            did = hash(self.documents[idx].page_content[:200])
            if did in top_ids and did not in seen:
                seen.add(did)
                results.append(self.documents[idx])
        return results[:k]
    
    def answer(self, query: str, use_hybrid: bool = True) -> dict:
        docs = self.hybrid_search(query) if use_hybrid else self.vectorstore.similarity_search(query, k=5)
        context = "\n---\n".join([f"[{d.metadata.get('source_file','?')}]\n{d.page_content}" for d in docs])
        prompt = ChatPromptTemplate.from_messages([
            ("system", "基于上下文回答。不足则说明。\n上下文：{context}"),
            ("human", "{query}")
        ])
        answer = (prompt | self.llm | StrOutputParser()).invoke({"query": query, "context": context})
        return {"answer": answer, "sources": [d.metadata.get("source_file","") for d in docs]}
```
