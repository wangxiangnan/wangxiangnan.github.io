# 第11章：RediSearch——全文搜索引擎

## 引言：Redis 里的 Elasticsearch

到目前为止，Redis 查数据基本靠"精确匹配"。想找"所有名字包含手机且价格在 2000-5000 之间的商品"？祝你好运。

**RediSearch** 让你在 Redis 中拥有类似 Elasticsearch 的搜索能力——但速度快得多（因为数据在内存里）。

## 11.1 安装 RediSearch

```bash
# 推荐：使用 Redis Stack（包含所有模块）
docker run -d --name redis-stack -p 6379:6379 redis/redis-stack:latest

# 或加载模块
# redis.conf: loadmodule /path/to/redisearch.so
redis-cli> MODULE LIST

const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function createSearchIndex() {
  const products = [

    { id: 'product:2', name: 'MacBook Pro 16', brand: 'Apple', price: 19999, category: 'laptop', desc: 'M3 Max芯片 专业级性能' },
    { id: 'product:3', name: 'Galaxy S24 Ultra', brand: 'Samsung', price: 9999, category: 'phone', desc: '骁龙8 Gen3 AI功能强大' },
    { id: 'product:4', name: 'ThinkPad X1 Carbon', brand: 'Lenovo', price: 12999, category: 'laptop', desc: '超轻商务本 14英寸' },
    { id: 'product:5', name: 'Pixel 8 Pro', brand: 'Google', price: 6999, category: 'phone', desc: 'Tensor G3芯片 AI摄影' },
    { id: 'product:6', name: 'AirPods Pro 2', brand: 'Apple', price: 1899, category: 'audio', desc: '主动降噪 空间音频' },
    { id: 'product:7', name: 'Sony WH-1000XM5', brand: 'Sony', price: 2499, category: 'audio', desc: '业界最佳降噪耳机' },
    { id: 'product:8', name: 'iPad Pro 12.9', brand: 'Apple', price: 8999, category: 'tablet', desc: 'M2芯片 Mini-LED显示屏' },
  ];

  const pipeline = redis.pipeline();
  products.forEach((p) => {
    pipeline.hset(p.id, { name: p.name, brand: p.brand, price: p.price, category: p.category, description: p.desc });
  });
  await pipeline.exec();
  console.log(`已写入 ${products.length} 个商品\n`);

  try {
    await redis.call(
      'FT.CREATE', 'idx:products', 'ON', 'HASH', 'PREFIX', '1', 'product:',
      'SCHEMA',
        'name', 'TEXT', 'WEIGHT', '2.0',
        'brand', 'TAG',
        'price', 'NUMERIC', 'SORTABLE',
        'category', 'TAG',
        'description', 'TEXT'
    );
    console.log('搜索索引创建成功');
  } catch (err) {
    if (err.message.includes('Index already exists')) console.log('索引已存在');
    else throw err;
  }

  await new Promise((r) => setTimeout(r, 500));
  await redis.quit();
}

createSearchIndex().catch(console.error);
```

### 字段类型说明

| 字段类型 | 用途 | 示例 |
|----------|------|------|
| `TEXT` | 全文搜索（分词、模糊匹配） | 产品名称、描述 |
| `TAG` | 精确匹配 | 品牌、分类 |
| `NUMERIC` | 数值范围查询 | 价格、评分 |
| `GEO` | 地理位置查询 | 经纬度 |
| `VECTOR` | 向量相似度搜索 | 嵌入向量 |

## 11.3 搜索查询：FT.SEARCH

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

function printResults(result) {
  const total = result[0];
  console.log(`  共 ${total} 条:`);
  for (let i = 1; i < result.length; i += 2) {
    const obj = {};
    const fields = result[i + 1];
    for (let j = 0; j < fields.length; j += 2) obj[fields[j]] = fields[j + 1];
    console.log(`  [${result[i]}] ${obj.name} | ${obj.brand} | ${obj.price}`);
  }
}

async function searchDemo() {
  console.log('搜索: "Pro"');
  printResults(await redis.call('FT.SEARCH', 'idx:products', 'Pro'));

  console.log('\nname 包含 "iPhone"');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '@name:iPhone'));

  console.log('\nbrand = Apple');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '@brand:{Apple}'));

  console.log('\nprice 5000-10000');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '@price:[5000 10000]'));

  console.log('\nApple + price < 10000');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '@brand:{Apple} @price:[0 10000]'));

  console.log('\n分页: 每页3条');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '*', 'LIMIT', '0', '3'));

  console.log('\n按价格升序');
  printResults(await redis.call('FT.SEARCH', 'idx:products', '*', 'SORTBY', 'price', 'ASC'));

  console.log('\n高亮搜索');
  printResults(await redis.call('FT.SEARCH', 'idx:products', 'Pro',
    'HIGHLIGHT', 'FIELDS', '1', 'name', 'TAGS', '<b>', '</b>'));

  await redis.quit();
}

searchDemo().catch(console.error);
```

## 11.4 聚合查询：FT.AGGREGATE

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function aggregateDemo() {
  console.log('按品牌分组统计:\n');
  const result = await redis.call(
    'FT.AGGREGATE', 'idx:products', '*',
    'GROUPBY', '1', '@brand',
      'REDUCE', 'COUNT', '0', 'AS', 'count',
      'REDUCE', 'AVG', '1', '@price', 'AS', 'avg_price',
    'SORTBY', '2', '@count', 'DESC'
  );

  for (let i = 1; i < result.length; i++) {
    const obj = {};
    for (let j = 0; j < result[i].length; j += 2) obj[result[i][j]] = result[i][j + 1];
    console.log(`  ${obj.brand}: ${obj.count}件 | 均价 ${parseFloat(obj.avg_price).toFixed(0)}`);
  }

  await redis.quit();
}

aggregateDemo().catch(console.error);
```

## 11.5 索引管理

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function indexManagement() {
  const indexes = await redis.call('FT._LIST');
  console.log('所有索引:', indexes);

  const info = await redis.call('FT.INFO', 'idx:products');
  console.log('\n索引详情:');
  for (let i = 0; i < info.length; i += 2) {
    if (typeof info[i + 1] !== 'object') console.log(`  ${info[i]}: ${info[i + 1]}`);
  }

  await redis.call('FT.ALIASADD', 'products', 'idx:products');
  console.log('\n别名已添加');

  const result = await redis.call('FT.SEARCH', 'products', '*');
  console.log(`通过别名搜索到 ${result[0]} 条结果`);

  await redis.quit();
}

indexManagement().catch(console.error);
```

## 11.6 实战：商品搜索引擎

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

class ProductSearchEngine {
  constructor(redisClient) {
    this.redis = redisClient;
    this.indexName = 'idx:shop_products';
  }

  async initIndex() {
    try {
      await this.redis.call(
        'FT.CREATE', this.indexName, 'ON', 'HASH', 'PREFIX', '1', 'shop:product:',
        'STOPWORDS', '0',
        'SCHEMA',
          'title', 'TEXT', 'WEIGHT', '5.0',
          'description', 'TEXT', 'WEIGHT', '1.0',
          'brand', 'TAG', 'category', 'TAG',
          'price', 'NUMERIC', 'SORTABLE',
          'rating', 'NUMERIC', 'SORTABLE',
          'stock', 'NUMERIC', 'SORTABLE',
          'tags', 'TAG', 'SEPARATOR', ','
      );
    } catch (err) {
      if (!err.message.includes('already exists')) throw err;
    }
  }

  async importProducts(products) {
    const pipeline = this.redis.pipeline();
    products.forEach((p) => {
      pipeline.hset(`shop:product:${p.id}`, {
        title: p.title, description: p.description,
        brand: p.brand, category: p.category,
        price: p.price, rating: p.rating || 0,
        stock: p.stock || 0, created_at: p.createdAt || Date.now(),
        tags: (p.tags || []).join(','),
      });
    });
    await pipeline.exec();
    console.log(`导入了 ${products.length} 个商品`);
  }

  async search(query, options = {}) {
    const { page = 1, pageSize = 10, sortBy = 'price', sortOrder = 'ASC', filters = {} } = options;
    let q = query || '*';
    if (filters.brand) q += ` @brand:{${filters.brand}}`;
    if (filters.category) q += ` @category:{${filters.category}}`;
    if (filters.minPrice !== undefined || filters.maxPrice !== undefined) {
      q += ` @price:[${filters.minPrice || 0} ${filters.maxPrice || '+inf'}]`;
    }
    if (filters.inStock) q += ` @stock:[1 +inf]`;

    const args = [this.indexName, q, 'LIMIT', String((page - 1) * pageSize), String(pageSize)];
    if (sortBy) args.push('SORTBY', sortBy, sortOrder);

    const result = await this.redis.call('FT.SEARCH', ...args);
    const products = [];
    for (let i = 1; i < result.length; i += 2) {
      const obj = { id: result[i].replace('shop:product:', '') };
      for (let j = 0; j < result[i + 1].length; j += 2) obj[result[i + 1][j]] = result[i + 1][j + 1];
      products.push(obj);
    }
    return { total: result[0], page, pageSize, products };
  }
}

async function main() {
  const engine = new ProductSearchEngine(redis);
  await engine.initIndex();

  await engine.importProducts([
    { id: '101', title: 'iPhone 15 Pro Max', description: 'A17 Pro芯片', brand: 'Apple', category: 'phone', price: 9999, rating: 4.8, stock: 50, tags: ['5G', '旗舰'] },
    { id: '102', title: 'Galaxy S24 Ultra', description: '骁龙8 Gen3', brand: 'Samsung', category: 'phone', price: 9699, rating: 4.7, stock: 30, tags: ['5G', 'AI'] },
    { id: '103', title: 'MacBook Air M3', description: 'M3芯片 18小时续航', brand: 'Apple', category: 'laptop', price: 12999, rating: 4.9, stock: 20, tags: ['轻薄', 'M3'] },
    { id: '104', title: 'Sony WH-1000XM5', description: '业界最佳降噪', brand: 'Sony', category: 'audio', price: 2499, rating: 4.8, stock: 100, tags: ['降噪'] },
    { id: '105', title: '小米14 Ultra', description: '徕卡光学', brand: 'Xiaomi', category: 'phone', price: 6499, rating: 4.6, stock: 0, tags: ['5G'] },
  ]);

  await new Promise((r) => setTimeout(r, 500));

  console.log('\n搜索: "Pro"');
  const r1 = await engine.search('Pro');
  r1.products.forEach((p) => console.log(`  ${p.title} - ${p.price}`));

  console.log('\nApple品牌 价格<10000');
  const r2 = await engine.search('', { filters: { brand: 'Apple', maxPrice: 10000 } });
  r2.products.forEach((p) => console.log(`  ${p.title} - ${p.price}`));

  console.log('\n有货的手机');
  const r3 = await engine.search('', { filters: { category: 'phone', inStock: true }, sortBy: 'rating', sortOrder: 'DESC' });
  r3.products.forEach((p) => console.log(`  ${p.title} - 评分${p.rating}`));

  await redis.quit();
}

main().catch(console.error);
```

## 小结

- **RediSearch** 是 Redis 模块生态中最重要的搜索组件
- 支持 TEXT、TAG、NUMERIC、GEO、VECTOR 字段类型
- FT.AGGREGATE 支持分组统计、聚合计算
- 适合**低延迟搜索**场景
