## 28.2 subprocess：执行系统命令

### 实用示例：批量重命名文件

```python
import os
from pathlib import Path

def batch_rename(directory: str, old_pattern: str, new_pattern: str):
    """批量重命名文件"""
    count = 0
    for file in Path(directory).iterdir():
        if file.is_file() and old_pattern in file.name:
            new_name = file.name.replace(old_pattern, new_pattern)
            file.rename(file.parent / new_name)
            print(f"  {file.name} -> {new_name}")
            count += 1
    print(f"\n共重命名 {count} 个文件")

# 使用示例
# batch_rename("./photos", "IMG_", "2024_vacation_")
```

## 28.3 schedule：定时任务

`schedule` 是一个轻量的定时任务库，语法非常直观——就像用英语描述任务一样。

```bash
pip install schedule
```

```python
import schedule
import time
from datetime import datetime

def job():
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    print(f"[{now}] 任务执行中...")

# 各种调度方式
schedule.every(10).seconds.do(job)           # 每 10 秒
schedule.every(1).minutes.do(job)            # 每 1 分钟
schedule.every().hour.do(job)                # 每小时
schedule.every().day.at("10:30").do(job)     # 每天 10:30
schedule.every().monday.do(job)              # 每周一
schedule.every().wednesday.at("13:15").do(job)  # 每周三 13:15

# 运行调度器（主循环）
print("定时任务已启动，按 Ctrl+C 停止...")
try:
    while True:
        schedule.run_pending()
        time.sleep(1)
except KeyboardInterrupt:
    print("\n已停止")
```

## 28.6 自动发送邮件

Python 标准库自带了邮件发送功能，不需要额外安装。

```python
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.application import MIMEApplication
import os

def send_email(
    to_email: str,
    subject: str,
    body: str,
    attachment_path: str | None = None,
):
    """发送带附件的邮件"""
    # 从环境变量读取配置
    smtp_server = os.getenv("SMTP_SERVER", "smtp.gmail.com")
    smtp_port = int(os.getenv("SMTP_PORT", "587"))
    from_email = os.getenv("EMAIL_USER")
    password = os.getenv("EMAIL_PASSWORD")

    if not from_email or not password:
        print("错误: 请设置 EMAIL_USER 和 EMAIL_PASSWORD 环境变量")
        return

    # 构建邮件
    msg = MIMEMultipart()
    msg["From"] = from_email
    msg["To"] = to_email
    msg["Subject"] = subject

    # 正文
    msg.attach(MIMEText(body, "html", "utf-8"))

    # 附件
    if attachment_path and os.path.exists(attachment_path):
        with open(attachment_path, "rb") as f:
            part = MIMEApplication(f.read())
            part.add_header(
                "Content-Disposition",
                "attachment",
                filename=os.path.basename(attachment_path),
            )
            msg.attach(part)

    # 发送
    try:
        with smtplib.SMTP(smtp_server, smtp_port) as server:
            server.starttls()  # 启用加密
            server.login(from_email, password)
            server.send_message(msg)
        print(f"邮件已成功发送至 {to_email}")
    except Exception as e:
        print(f"发送失败: {e}")
```

## 28.7 实战：自动生成周报 Excel + 发邮件

把前面的知识组合起来，完成一个实用的自动化项目：自动从数据源生成周报 Excel，并发送邮件给领导。

### 完整项目代码

```python
import os
import random
from datetime import datetime, timedelta
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment, PatternFill, Border, Side
from openpyxl.chart import BarChart, Reference

# ========== 配置 ==========
REPORT_DIR = "reports"
WEEK_START = datetime(2024, 1, 1)  # 本周一

# ========== 模拟数据源 ==========
def generate_weekly_data() -> list[dict]:
    """生成一周的销售数据（模拟从数据库/系统导出）"""
    products = ["智能手表", "蓝牙耳机", "充电宝", "数据线", "手机壳"]
    data = []
    for i in range(7):
        date = WEEK_START + timedelta(days=i)
        for product in products:
            data.append({
                "date": date.strftime("%Y-%m-%d"),
                "weekday": ["周一", "周二", "周三", "周四", "周五", "周六", "周日"][i],
                "product": product,
                "sales": random.randint(10, 200),
                "revenue": round(random.uniform(1000, 20000), 2),
            })
    return data

# ========== 生成 Excel 报告 ==========
def create_report_excel(data: list[dict], filename: str) -> str:
    """根据数据生成周报 Excel"""
    wb = Workbook()

    # ---- 样式定义 ----
    header_font = Font(name="微软雅黑", bold=True, size=11, color="FFFFFF")
    header_fill = PatternFill(start_color="4472C4", end_color="4472C4", fill_type="solid")
    title_font = Font(name="微软雅黑", bold=True, size=16)
    thin_border = Border(
        left=Side(style="thin"), right=Side(style="thin"),
        top=Side(style="thin"), bottom=Side(style="thin"),
    )

    # ---- Sheet 1: 每日明细 ----
    ws1 = wb.active
    ws1.title = "每日明细"

    # 标题
    ws1.merge_cells("A1:F1")
    ws1["A1"] = f"销售周报 {WEEK_START.strftime('%Y.%m.%d')} - {(WEEK_START + timedelta(days=6)).strftime('%Y.%m.%d')}"
    ws1["A1"].font = title_font
    ws1["A1"].alignment = Alignment(horizontal="center")

    # 表头
    headers = ["日期", "星期", "产品", "销量", "收入", "备注"]
    for col, header in enumerate(headers, 1):
        cell = ws1.cell(row=3, column=col, value=header)
        cell.font = header_font
        cell.fill = header_fill
        cell.alignment = Alignment(horizontal="center")
        cell.border = thin_border

    # 数据
    for i, row in enumerate(data, 4):
        ws1.cell(row=i, column=1, value=row["date"]).border = thin_border
        ws1.cell(row=i, column=2, value=row["weekday"]).border = thin_border
        ws1.cell(row=i, column=3, value=row["product"]).border = thin_border
        ws1.cell(row=i, column=4, value=row["sales"]).border = thin_border
        ws1.cell(row=i, column=5, value=row["revenue"]).border = thin_border
        ws1.cell(row=i, column=6, value="").border = thin_border

    # 列宽
    for col, width in zip("ABCDEF", [12, 8, 12, 10, 12, 15]):
        ws1.column_dimensions[col].width = width

    # ---- Sheet 2: 产品汇总 ----
    ws2 = wb.create_sheet("产品汇总")
    ws2["A1"] = "产品销售汇总"
    ws2["A1"].font = title_font

    # 按产品汇总
    product_summary = {}
    for row in data:
        p = row["product"]
        if p not in product_summary:
            product_summary[p] = {"sales": 0, "revenue": 0}
        product_summary[p]["sales"] += row["sales"]
        product_summary[p]["revenue"] += row["revenue"]

    summary_headers = ["产品", "总销量", "总收入", "平均单价"]
    for col, header in enumerate(summary_headers, 1):
        cell = ws2.cell(row=3, column=col, value=header)
        cell.font = header_font
        cell.fill = header_fill
        cell.border = thin_border

    for i, (product, stats) in enumerate(sorted(product_summary.items(), key=lambda x: -x[1]["revenue"]), 4):
        ws2.cell(row=i, column=1, value=product).border = thin_border
        ws2.cell(row=i, column=2, value=stats["sales"]).border = thin_border
        ws2.cell(row=i, column=3, value=round(stats["revenue"], 2)).border = thin_border
        avg_price = round(stats["revenue"] / stats["sales"], 2) if stats["sales"] else 0
        ws2.cell(row=i, column=4, value=avg_price).border = thin_border

    # 添加柱状图
    chart = BarChart()
    chart.title = "产品销量对比"
    chart.y_axis.title = "销量"
    data_ref = Reference(ws2, min_col=2, min_row=3, max_row=3 + len(product_summary))
    cats_ref = Reference(ws2, min_col=1, min_row=4, max_row=3 + len(product_summary))
    chart.add_data(data_ref, titles_from_data=True)
    chart.set_categories(cats_ref)
    chart.shape = 4
    ws2.add_chart(chart, "A10")

    for col, width in zip("ABCD", [15, 10, 12, 12]):
        ws2.column_dimensions[col].width = width

    # ---- Sheet 3: 每日趋势 ----
    ws3 = wb.create_sheet("每日趋势")
    ws3["A1"] = "每日销售趋势"
    ws3["A1"].font = title_font

    daily_summary = {}
    for row in data:
        d = row["date"]
        if d not in daily_summary:
            daily_summary[d] = {"sales": 0, "revenue": 0}
        daily_summary[d]["sales"] += row["sales"]
        daily_summary[d]["revenue"] += row["revenue"]

    trend_headers = ["日期", "总销量", "总收入"]
    for col, header in enumerate(trend_headers, 1):
        cell = ws3.cell(row=3, column=col, value=header)
        cell.font = header_font
        cell.fill = header_fill
        cell.border = thin_border

    for i, (date, stats) in enumerate(sorted(daily_summary.items()), 4):
        ws3.cell(row=i, column=1, value=date).border = thin_border
        ws3.cell(row=i, column=2, value=stats["sales"]).border = thin_border
        ws3.cell(row=i, column=3, value=round(stats["revenue"], 2)).border = thin_border

    for col, width in zip("ABC", [12, 10, 12]):
        ws3.column_dimensions[col].width = width

    # 保存
    os.makedirs(REPORT_DIR, exist_ok=True)
    filepath = os.path.join(REPORT_DIR, filename)
    wb.save(filepath)
    print(f"Excel 报告已生成: {filepath}")
    return filepath

# ========== 生成邮件正文 ==========
def generate_email_body(data: list[dict]) -> str:
    """生成 HTML 格式的邮件正文"""
    total_sales = sum(row["sales"] for row in data)
    total_revenue = sum(row["revenue"] for row in data)
    avg_daily = round(total_revenue / 7, 2)

    # 找出销量冠军
    product_sales = {}
    for row in data:
        product_sales[row["product"]] = product_sales.get(row["product"], 0) + row["sales"]
    top_product = max(product_sales, key=product_sales.get)

    body = f"""
    <h2>本周销售周报摘要</h2>
    <p>报告周期: {WEEK_START.strftime('%Y-%m-%d')} ~ {(WEEK_START + timedelta(days=6)).strftime('%Y-%m-%d')}</p>

    <table border="1" cellpadding="8" cellspacing="0" style="border-collapse:collapse;">
        <tr style="background-color:#4472C4; color:white;">
            <th>指标</th><th>数值</th>
        </tr>
        <tr><td>总销量</td><td>{total_sales:,} 件</td></tr>
        <tr><td>总收入</td><td>¥{total_revenue:,.2f}</td></tr>
        <tr><td>日均收入</td><td>¥{avg_daily:,.2f}</td></tr>
        <tr><td>销量冠军</td><td>{top_product}（{product_sales[top_product]:,} 件）</td></tr>
    </table>

    <p><em>详细数据请查看附件中的 Excel 文件。</em></p>
    """
    return body

# ========== 主程序 ==========
def main():
    print("=" * 50)
    print("  自动化周报生成器")
    print("=" * 50)

    # 1. 获取数据
    print("\n[1/3] 获取本周销售数据...")
    data = generate_weekly_data()
    print(f"  共 {len(data)} 条记录")

    # 2. 生成 Excel
    week_num = WEEK_START.isocalendar()[1]
    filename = f"销售周报_2024年第{week_num}周.xlsx"
    print(f"\n[2/3] 生成 Excel 报告: {filename}")
    filepath = create_report_excel(data, filename)

    # 3. 准备邮件
```

## 28.8 小结

本章我们学习了 Python 在自动化和办公领域的应用：

- **os 和 pathlib** 提供了文件系统的完整操作能力
- **subprocess** 让你调用系统命令，串联各种工具

- **smtplib** 可以自动发送邮件

自动化的核心理念是：把重复的、机械的工作交给代码，把你的时间留给需要思考和创造力的事情。

```python
import os
import time
from pathlib import Path
from datetime import datetime, timedelta

def cleanup_old_logs(directory: str, max_age_days: int = 30):
    """清理超过指定天数未修改的 .log 文件"""
    cutoff = datetime.now() - timedelta(days=max_age_days)
    to_delete = []

    for file in Path(directory).glob("**/*.log"):
        mtime = datetime.fromtimestamp(os.path.getmtime(file))
        if mtime < cutoff:
            to_delete.append((file, mtime))

    if not to_delete:
        print("没有需要清理的旧日志文件。")
        return

    print(f"以下 {len(to_delete)} 个日志文件超过 {max_age_days} 天未修改:")
    for file, mtime in to_delete:
        size_kb = file.stat().st_size / 1024
        print(f"  {file} ({size_kb:.1f} KB, 修改于 {mtime:%Y-%m-%d})")

    # 确认后删除
    confirm = input("\n确认删除？(y/n): ")
    if confirm.lower() == "y":
        for file, _ in to_delete:
            file.unlink()
            print(f"  已删除: {file}")
        print(f"\n共删除 {len(to_delete)} 个文件")
    else:
        print("已取消。")
```

## 28.11 常见错误与避坑

**1. 文件编码问题**

```python
# 错误：在 Windows 上读取 UTF-8 文件时默认用 GBK 解码
with open("data.csv", "r") as f:
    content = f.read()  # 可能报 UnicodeDecodeError

# 正确：明确指定编码
with open("data.csv", "r", encoding="utf-8") as f:
    content = f.read()
```

**2. 路径拼接不要用字符串连接**

```python
# 错误：不同操作系统路径分隔符不同
path = "reports/" + "2024/" + "report.xlsx"

# 正确：使用 os.path.join 或 pathlib
import os
path = os.path.join("reports", "2024", "report.xlsx")

# 更推荐 pathlib
from pathlib import Path
path = Path("reports") / "2024" / "report.xlsx"
```

**3. openpyxl 中单元格索引从 1 开始**

```python
# openpyxl 的行列索引从 1 开始（不是 0！）
ws.cell(row=1, column=1)  # 这是 A1 单元格

# 这与 Python 列表从 0 开始不同，容易搞混
data = ["a", "b", "c"]
data[0]  # "a"（索引从 0 开始）
ws.cell(row=1, column=1)  # 第一行第一列（索引从 1 开始）
```

**4. 忘记关闭文件**

```python
# 错误：没有使用 with 语句，可能忘记关闭
f = open("data.txt", "r")
content = f.read()
# 忘记 f.close()...

# 正确：始终使用 with 语句
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()
# 自动关闭
```

**5. subprocess 在 Windows 上的差异**

```python
import subprocess
import platform

# 不同操作系统的命令不同
if platform.system() == "Windows":
    result = subprocess.run(["dir"], shell=True, capture_output=True, text=True)
else:
    result = subprocess.run(["ls", "-la"], capture_output=True, text=True)

# shell=True 有安全风险（命令注入），尽量用列表形式
# 错误（危险）
subprocess.run(f"ping {user_input}", shell=True)
# 正确
subprocess.run(["ping", "-c", "4", user_input])
```
