# 第25章 生产部署 —— 从Demo到Production

## 25.1 开篇：Demo vs Production的差距

你已经构建了功能完整的DocMind Agent，本地跑得非常好。现在老板说："上线吧！"——这时候你会发现Demo和Production之间有一道巨大的鸿沟：

| 维度 | Demo | Production |
| ------ | ------ | ----------- |
| 用户数 | 1（你自己） | 100+并发 |
| 可靠性 | 崩了重启 | 99.9%可用 |
| 性能 | 等几秒没事 | 首字<1秒 |
| 安全 | 没人在意 | 每请求都检查 |
| 监控 | 看终端 | 实时告警 |
| 成本 | 免费额度 | 每月数千美元 |

这一章，我们把DocMind从Demo变成生产级应用。

## 25.1.1 生产部署的核心原则

在深入代码之前，让我们确立几个核心原则：

1. **可靠性优先**：任何单点故障都是不可接受的。每个组件都要有降级方案。
2. **可观测性**：如果你不能观察到系统内部发生了什么，你就无法修复它。
3. **渐进式发布**：永远不要一次性把新版本推给所有用户。
4. **成本意识**：LLM API 调用是真实的花费，需要精细化管理。
5. **安全纵深**：不依赖单一安全措施，多层防护才是正道。

这些原则将贯穿本章的所有实践。

from collections import defaultdict


class RateLimiter:
    def __init__(self, max_requests=30, window=60):
        self.max_requests = max_requests
        self.window = window
        self.requests = defaultdict(list)
    
    def check(self, user_id: str) -> bool:
        now = time.time()
        self.requests[user_id] = [t for t in self.requests[user_id] if now - t < self.window]
        if len(self.requests[user_id]) >= self.max_requests:
            return False
        self.requests[user_id].append(now)
        return True

### 25.5.1 语义缓存（Semantic Cache）

```python
import hashlib
import json
import time
from typing import Optional


class SemanticCache:
    """语义缓存 — 相似问题直接返回缓存答案"""
    
    def __init__(self, similarity_threshold=0.92, ttl=3600):
        self.threshold = similarity_threshold
        self.ttl = ttl  # 缓存有效期(秒)
        self.cache = {}  # 生产环境用Redis
        self.embeddings = None  # 延迟加载
    
    def _get_embedding(self, text: str):
        if not self.embeddings:
            from langchain_openai import OpenAIEmbeddings
            self.embeddings = OpenAIEmbeddings()
        return self.embeddings.embed_query(text)
    
    def get(self, query: str) -> Optional[str]:
        """查找缓存"""
        import numpy as np
        
        query_emb = self._get_embedding(query)
        now = time.time()
        
        for key, entry in self.cache.items():
            # TTL检查
            if now - entry["timestamp"] > self.ttl:
                continue
            
            # 语义相似度检查
            similarity = np.dot(query_emb, entry["embedding"]) / (
                np.linalg.norm(query_emb) * np.linalg.norm(entry["embedding"])
            )
            
            if similarity >= self.threshold:
                entry["hits"] += 1
                return entry["answer"]
        
        return None
    
    def set(self, query: str, answer: str):
        """写入缓存"""
        key = hashlib.md5(query.encode()).hexdigest()
        self.cache[key] = {
            "embedding": self._get_embedding(query),
            "answer": answer,
            "timestamp": time.time(),
            "hits": 0,
            "query": query
        }
    
    def stats(self) -> dict:
        total = len(self.cache)
        hits = sum(e["hits"] for e in self.cache.values())
        return {"entries": total, "total_hits": hits}


# 使用示例
cache = SemanticCache(similarity_threshold=0.92, ttl=3600)

def cached_ask(question: str) -> str:
    # 先查缓存
    cached = cache.get(question)
    if cached:
        print(f"[缓存命中] 相似度查询返回缓存结果")
        return cached
    
    # 缓存未命中，调用Agent
    answer = docmind.ask(question)
    
    # 写入缓存
    cache.set(question, answer)
    return answer
```

class TokenOptimizer:
    """Token使用优化器"""
    
    def __init__(self, max_context_tokens=4000):
        self.max_tokens = max_context_tokens
    
    def optimize_context(self, documents: list, query: str) -> list:
        """智能选择最相关的文档片段，控制总Token数"""
        # 按相关性排序（假设已排序）
        selected = []
        total_tokens = 0
        
        for doc in documents:
            doc_tokens = len(doc.page_content) // 4  # 粗略估算
            if total_tokens + doc_tokens > self.max_tokens:
                break
            selected.append(doc)
            total_tokens += doc_tokens
        
        return selected
    
    def compress_answer(self, answer: str, max_length: int = 500) -> str:
        """压缩过长的回答"""
        if len(answer) <= max_length:
            return answer
        
        # 简单策略：截断到最近的完整句子
        truncated = answer[:max_length]
        last_period = max(truncated.rfind("。"), truncated.rfind("."), truncated.rfind("\n"))
        if last_period > max_length * 0.7:
            return truncated[:last_period + 1]
        return truncated + "..."

"""DocMind 监控系统"""

import time
from collections import defaultdict
from dataclasses import dataclass, field


@dataclass
class Metrics:
    """应用指标"""
    total_requests: int = 0
    successful_requests: int = 0
    failed_requests: int = 0
    total_latency_ms: float = 0
    total_tokens_used: int = 0
    cache_hits: int = 0
    errors: dict = field(default_factory=lambda: defaultdict(int))
    
    @property
    def avg_latency(self):
        return self.total_latency_ms / max(self.total_requests, 1)
    
    @property
    def success_rate(self):
        return self.successful_requests / max(self.total_requests, 1)
    
    @property
    def cache_hit_rate(self):
        return self.cache_hits / max(self.total_requests, 1)
    
    def to_dict(self):
        return {
            "total_requests": self.total_requests,
            "success_rate": round(self.success_rate * 100, 1),
            "avg_latency_ms": round(self.avg_latency, 1),
            "cache_hit_rate": round(self.cache_hit_rate * 100, 1),
            "total_tokens": self.total_tokens_used,
            "errors": dict(self.errors)
        }


# Prometheus兼容的指标导出
class PrometheusExporter:

class AlertManager:
    """告警管理器"""
    
    def __init__(self, thresholds=None):
        self.thresholds = thresholds or {
            "error_rate": 0.05,       # 错误率超过5%告警
            "avg_latency_ms": 5000,   # 平均延迟超过5秒告警
            "success_rate": 0.95,     # 成功率低于95%告警
        }
    
    def check(self, metrics: Metrics) -> list:
        alerts = []
        
        if metrics.success_rate < self.thresholds["success_rate"]:
            alerts.append({
                "level": "critical",
                "message": f"成功率过低: {metrics.success_rate*100:.1f}%",
                "threshold": self.thresholds["success_rate"]
            })
        
        if metrics.avg_latency > self.thresholds["avg_latency_ms"]:
            alerts.append({
                "level": "warning",
                "message": f"延迟过高: {metrics.avg_latency:.0f}ms",
                "threshold": self.thresholds["avg_latency_ms"]
            })
        
        return alerts

class CostTracker:
    """成本追踪器"""
    
    # 每1K token的价格（美元）
    PRICING = {
        "gpt-4o": {"input": 0.005, "output": 0.015},
        "gpt-4o-mini": {"input": 0.00015, "output": 0.0006},
        "gpt-3.5-turbo": {"input": 0.0005, "output": 0.0015},
    }
    
    def __init__(self, monthly_budget=500):
        self.budget = monthly_budget
        self.usage = defaultdict(lambda: {"input_tokens": 0, "output_tokens": 0, "cost": 0})
    
    def record(self, model: str, input_tokens: int, output_tokens: int):
        pricing = self.PRICING.get(model, self.PRICING["gpt-4o"])
        cost = (input_tokens / 1000 * pricing["input"] + 
                output_tokens / 1000 * pricing["output"])
        self.usage[model]["input_tokens"] += input_tokens
        self.usage[model]["output_tokens"] += output_tokens
        self.usage[model]["cost"] += cost
    
    def total_cost(self) -> float:
        return sum(u["cost"] for u in self.usage.values())
    
    def budget_remaining(self) -> float:
        return self.budget - self.total_cost()
    
    def should_use_premium_model(self) -> bool:
        """是否应该使用高级模型"""
        return self.budget_remaining() > self.budget * 0.3
    
    def report(self) -> dict:
        return {
            "total_cost": round(self.total_cost(), 2),
            "budget": self.budget,
            "remaining": round(self.budget_remaining(), 2),
            "by_model": {m: {"cost": round(u["cost"], 2), "tokens": u["input_tokens"] + u["output_tokens"]}
                        for m, u in self.usage.items()}
        }

"""灰度发布管理器"""


class CanaryDeployment:
    """金丝雀发布"""
    
    def __init__(self, old_agent, new_agent, traffic_ratio=0.1):
        self.old_agent = old_agent
        self.new_agent = new_agent
        self.ratio = traffic_ratio  # 新版本的流量比例
        self.metrics = {"old": [], "new": []}
    
    def ask(self, query: str) -> dict:
        """根据流量比例路由到不同版本"""
        import random
        
        if random.random() < self.ratio:
            # 新版本
            answer = self.new_agent.ask(query)
            self.metrics["new"].append({"query": query, "answer": answer})
            return {"answer": answer, "version": "new"}
        else:
            # 旧版本
            answer = self.old_agent.ask(query)
            self.metrics["old"].append({"query": query, "answer": answer})
            return {"answer": answer, "version": "old"}
    
    def increase_traffic(self, step=0.1):
        """逐步增加新版本流量"""
        self.ratio = min(1.0, self.ratio + step)
        print(f"新版本流量调整为: {self.ratio*100:.0f}%")
    
    def rollback(self):
        """回滚到旧版本"""
        self.ratio = 0
        print("已回滚到旧版本")
    
    def compare_metrics(self) -> dict:
        """对比两个版本的指标"""
        return {
            "old_queries": len(self.metrics["old"]),
            "new_queries": len(self.metrics["new"]),
            "current_ratio": self.ratio
        }


class BlueGreenDeployment:
    """蓝绿部署"""
    
    def __init__(self, blue_agent, green_agent):
        self.blue = blue_agent
        self.green = green_agent
        self.active = "blue"  # 当前活跃版本
    
    def ask(self, query: str) -> str:
        if self.active == "blue":
            return self.blue.ask(query)
        return self.green.ask(query)
    
    def switch(self):
        """切换活跃版本"""
        self.active = "green" if self.active == "blue" else "blue"
        print(f"切换到 {self.active} 版本")
    
    def health_check(self) -> dict:
        """健康检查"""
        results = {}
        for name, agent in [("blue", self.blue), ("green", self.green)]:
            try:
                agent.ask("test")
                results[name] = "healthy"
            except:
                results[name] = "unhealthy"
        return {"active": self.active, "status": results}
