# 第3章：五大数据类型——Redis的数据结构课

## 引言：Redis 的"五虎将"

如果说 Redis 是一支军队，那五大数据类型就是它的五虎上将，各司其职：

- **String（字符串）**：万能型选手，啥都能干
- **Hash（哈希）**：存对象的老手，JSON 的最佳替身
- **List（列表）**：队列和栈的化身，有序可重复
- **Set（集合）**：去重专家，集合运算大师
- **Sorted Set（有序集合）**：排行榜之王，自带排序

就像 JavaScript 有 Object、Array、Map、Set 一样，Redis 也有自己的数据结构。区别在于：JS 的数据结构存在内存里，进程一死就没了；Redis 的数据结构存在独立的服务里，跨进程共享，还能持久化。

## 3.1 String（字符串）：最简单的类型，也是最强大的

String 是 Redis 最基础的类型，就像 JavaScript 里的 `string`——但别小看它，它不仅能存文本，还能存数字、JSON、甚至二进制数据（最大 512MB）。

**编程比喻**：String 就像一个超级变量——你可以用它存字符串、数字、序列化后的对象，甚至用来做分布式锁。

### 基本操作

```js
const Redis = require('ioredis');
const redis = new Redis();

async function stringDemo() {
  // ========== SET / GET ==========
  // 基本设置和获取
  await redis.set('name', '张三');
  const name = await redis.get('name');
  console.log('姓名：', name); // 张三

  // ========== SETEX：设置并指定过期时间（秒） ==========
  await redis.setex('token:abc123', 3600, 'user-session-data');
  console.log('Token TTL：', await redis.ttl('token:abc123')); // 约3600

  // ========== SETNX：不存在才设置（分布式锁的基础） ==========
  const result1 = await redis.setnx('lock', 'held-by-server-1');
  console.log('第一次 SETNX：', result1); // 1 (成功)
  const result2 = await redis.setnx('lock', 'held-by-server-2');
  console.log('第二次 SETNX：', result2); // 0 (失败，已存在)

  // ========== MSET / MGET：批量操作 ==========
  await redis.mset('user:1:name', 'Alice', 'user:1:age', '25', 'user:1:city', 'Beijing');
  const [uName, uAge, uCity] = await redis.mget('user:1:name', 'user:1:age', 'user:1:city');
  console.log(`用户信息：${uName}, ${uAge}岁, 来自${uCity}`);

  // ========== INCR / DECR：原子递增/递减 ==========
  await redis.set('counter', '0');
  await redis.incr('counter');       // +1
  await redis.incr('counter');       // +1
  await redis.incrby('counter', 10); // +10
  const count = await redis.get('counter');
  console.log('计数器：', count); // 12

  await redis.decr('counter');        // -1
  await redis.decrby('counter', 5);   // -5
  console.log('递减后：', await redis.get('counter')); // 6

  // ========== APPEND：追加字符串 ==========
  await redis.set('msg', 'Hello');
  await redis.append('msg', ' World');
  console.log('追加后：', await redis.get('msg')); // "Hello World"

  // ========== STRLEN：获取字符串长度 ==========
  const len = await redis.strlen('msg');
  console.log('字符串长度：', len); // 11

  // ========== GETRANGE / SETRANGE：子串操作 ==========
  const sub = await redis.getrange('msg', 0, 4);
  console.log('子串(0-4)：', sub); // "Hello"

  await redis.setrange('msg', 6, 'Redis');
  console.log('替换后：', await redis.get('msg')); // "Hello Redis"

  // 清理
  await redis.del('name', 'token:abc123', 'lock', 'counter', 'msg',
    'user:1:name', 'user:1:age', 'user:1:city');
  await redis.quit();
}

stringDemo();
```

## 3.2 Hash（哈希）：对象存储的最佳选择

Hash 就像 JavaScript 的 Object——键值对的集合。区别在于：JS 的 Object 存在进程内存里，Redis 的 Hash 存在 Redis 里，跨进程共享。

**编程比喻**：Hash 就像一个"迷你数据库表"——每个 Hash 是一张表，field 是列名，value 是列值。

```js
const Redis = require('ioredis');
const redis = new Redis();

async function hashDemo() {
  // ========== HSET / HGET：设置/获取单个字段 ==========
  await redis.hset('user:1001', 'name', '李四');
  await redis.hset('user:1001', 'age', '28');
  await redis.hset('user:1001', 'email', 'lisi@example.com');

  const userName = await redis.hget('user:1001', 'name');
  console.log('用户名：', userName); // 李四

  // ========== HMSET / HMGET：批量设置/获取多个字段 ==========
  await redis.hmset('user:1002', {
    name: '王五',
    age: '32',
    city: 'Shanghai',
    role: 'developer',
  });

  const [rName, rCity, rRole] = await redis.hmget('user:1002', 'name', 'city', 'role');
  console.log(`用户：${rName}, 城市：${rCity}, 角色：${rRole}`);

  // ========== HGETALL：获取所有字段和值 ==========
  const allFields = await redis.hgetall('user:1001');
  console.log('用户1001完整信息：', allFields);
  // { name: '李四', age: '28', email: 'lisi@example.com' }

  // ========== HINCRBY：字段值原子递增 ==========
  await redis.hset('user:1001', 'loginCount', '0');
  await redis.hincrby('user:1001', 'loginCount', 1);
  await redis.hincrby('user:1001', 'loginCount', 1);
  const loginCount = await redis.hget('user:1001', 'loginCount');
  console.log('登录次数：', loginCount); // 2

  // ========== HEXISTS：判断字段是否存在 ==========
  const hasEmail = await redis.hexists('user:1001', 'email');
  console.log('有email字段？', hasEmail); // 1 (true)

  const hasPhone = await redis.hexists('user:1001', 'phone');
  console.log('有phone字段？', hasPhone); // 0 (false)

  // ========== HKEYS / HVALS / HLEN ==========
  const keys = await redis.hkeys('user:1001');
  console.log('所有字段名：', keys);

  const values = await redis.hvals('user:1001');
  console.log('所有字段值：', values);

  const fieldCount = await redis.hlen('user:1001');
  console.log('字段数量：', fieldCount);

  // ========== HDEL：删除字段 ==========
  await redis.hdel('user:1001', 'email');
  console.log('删除email后：', await redis.hgetall('user:1001'));

  // 清理
  await redis.del('user:1001', 'user:1002');
  await redis.quit();
}

hashDemo();
```

> [!TIP] 提示
> ：Hash vs String 存对象的选择——如果你需要频繁修改对象的某个字段，用 Hash（`HSET user:1 name "Alice"`）；如果你只是整体读写，用 String + JSON（`SET user:1 '{"name":"Alice"}'`）。Hash 的优势是只修改一个字段时不用传输整个对象。

## 3.3 List（列表）：消息队列的原型

List 就像一个 JavaScript 的 Array——有序、可重复、支持头尾操作。你可以把它当成栈（LIFO）或队列（FIFO）来用。

**编程比喻**：List 就像食堂排队的队伍——你可以从队尾加入（RPUSH），从队首离开（LPOP），也可以插队到队首（LPUSH），还可以看看队伍里都有谁（LRANGE）。

```js
const Redis = require('ioredis');
const redis = new Redis();

async function listDemo() {
  // ========== LPUSH / RPUSH：从头部/尾部添加 ==========
  await redis.lpush('queue:tasks', 'task-1', 'task-2', 'task-3');
  // 队列状态：[task-3, task-2, task-1]（LPUSH是插到头部）

  await redis.rpush('queue:tasks', 'task-4', 'task-5');
  // 队列状态：[task-3, task-2, task-1, task-4, task-5]

  // ========== LRANGE：获取范围内的元素 ==========
  const all = await redis.lrange('queue:tasks', 0, -1);
  console.log('所有任务：', all);
  // ['task-3', 'task-2', 'task-1', 'task-4', 'task-5']

  // ========== LPOP / RPOP：从头部/尾部取出 ==========
  const first = await redis.lpop('queue:tasks');
  console.log('LPOP：', first); // 'task-3'

  const last = await redis.rpop('queue:tasks');
  console.log('RPOP：', last); // 'task-5'

  // ========== LLEN：获取列表长度 ==========
  const length = await redis.llen('queue:tasks');
  console.log('剩余任务数：', length); // 3

  // ========== LINDEX：按索引获取 ==========
  const second = await redis.lindex('queue:tasks', 1);
  console.log('第2个任务：', second);

  // ========== LTRIM：裁剪列表（保留指定范围） ==========
  // 只保留前10个元素（常用于"最新消息"功能）
  await redis.lpush('recent:messages',
    'msg-1', 'msg-2', 'msg-3', 'msg-4', 'msg-5',
    'msg-6', 'msg-7', 'msg-8', 'msg-9', 'msg-10',
    'msg-11', 'msg-12'
  );
  await redis.ltrim('recent:messages', 0, 9); // 只保留前10条
  const recentCount = await redis.llen('recent:messages');
  console.log('最近消息数量：', recentCount); // 10

  // ========== BLPOP / BRPOP：阻塞式弹出（消息队列的核心） ==========
  console.log('等待任务中...（3秒超时）');
  // 如果队列为空，会阻塞等待直到有新元素或超时
  const result = await redis.blpop('queue:empty', 3);
  console.log('BLPOP 结果：', result); // null（超时了）

  // 清理
  await redis.del('queue:tasks', 'recent:messages');
  await redis.quit();
}

listDemo();
```

### 实战：简单的任务队列

```js
const Redis = require('ioredis');

async function taskQueue() {
  const producer = new Redis(); // 生产者
  const consumer = new Redis(); // 消费者

  // 生产者：投递任务
  const tasks = ['发送邮件', '生成报表', '推送通知', '清理日志'];
  for (const task of tasks) {
    await producer.rpush('job:queue', JSON.stringify({
      task,
      createdAt: Date.now(),
    }));
    console.log(`[生产者] 投递任务：${task}`);
  }

  // 消费者：处理任务
  let processed = 0;
  while (true) {
    const item = await consumer.lpop('job:queue');
    if (!item) break; // 队列空了

    const job = JSON.parse(item);
    console.log(`[消费者] 处理任务：${job.task}`);
    processed++;
  }

  console.log(`\n共处理了 ${processed} 个任务`);

  await producer.quit();
  await consumer.quit();
}

taskQueue();
```

## 3.4 Set（集合）：去重神器

Set 就像 JavaScript 的 `Set`——元素唯一、无序。Redis 的 Set 还额外支持集合运算（交集、并集、差集），这可是 JS Set 没有的超能力。

**编程比喻**：Set 就像一个 VIP 名单——每个人只能出现一次，而且你可以快速判断某人在不在名单上。

```js
const Redis = require('ioredis');
const redis = new Redis();

async function setDemo() {
  // ========== SADD：添加成员 ==========
  await redis.sadd('user:1:tags', '前端', 'Node.js', 'TypeScript', 'Redis');
  await redis.sadd('user:2:tags', '后端', 'Java', 'Redis', 'Docker');

  // ========== SMEMBERS：获取所有成员 ==========
  const user1Tags = await redis.smembers('user:1:tags');
  console.log('用户1的标签：', user1Tags);

  // ========== SISMEMBER：判断成员是否存在 ==========
  const hasFrontend = await redis.sismember('user:1:tags', '前端');
  console.log('用户1有"前端"标签？', hasFrontend === 1 ? '是' : '否'); // 是

  // ========== SCARD：获取集合大小 ==========
  const count = await redis.scard('user:1:tags');
  console.log('用户1的标签数量：', count);

  // ========== SREM：移除成员 ==========
  await redis.srem('user:1:tags', 'TypeScript');
  console.log('移除TS后：', await redis.smembers('user:1:tags'));

  // ========== SUNION：并集 ==========
  const allTags = await redis.sunion('user:1:tags', 'user:2:tags');
  console.log('两个用户的所有标签：', allTags);

  // ========== SINTER：交集（共同兴趣！） ==========
  const commonTags = await redis.sinter('user:1:tags', 'user:2:tags');
  console.log('共同标签：', commonTags); // ['Redis']

  // ========== SDIFF：差集 ==========
  const onlyUser1 = await redis.sdiff('user:1:tags', 'user:2:tags');
  console.log('用户1独有的标签：', onlyUser1);

  // ========== SUNIONSTORE / SINTERSTORE：将结果保存到新集合 ==========
  await redis.sinterstore('common:interests', 'user:1:tags', 'user:2:tags');
  console.log('保存的共同兴趣：', await redis.smembers('common:interests'));

  // ========== SRANDMEMBER：随机获取成员（抽奖！） ==========
  await redis.sadd('lottery:participants', '张三', '李四', '王五', '赵六', '孙七');
  const winner = await redis.srandmember('lottery:participants');
  console.log('🎉 中奖者：', winner);

  // 随机抽取2个（不弹出）
  const twoWinners = await redis.srandmember('lottery:participants', 2);
  console.log('🎉 两位幸运儿：', twoWinners);

  // ========== SPOP：随机弹出（抽奖并移除） ==========
  const popped = await redis.spop('lottery:participants');
  console.log('弹出并移除：', popped);

  // 清理
  await redis.del('user:1:tags', 'user:2:tags', 'common:interests', 'lottery:participants');
  await redis.quit();
}

setDemo();
```

## 3.5 Sorted Set（有序集合）：排行榜的秘密

Sorted Set（简称 ZSet）是 Redis 最独特的数据结构之一。它像 Set 一样成员唯一，但每个成员都有一个**分数（score）**，Redis 会根据分数自动排序。

**编程比喻**：ZSet 就像一个自带排序功能的 Map——key 是成员，value 是分数，而且永远按分数排好序。就像学校的排名榜，每次考试完自动重新排名。

```js
const Redis = require('ioredis');
const redis = new Redis();

async function sortedSetDemo() {
  // ========== ZADD：添加成员及分数 ==========
  await redis.zadd('leaderboard',
    100, 'Alice',
    85, 'Bob',
    92, 'Charlie',
    78, 'David',
    95, 'Eve',
    88, 'Frank',
    105, 'Grace'
  );

  // ========== ZCARD：成员数量 ==========
  console.log('参与者数量：', await redis.zcard('leaderboard')); // 7

  // ========== ZSCORE：获取某个成员的分数 ==========
  const aliceScore = await redis.zscore('leaderboard', 'Alice');
  console.log('Alice 的分数：', aliceScore); // 100

  // ========== ZRANK / ZREVRANK：获取排名（从0开始） ==========
  const aliceRank = await redis.zrank('leaderboard', 'Alice');
  console.log('Alice 排名（升序）：', aliceRank + 1); // 排名6

  const aliceRevRank = await redis.zrevrank('leaderboard', 'Alice');
  console.log('Alice 排名（降序/实际排名）：', aliceRevRank + 1); // 排名2

  // ========== ZRANGE：按排名范围获取（升序） ==========
  const all = await redis.zrange('leaderboard', 0, -1, 'WITHSCORES');
  console.log('所有成员（升序）：', all);

  // ========== ZREVRANGE：按排名范围获取（降序） ==========
  const top3 = await redis.zrevrange('leaderboard', 0, 2, 'WITHSCORES');
  console.log('🏆 前3名：', top3);
  // ['Grace', '105', 'Alice', '100', 'Eve', '95']

  // ========== ZRANGEBYSCORE：按分数范围获取 ==========
  const range80to95 = await redis.zrangebyscore('leaderboard', 80, 95, 'WITHSCORES');
  console.log('80-95分的玩家：', range80to95);

  // ========== ZINCRBY：增加分数 ==========
  await redis.zincrby('leaderboard', 20, 'Bob'); // Bob +20分
  console.log('Bob 加分后：', await redis.zscore('leaderboard', 'Bob')); // 105

  // ========== ZCOUNT：统计分数范围内的成员数 ==========
  const countAbove90 = await redis.zcount('leaderboard', 90, '+inf');
  console.log('90分以上的人数：', countAbove90);

  // ========== ZREM：移除成员 ==========
  await redis.zrem('leaderboard', 'David');
  console.log('移除David后剩余：', await redis.zcard('leaderboard'));

  // ========== ZREMRANGEBYSCORE：按分数范围移除 ==========
  // await redis.zremrangebyscore('leaderboard', 0, 50); // 移除50分以下的

  // 清理
  await redis.del('leaderboard');
  await redis.quit();
}

sortedSetDemo();
```

### 实战：游戏排行榜系统

```js
const Redis = require('ioredis');
const redis = new Redis();

class GameLeaderboard {
  constructor(redis, gameName) {
    this.redis = redis;
    this.key = `leaderboard:${gameName}`;
  }

  // 提交分数（取最高分）
  async submitScore(playerId, score) {
    const currentScore = await this.redis.zscore(this.key, playerId);
    if (!currentScore || score > parseFloat(currentScore)) {
      await this.redis.zadd(this.key, score, playerId);
      return true; // 更新了分数
    }
    return false; // 分数没有提高
  }

  // 获取 Top N
  async getTopN(n = 10) {
    const results = await this.redis.zrevrange(this.key, 0, n - 1, 'WITHSCORES');
    const leaderboard = [];
    for (let i = 0; i < results.length; i += 2) {
      leaderboard.push({
        rank: i / 2 + 1,
        player: results[i],
        score: parseInt(results[i + 1]),
      });
    }
    return leaderboard;
  }

  // 获取玩家排名
  async getPlayerRank(playerId) {
    const rank = await this.redis.zrevrank(this.key, playerId);
    const score = await this.redis.zscore(this.key, playerId);
    if (rank === null) return null;
    return { rank: rank + 1, score: parseInt(score) };
  }

  // 获取排名在某个玩家前后各N名的玩家
  async getNeighbors(playerId, n = 2) {
    const rank = await this.redis.zrevrank(this.key, playerId);
    if (rank === null) return null;

    const start = Math.max(0, rank - n);
    const end = rank + n;
    const results = await this.redis.zrevrange(this.key, start, end, 'WITHSCORES');

    const neighbors = [];
    for (let i = 0; i < results.length; i += 2) {
      neighbors.push({
        rank: start + i / 2 + 1,
        player: results[i],
        score: parseInt(results[i + 1]),
        isCurrentPlayer: results[i] === playerId,
      });
    }
    return neighbors;
  }
}

async function main() {
  const lb = new GameLeaderboard(redis, 'space-shooter');

  // 模拟玩家提交分数
  const players = [
    { id: 'player:dragon', score: 12500 },
    { id: 'player:phoenix', score: 15800 },
    { id: 'player:tiger', score: 9200 },
    { id: 'player:wolf', score: 11000 },
    { id: 'player:eagle', score: 14300 },
    { id: 'player:shark', score: 13100 },
    { id: 'player:lion', score: 16200 },
    { id: 'player:bear', score: 8700 },
  ];

  for (const p of players) {
    await lb.submitScore(p.id, p.score);
  }

  // 查看排行榜
  console.log('=== 🏆 游戏排行榜 Top 5 ===');
  const top5 = await lb.getTopN(5);
  top5.forEach(item => {
    console.log(`  #${item.rank} ${item.player} - ${item.score}分`);
  });

  // 查看某个玩家的排名
  const wolfRank = await lb.getPlayerRank('player:wolf');
  console.log(`\nplayer:wolf 的排名：第${wolfRank.rank}名，${wolfRank.score}分`);

  // 查看某玩家附近的排名
  console.log('\nplayer:shark 附近排名：');
  const neighbors = await lb.getNeighbors('player:shark', 2);
  neighbors.forEach(n => {
    const marker = n.isCurrentPlayer ? ' ← 当前玩家' : '';
    console.log(`  #${n.rank} ${n.player} - ${n.score}分${marker}`);
  });

  await redis.del('leaderboard:space-shooter');
  await redis.quit();
}

main();
```

## 3.6 类型选择速查表

| 场景 | 推荐类型 | 示例 Key |
|------|----------|----------|
| 缓存一个 JSON 对象 | String | `cache:user:1001` |
| 存储用户详细信息 | Hash | `user:1001` |
| 最新消息列表 | List | `news:latest` |
| 任务队列 | List | `queue:emails` |
| 标签系统 | Set | `article:123:tags` |
| 共同好友/兴趣 | Set | `friends:userA` |
| 游戏排行榜 | Sorted Set | `leaderboard:game1` |
| 延时队列 | Sorted Set | `delayed:jobs` |
| 计数器 | String | `counter:page:123` |
| 分布式锁 | String | `lock:resource:xyz` |

## 3.7 小结

> **五虎将各有所长，选对了事半功倍**

- **String**：万金油，适合缓存、计数、分布式锁
- **Hash**：存对象首选，可以单独修改字段，节省带宽
- **List**：有序可重复，适合队列、栈、最新消息
- **Set**：自动去重，集合运算（交/并/差）是杀手锏
- **Sorted Set**：自动排序，排行榜和延时队列的不二之选
