### Series：一维带标签的数组

```python
import pandas as pd

# 从列表创建 Series
prices = pd.Series([89.0, 79.0, 99.0, 128.0], name="价格")
print(prices)
# 输出:
# 0     89.0
# 1     79.0
# 2     99.0
# 3    128.0
# Name: 价格, dtype: float64

# 自定义索引
prices = pd.Series(
    [89.0, 79.0, 99.0, 128.0],
    index=["Python入门", "Flask实战", "FastAPI指南", "算法导论"],
    name="价格"
)
print(prices["Flask实战"])  # 79.0
print(prices[prices > 90])  # 筛选大于 90 的
```

### 数据选择与过滤

## 27.2 数据清洗：脏数据是常态

现实世界的数据永远不会是完美的。缺失值、重复数据、格式不一致、异常值……数据清洗往往占据数据分析 60%-80% 的时间。把它想象成做菜前的洗菜、切菜——虽然不 glamorous，但不做这一步，后面的烹饪全白搭。

### 处理缺失值

```python
import pandas as pd
import numpy as np

# 创建包含缺失值的数据
df = pd.DataFrame({
    "姓名": ["张三", "李四", "王五", "赵六", "钱七"],
    "年龄": [28, np.nan, 35, 42, np.nan],
    "城市": ["北京", "上海", None, "广州", "深圳"],
    "薪资": [15000, 22000, np.nan, 18000, 25000],
})

# 查看缺失情况
print(df.isnull().sum())  # 每列缺失值个数
print(f"\n缺失比例:\n{df.isnull().mean() * 100:.1f}%")

# 策略 1: 删除含缺失值的行
df_dropped = df.dropna()

# 策略 2: 填充缺失值
df_filled = df.copy()
df_filled["年龄"] = df_filled["年龄"].fillna(df["年龄"].median())  # 用中位数填充
df_filled["城市"] = df_filled["城市"].fillna("未知")              # 用固定值填充
df_filled["薪资"] = df_filled["薪资"].interpolate()               # 插值填充

# 策略 3: 向前/向后填充（适合时间序列）
df_ffill = df.fillna(method="ffill")  # 用前一个有效值填充
```

## 27.3 数据聚合与合并

### groupby：分组统计

`groupby` 是数据分析中最强大的操作之一。它的逻辑很简单：把数据按某个字段分组，然后对每组分别做统计。就像把一个班级的学生按性别分组，分别计算平均成绩。

```python
import pandas as pd

df = pd.DataFrame({
    "部门": ["技术", "技术", "市场", "市场", "技术", "市场"],
    "姓名": ["张三", "李四", "王五", "赵六", "钱七", "孙八"],
    "薪资": [25000, 22000, 18000, 20000, 28000, 15000],
    "绩效": [90, 85, 78, 88, 95, 72],
})

# 按部门分组，计算平均薪资和绩效
dept_stats = df.groupby("部门").agg(
    平均薪资=("薪资", "mean"),
    最高薪资=("薪资", "max"),
    平均绩效=("绩效", "mean"),
    人数=("姓名", "count"),
)
print(dept_stats)

# 多列分组
df2 = pd.DataFrame({
    "年份": [2023, 2023, 2024, 2024, 2023, 2024],
    "季度": ["Q1", "Q2", "Q1", "Q2", "Q1", "Q2"],
    "销售额": [100, 120, 150, 180, 110, 200],
})

result = df2.groupby(["年份", "季度"])["销售额"].sum().unstack()
print(result)
```

### merge：合并表格

`merge` 就像 SQL 中的 JOIN，把两个表格按照共同的列连接起来。

```python
# 订单表
orders = pd.DataFrame({
    "订单号": ["A001", "A002", "A003"],
    "客户ID": ["C01", "C02", "C01"],
    "金额": [299, 159, 89],
})

# 客户表
customers = pd.DataFrame({
    "客户ID": ["C01", "C02", "C03"],
    "姓名": ["张三", "李四", "王五"],
    "城市": ["北京", "上海", "广州"],
})

# 内连接（只保留两边都有的）
merged = orders.merge(customers, on="客户ID", how="inner")
print("内连接:")
print(merged)

# 左连接（保留左表所有行）
merged_left = orders.merge(customers, on="客户ID", how="left")
print("\n左连接:")
print(merged_left)
```

### pivot_table：数据透视表

如果你用过 Excel 的数据透视表，pandas 的 `pivot_table` 就是同样的概念。

## 27.6 实战：分析一份公开数据集

让我们用一个完整的项目来实践。我们将分析一份模拟的电商销售数据集，从数据加载到可视化报告。

### 完整项目代码

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime, timedelta

# ========== 第一步：生成模拟数据 ==========
def generate_sales_data(n: int = 5000) -> pd.DataFrame:
    """生成模拟的电商销售数据"""
    np.random.seed(42)

    categories = ["电子产品", "图书", "服装", "食品", "家居"]
    regions = ["华东", "华南", "华北", "西南", "华中"]

    dates = pd.date_range("2024-01-01", "2024-12-31", freq="h")
    random_dates = np.random.choice(dates, size=n)

    data = {
        "订单日期": random_dates,
        "商品类别": np.random.choice(categories, size=n),
        "销售区域": np.random.choice(regions, size=n),
        "销售金额": np.round(np.random.lognormal(mean=5, sigma=1, size=n), 2),
        "数量": np.random.randint(1, 20, size=n),
    }
    df = pd.DataFrame(data)

    # 添加一些缺失值（模拟真实场景）
    mask = np.random.random(n) < 0.05
    df.loc[mask, "销售金额"] = np.nan

    return df

# ========== 第二步：数据清洗 ==========
def clean_data(df: pd.DataFrame) -> pd.DataFrame:
    """清洗数据"""
    print(f"原始数据: {df.shape[0]} 行")
    print(f"缺失值统计:\n{df.isnull().sum()}")

    # 填充缺失的销售金额（用该类别的中位数）
    df["销售金额"] = df.groupby("商品类别")["销售金额"].transform(
        lambda x: x.fillna(x.median())
    )

    # 转换日期
    df["订单日期"] = pd.to_datetime(df["订单日期"])

    # 添加衍生列
    df["月份"] = df["订单日期"].dt.month
    df["星期"] = df["订单日期"].dt.dayofweek
    df["季度"] = df["订单日期"].dt.quarter

    print(f"\n清洗后: {df.shape[0]} 行, 缺失值: {df.isnull().sum().sum()}")
    return df

# ========== 第三步：分析与可视化 ==========
def analyze_and_visualize(df: pd.DataFrame):
    """分析数据并生成图表"""
    # 设置图表风格
    plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei"]
    plt.rcParams["axes.unicode_minus"] = False
    sns.set_theme(style="whitegrid")

    fig, axes = plt.subplots(2, 2, figsize=(16, 12))
    fig.suptitle("2024 年电商销售分析报告", fontsize=18, fontweight="bold")

    # 图 1: 各类别销售额
    category_sales = df.groupby("商品类别")["销售金额"].sum().sort_values(ascending=False)
    axes[0, 0].bar(category_sales.index, category_sales.values, color=sns.color_palette("viridis", len(category_sales)))
    axes[0, 0].set_title("各类别销售总额")
    axes[0, 0].set_ylabel("销售额")
    for i, v in enumerate(category_sales.values):
        axes[0, 0].text(i, v + 5000, f"¥{v/10000:.1f}万", ha="center", fontsize=9)

    # 图 2: 月度销售趋势
    monthly_sales = df.groupby("月份")["销售金额"].sum()
    axes[0, 1].plot(monthly_sales.index, monthly_sales.values, marker="o", linewidth=2)
    axes[0, 1].set_title("月度销售趋势")
    axes[0, 1].set_xlabel("月份")
    axes[0, 1].set_ylabel("销售额")
    axes[0, 1].set_xticks(range(1, 13))

    # 图 3: 区域销售占比
    region_sales = df.groupby("销售区域")["销售金额"].sum()
    axes[1, 0].pie(region_sales.values, labels=region_sales.index,
                    autopct="%1.1f%%", startangle=90, colors=sns.color_palette("pastel"))
    axes[1, 0].set_title("区域销售占比")

    # 图 4: 各类别×区域的交叉分析热力图
    cross_table = df.pivot_table(
        values="销售金额", index="商品类别",
        columns="销售区域", aggfunc="sum"
    )
    sns.heatmap(cross_table, annot=True, fmt=".0f", cmap="YlOrRd", ax=axes[1, 1])
    axes[1, 1].set_title("类别×区域 销售热力图")

    plt.tight_layout()
    plt.savefig("sales_analysis.png", dpi=150, bbox_inches="tight")
    plt.show()

# ========== 第四步：生成文字报告 ==========
def generate_report(df: pd.DataFrame) -> str:
    """生成分析摘要报告"""
    total_sales = df["销售金额"].sum()
    avg_order = df["销售金额"].mean()
    top_category = df.groupby("商品类别")["销售金额"].sum().idxmax()
    top_region = df.groupby("销售区域")["销售金额"].sum().idxmax()
    best_month = df.groupby("月份")["销售金额"].sum().idxmax()

    report = f"""
{'='*50}
  2024 年电商销售分析报告
{'='*50}

数据概况:
  总订单数: {len(df):,}
  销售总额: ¥{total_sales:,.2f}
  平均订单金额: ¥{avg_order:,.2f}

关键发现:
  最畅销类别: {top_category}
  最强区域: {top_region}
  销售最好月份: {best_month}月

月度销售:
"""
    monthly = df.groupby("月份")["销售金额"].sum()
    for month, sales in monthly.items():
        bar = "█" * int(sales / monthly.max() * 30)
        report += f"  {month:2d}月: ¥{sales:>12,.2f} {bar}\n"

    return report

# ========== 主程序 ==========
if __name__ == "__main__":
    # 生成数据
    print("正在生成模拟数据...")
    raw_df = generate_sales_data(5000)

    # 清洗
    print("正在清洗数据...")
    clean_df = clean_data(raw_df)

    # 可视化
    print("正在生成图表...")
    analyze_and_visualize(clean_df)

    # 文字报告
    print(generate_report(clean_df))

    # 保存清洗后的数据
    clean_df.to_csv("cleaned_sales_data.csv", index=False, encoding="utf-8-sig")
    print("\n清洗后的数据已保存到 cleaned_sales_data.csv")
```

## 27.7 小结

本章我们学习了 Python 数据分析的核心工具和方法：

- **pandas** 是数据处理的核心，Series 和 DataFrame 覆盖了绝大部分数据结构需求
- **数据清洗**是数据分析的前置步骤，处理缺失值、重复和类型转换是基本功
- **groupby、merge、pivot_table** 是数据分析的三板斧，分别对应分组统计、表合并和透视分析
- **matplotlib 和 seaborn** 让数据可视化变得简单高效
- **Jupyter Notebook** 提供了交互式的分析环境

记住：数据分析不只是技术活，更重要的是提出好问题。工具帮你算，但问什么问题、如何解读结果，需要你的业务理解和思考能力。

**第 1 题：**

```python
import pandas as pd

df = pd.DataFrame({
    "姓名": ["小明", "小红", "小刚", "小丽", "小强",
             "小芳", "小伟", "小雪", "小龙", "小燕"],
    "语文": [85, 92, 78, 88, 95, 90, 82, 96, 75, 88],
    "数学": [90, 88, 85, 92, 78, 95, 88, 90, 82, 76],
    "英语": [82, 90, 92, 85, 88, 80, 95, 88, 90, 85],
})

# 计算平均分
df["平均分"] = df[["语文", "数学", "英语"]].mean(axis=1).round(1)

# 找最高分
top_student = df.loc[df["平均分"].idxmax()]
print(f"平均分最高的学生: {top_student['姓名']}，平均分: {top_student['平均分']}")

# 打印排名
print("\n成绩排名:")
print(df.sort_values("平均分", ascending=False)[["姓名", "平均分"]].to_string(index=False))
```

**第 2 题：**

```python
import pandas as pd
import numpy as np

df = pd.DataFrame({
    "姓名": ["张三", "李四", "王五", "赵六", "钱七"],
    "成绩": [85, np.nan, 92, np.nan, 78],
})

# 策略 1: 删除
s1 = df.dropna()
print(f"删除缺失值: {s1['成绩'].tolist()}")

# 策略 2: 用均值填充
s2 = df.copy()
s2["成绩"] = s2["成绩"].fillna(df["成绩"].mean())
print(f"均值填充: {s2['成绩'].round(1).tolist()}")

# 策略 3: 用中位数填充
s3 = df.copy()
s3["成绩"] = s3["成绩"].fillna(df["成绩"].median())
print(f"中位数填充: {s3['成绩'].tolist()}")

# 策略 4: 用前一个值填充
s4 = df.fillna(method="ffill")
print(f"前值填充: {s4['成绩'].tolist()}")
```

**第 6 题：**

```python
import pandas as pd

orders = pd.DataFrame({
    "订单号": [1, 2, 3, 4],
    "客户ID": ["A", "B", "C", "A"],
    "金额": [100, 200, 150, 300],
})

customers = pd.DataFrame({
    "客户ID": ["A", "B", "D"],
    "姓名": ["张三", "李四", "王五"],
})

# 内连接：只保留两个表都有客户 ID 的行（A 和 B）
inner = orders.merge(customers, on="客户ID", how="inner")
print("内连接（只保留匹配行）:")
print(inner)

# 左连接：保留左表所有行，右表没有匹配的填 NaN
left = orders.merge(customers, on="客户ID", how="left")
print("\n左连接（保留左表所有行）:")
print(left)
# 客户 C 在右表中不存在，姓名列会是 NaN
```

## 27.10 常见错误与避坑

**1. 忘记设置中文字体，图表中文显示为方块**

```python
# 在每个脚本或 Notebook 的开头加上
import matplotlib.pyplot as plt
plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei", "Arial Unicode MS"]
plt.rcParams["axes.unicode_minus"] = False  # 解决负号显示问题
```

**2. 链式赋值导致 SettingWithCopyWarning**

```python
# 容易出错
filtered = df[df["价格"] > 100]
filtered["折扣"] = filtered["价格"] * 0.9  # 可能产生警告

# 正确做法：使用 .copy()
filtered = df[df["价格"] > 100].copy()
filtered["折扣"] = filtered["价格"] * 0.9
```

**3. 混淆 loc 和 iloc**

```python
# loc 基于标签（索引值）
df.loc[2]      # 索引标签为 2 的行

# iloc 基于位置（整数位置）
df.iloc[2]     # 第 3 行（0 开始）

# 当索引不是连续整数时，两者结果可能完全不同
df = pd.DataFrame({"值": [10, 20, 30]}, index=[5, 2, 8])
print(df.loc[2])   # 值: 20（索引标签 2）
print(df.iloc[2])  # 值: 30（第 3 个位置）
```

**4. 大文件读取时内存溢出**

```python
# 错误：一次性读取超大 CSV，内存爆了
df = pd.read_csv("huge_file.csv")

# 正确：分块读取
chunks = pd.read_csv("huge_file.csv", chunksize=10000)
for chunk in chunks:
    process(chunk)  # 每次处理 10000 行
```

**5. 不检查数据类型就直接计算**

```python
# 数字可能是字符串类型
df["价格"].dtype  # object（字符串！）
df["价格"].sum()  # 不会报错，但结果是字符串拼接

# 先转换类型
df["价格"] = pd.to_numeric(df["价格"], errors="coerce")
df["价格"].sum()  # 现在是正确的数值求和
```

**6. 图表忘记调用 plt.show()**

在普通 Python 脚本中，`plt.show()` 是必须的，否则图表不会显示。在 Jupyter Notebook 中，如果使用 `%matplotlib inline`，则会自动显示。
