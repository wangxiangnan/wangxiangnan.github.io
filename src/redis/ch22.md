# 第22章：排行榜与计数器——Sorted Set的实战

## 引言：Redis 里的瑞士军刀

String 是万能胶水，Sorted Set 是瑞士军刀——不仅存储数据，还自动排序。想象一个永远自动排序的成绩单，新成绩加入就自动重排，瞬间告诉你"你排第几"。

Redis 还有 HyperLogLog（12KB 统计数十亿独立事件）、BitMap（极少空间管理海量布尔状态）和 Geo（附近的人/店）。

## 22.1 排行榜系统设计

Sorted Set 是专门为排行榜设计的数据结构：自动排序、O(log N) 插入、丰富的排名查询。

### 基本操作

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function basicSortedSetOps() {
  const boardKey = 'leaderboard:game:alltime';

  await redis.zadd(boardKey, 9850, 'Alice');
  await redis.zadd(boardKey, 8720, 'Bob');
  await redis.zadd(boardKey, 7630, 'Charlie');
  await redis.zadd(boardKey, 6540, 'Diana', 5450, 'Eve', 4360, 'Frank');

  // 增加分数
  await redis.zincrby(boardKey, 500, 'Charlie');
  console.log('Charlie 新分数:', await redis.zscore(boardKey, 'Charlie'));

  // 获取排名
  const aliceRank = await redis.zrevrank(boardKey, 'Alice');
  console.log(`Alice 排名: #${aliceRank + 1}`);

  // Top 3
  const top3 = await redis.zrevrange(boardKey, 0, 2, 'WITHSCORES');
  console.log('Top 3:', top3);

  // 总人数
  console.log('总玩家数:', await redis.zcard(boardKey));

  // 分数区间
  console.log('7000+分:', await redis.zrangebyscore(boardKey, 7000, 10000));
}
```

### 多维度排行榜

```js
class MultiDimensionLeaderboard {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  _getKey(game, dimension) {
    const now = new Date();
    switch (dimension) {
      case 'daily':
        return `lb:${game}:daily:${now.toISOString().slice(0, 10)}`;
      case 'weekly': {
        const monday = new Date(now);
        monday.setDate(now.getDate() - now.getDay() + 1);
        return `lb:${game}:weekly:${monday.toISOString().slice(0, 10)}`;
      }
      case 'monthly':
        return `lb:${game}:monthly:${now.toISOString().slice(0, 7)}`;
      case 'alltime':
        return `lb:${game}:alltime`;
    }
  }

  async recordScore(game, player, score) {
    const dims = ['daily', 'weekly', 'monthly', 'alltime'];
    const pipeline = this.redis.pipeline();

    for (const dim of dims) {
      const key = this._getKey(game, dim);
      pipeline.zadd(key, score, player);
      if (dim === 'daily') pipeline.expire(key, 86400 + 3600);
      else if (dim === 'weekly') pipeline.expire(key, 604800 + 3600);
      else if (dim === 'monthly') pipeline.expire(key, 2592000 + 3600);
    }

    await pipeline.exec();
  }

  async getTopN(game, dimension, n = 10) {
    const key = this._getKey(game, dimension);
    const results = await this.redis.zrevrange(key, 0, n - 1, 'WITHSCORES');

    const leaderboard = [];
    for (let i = 0; i < results.length; i += 2) {
      leaderboard.push({
        rank: i / 2 + 1,
        player: results[i],
        score: parseInt(results[i + 1]),
      });
    }
    return leaderboard;
  }

  async getPlayerContext(game, dimension, player, contextSize = 2) {
    const key = this._getKey(game, dimension);
    const rank = await this.redis.zrevrank(key, player);
    if (rank === null) return null;

    const score = await this.redis.zscore(key, player);
    const total = await this.redis.zcard(key);
    const start = Math.max(0, rank - contextSize);
    const end = rank + contextSize;
    const context = await this.redis.zrevrange(key, start, end, 'WITHSCORES');

    const players = [];
    for (let i = 0; i < context.length; i += 2) {
      players.push({
        rank: start + i / 2 + 1,
        player: context[i],
        score: parseInt(context[i + 1]),
        isCurrentPlayer: context[i] === player,
      });
    }

    return {
      player,
      score: parseInt(score),
      rank: rank + 1,
      totalPlayers: total,
      percentile: ((total - rank) / total * 100).toFixed(1),
      context: players,
    };
  }
}

async function leaderboardDemo() {
  const lb = new MultiDimensionLeaderboard(redis);
  const players = [
    { name: '剑圣', scores: [1500, 2300, 800, 1200] },
    { name: '暗影猎手', scores: [1800, 1600, 2100, 900] },
    { name: '风暴法师', scores: [2200, 1900, 1700, 2500] },
  ];

  for (const p of players) {
    for (const score of p.scores) await lb.recordScore('arena', p.name, score);
  }

  const topAll = await lb.getTopN('arena', 'alltime', 5);
  topAll.forEach((e) => console.log(`  #${e.rank} ${e.player}: ${e.score} 分`));

  const ctx = await lb.getPlayerContext('arena', 'alltime', '暗影猎手');
  if (ctx) console.log(`  ${ctx.player}: 排名 #${ctx.rank}/${ctx.totalPlayers}, 超越 ${ctx.percentile}%`);
}
```

```js
// score = 实际分数 * 10^13 + (MAX_TIMESTAMP - 当前时间戳)
function buildScore(realScore) {
  return realScore * 1e13 + (9999999999999 - Date.now());
}

function parseScore(compositeScore) {
  return Math.floor(compositeScore / 1e13);
}

const score1 = buildScore(1000);
await new Promise((r) => setTimeout(r, 100));
const score2 = buildScore(1000);
console.log('score1 > score2:', score1 > score2); // 先提交排名更高
```

### 分布式计数器

```js
class DistributedCounter {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.prefix = options.prefix || 'counter';
    this.shardCount = options.shardCount || 10;
  }

  _shardKey(name, requestId) {
    let hash = 0;
    for (let i = 0; i < requestId.length; i++) {
      hash = ((hash << 5) - hash + requestId.charCodeAt(i)) | 0;
    }
    return `${this.prefix}:${name}:shard:${Math.abs(hash) % this.shardCount}`;
  }

  async increment(name, requestId, amount = 1, ttlSeconds = 86400) {
    const key = this._shardKey(name, requestId);
    const result = await this.redis.incrby(key, amount);
    if (result === amount) await this.redis.expire(key, ttlSeconds);
    return result;
  }

  async getTotal(name) {
    const pipeline = this.redis.pipeline();
    for (let i = 0; i < this.shardCount; i++) {
      pipeline.get(`${this.prefix}:${name}:shard:${i}`);
    }
    const results = await pipeline.exec();
    let total = 0;
    for (const [, value] of results) {
      if (value) total += parseInt(value);
    }
    return total;
  }
}

const counter = new DistributedCounter(redis, { prefix: 'api', shardCount: 5 });
for (let i = 0; i < 1000; i++) {
  await counter.increment('calls:getUser', `req-${i}`);
}
console.log('总调用:', await counter.getTotal('calls:getUser')); // 1000
```

### 用户签到系统

```js
class CheckInSystem {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  _getKey(userId, year, month) {
    return `checkin:${userId}:${year}${String(month).padStart(2, '0')}`;
  }

  async checkIn(userId, date = new Date()) {
    const year = date.getFullYear();
    const month = date.getMonth() + 1;
    const day = date.getDate();
    const key = this._getKey(userId, year, month);

    const already = await this.redis.getbit(key, day);
    if (already) return { success: false, message: '今天已签到' };

    await this.redis.setbit(key, day, 1);
    await this.redis.expire(key, 90 * 86400);

    const streak = await this.getStreak(userId, date);
    return { success: true, streak };
  }

  async getStreak(userId, date = new Date()) {
    let streak = 0;
    const current = new Date(date);
    while (streak < 365) {
      const key = this._getKey(userId, current.getFullYear(), current.getMonth() + 1);
      if (!(await this.redis.getbit(key, current.getDate()))) break;
      streak++;
      current.setDate(current.getDate() - 1);
    }
    return streak;
  }

  async getMonthCheckIns(userId, year, month) {
    return this.redis.bitcount(this._getKey(userId, year, month));
  }
}

const checkin = new CheckInSystem(redis);
const result = await checkin.checkIn('user_42');
console.log('签到结果:', result);
```

## 22.6 实战：构建游戏排行榜系统

```js
class GameLeaderboardSystem {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  async recordMatch(playerId, score, location = null) {
    const pipeline = this.redis.pipeline();
    const dateKey = new Date().toISOString().slice(0, 10);
    const compositeScore = score * 1e13 + (9999999999999 - Date.now());

    pipeline.zadd('lb:global', compositeScore, playerId);
    pipeline.zadd(`lb:daily:${dateKey}`, compositeScore, playerId);
    pipeline.pfadd(`uv:daily:${dateKey}`, playerId);
    pipeline.incr(`stats:matches:${dateKey}`);

    if (location) {
      pipeline.geoadd('players:location', location.lng, location.lat, playerId);
    }

    await pipeline.exec();
  }

  async getLeaderboard(type = 'global', page = 1, pageSize = 20) {
    const key = type === 'global' ? 'lb:global' : `lb:daily:${type}`;
    const start = (page - 1) * pageSize;
    const results = await this.redis.zrevrange(key, start, start + pageSize - 1, 'WITHSCORES');
    const total = await this.redis.zcard(key);

    const entries = [];
    for (let i = 0; i < results.length; i += 2) {
      entries.push({
        rank: start + i / 2 + 1,
        playerId: results[i],
        score: Math.floor(parseFloat(results[i + 1]) / 1e13),
      });
    }

    return { entries, total, page, pageSize, totalPages: Math.ceil(total / pageSize) };
  }

  async getDailyStats() {
    const dateKey = new Date().toISOString().slice(0, 10);
    const [uv, matches] = await Promise.all([
      this.redis.pfcount(`uv:daily:${dateKey}`),
      this.redis.get(`stats:matches:${dateKey}`),
    ]);
    return { date: dateKey, uniquePlayers: uv, totalMatches: parseInt(matches) || 0 };
  }
}

const game = new GameLeaderboardSystem(redis);
const players = [
  { id: 'warrior_01', score: 2500, location: { lng: 116.40, lat: 39.92 } },
  { id: 'mage_02', score: 3200, location: { lng: 116.41, lat: 39.91 } },
  { id: 'archer_03', score: 2800, location: { lng: 116.39, lat: 39.93 } },
];

for (const p of players) await game.recordMatch(p.id, p.score, p.location);

const lb = await game.getLeaderboard('global');
lb.entries.forEach((e) => console.log(`#${e.rank} ${e.playerId}: ${e.score} 分`));
```

## 小结

| 数据结构 | 适用场景 | 内存效率 | 精度 |
|----------|----------|----------|------|
| Sorted Set | 排行榜 | 中等 | 精确 |

| BitMap | 签到/在线状态 | 极高 | 精确 |
