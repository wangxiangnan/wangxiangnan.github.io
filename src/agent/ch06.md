## 6.2 什么是 Embedding？— 文本的"语义指纹"

**Embedding = 将文本转换为固定长度的向量，语义相近的文本向量也相近。**

就像 Hash 函数，但有关键区别：

```python
# 传统 Hash（MD5/SHA）
hash("Docker部署")  # → "a3f2..."
hash("Docker部署!") # → "b7e1..." ← 微小变化，完全不同！

# Embedding
embed("Docker部署")   # → [0.8, 0.2, 0.1, ...]
embed("Docker部署!")  # → [0.79, 0.21, 0.12, ...] ← 微小变化，向量相近！
embed("容器部署")     # → [0.82, 0.18, 0.08, ...] ← 语义相近，向量相近！
embed("Python装饰器") # → [0.1, 0.9, 0.1, ...] ← 语义远，向量远！
```

## 6.3 余弦相似度 — 衡量语义距离

余弦相似度衡量两个向量的夹角，而不是距离。值域 [-1, 1]，越大越相似。

```python
import math

def cosine_similarity(a: list[float], b: list[float]) -> float:
    """计算余弦相似度
    
    cos(θ) = (A·B) / (|A|×|B|)
    
    - 1: 完全相同方向
    - 0: 正交（不相关）
    - -1: 完全相反方向
    """
    dot_product = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(x * x for x in b))
    
    if norm_a == 0 or norm_b == 0:
        return 0.0
    
    return dot_product / (norm_a * norm_b)

# 测试
print("=== 余弦相似度测试 ===")
print(f"Docker vs K8s: {cosine_similarity([0.8, 0.2, 0.1], [0.75, 0.25, 0.15]):.3f}")  # ~0.998
print(f"Docker vs Python: {cosine_similarity([0.8, 0.2, 0.1], [0.1, 0.9, 0.1]):.3f}")  # ~0.303
print(f"相同文本: {cosine_similarity([1, 0, 0], [1, 0, 0]):.3f}")  # 1.000
print(f"正交: {cosine_similarity([1, 0], [0, 1]):.3f}")  # 0.000
```

> 💡 **为什么用余弦而不是欧氏距离？**
>
> - 余弦只看方向，不看长度 → 不受文本长度影响
> - "Docker" 和 "一篇很长的Docker文档" 方向相同，余弦 ≈ 1
> - 但欧氏距离会因为长度差异而变大

## 6.7 RAG — 检索增强生成

**RAG = Retrieval Augmented Generation（检索增强生成）**

就像开卷考试 — 先查资料，再写答案。

```
RAG 流程:

用户问题
   ↓
[1] Embedding（问题转向量）
   ↓
[2] 向量搜索（找最相关的文档块）
   ↓
[3] 组装 Prompt（问题 + 检索到的文档）
   ↓
[4] LLM 生成回答
   ↓
最终答案 + 来源引用
```

### 为什么需要 RAG？


### 实操：OpenAI + Chroma

```python
# pip install openai chromadb
import os, chromadb
from openai import OpenAI
from dotenv import load_dotenv
load_dotenv()
client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

def embed(text):
    """生成文本Embedding"""
    return client.embeddings.create(
        model="text-embedding-3-small", input=text).data[0].embedding

# 文档数据
docs = [
    {"id":"d1","title":"FastAPI依赖注入","content":"Depends()实现依赖注入，支持函数和类。用于数据库连接、认证、配置等共享逻辑。","tags":["python"]},
    {"id":"d2","title":"Docker多阶段构建","content":"分离构建和运行环境，使用multi-stage build优化镜像大小。FROM builder → FROM runtime。","tags":["docker"]},
    {"id":"d3","title":"PostgreSQL索引","content":"B-tree索引适合等值和范围查询，Hash索引适合等值查询，GIN索引适合全文搜索。","tags":["db"]},
    {"id":"d4","title":"Redis缓存策略","content":"Cache Aside: 先查缓存，miss则查DB并回填。Write Through: 写缓存同时写DB。","tags":["redis"]},
    {"id":"d5","title":"K8s HPA","content":"Horizontal Pod Autoscaler根据CPU/内存使用率自动调整Pod副本数。","tags":["k8s"]},
    {"id":"d6","title":"Python asyncio","content":"async/await实现并发，适合I/O密集型。asyncio.gather并行执行多个协程。","tags":["python"]},
    {"id":"d7","title":"SQLAlchemy ORM","content":"relationship()处理表关联，joinedload()避免N+1查询问题。","tags":["python"]},
    {"id":"d8","title":"Nginx反向代理","content":"将请求分发到多个后端服务器，支持负载均衡、SSL终止、缓存。","tags":["nginx"]},
]

# 创建Chroma索引
cc = chromadb.Client()
col = cc.create_collection("docmind")
for d in docs:
    col.add(ids=[d["id"]],
            embeddings=[embed(f"{d['title']}\n{d['content']}")],
            documents=[d["content"]],
            metadatas=[{"title":d["title"],"tags":",".join(d["tags"])}])
print(f"已索引 {col.count()} 篇文档")

# 语义搜索
def search(q, n=3):
    r = col.query(query_embeddings=[embed(q)], n_results=n,
                  include=["documents","metadatas","distances"])
    return [{"title":r["metadatas"][0][i]["title"],
             "content":r["documents"][0][i],
             "rel":1-r["distances"][0][i]/2}
            for i in range(len(r["ids"][0]))]

# RAG 问答
def rag_answer(question: str, top_k: int = 3) -> dict:
    """完整的 RAG 流程"""
    
    # Step 1: 检索相关文档
    relevant_docs = search(question, top_k)
    
    # Step 2: 构建上下文
    context = "\n\n".join(
        f"## {d['title']}\n{d['content']}" for d in relevant_docs
    )
    
    # Step 3: 构建 Prompt
    system_prompt = f"""你是 DocMind，技术文档助手。
基于以下参考文档回答用户的问题。
如果参考文档中没有相关信息，请明确说明。

## 参考文档
{context}

## 回答规范
- 基于文档内容回答，不要编造
- 引用文档来源
- 如果不确定，标注 [UNCERTAIN]"""
    
    # Step 4: 调用 LLM
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": question}
        ],
        temperature=0.2  # 低温度，更确定
    )
    
    return {
        "answer": response.choices[0].message.content,
        "sources": [d["title"] for d in relevant_docs],
        "relevance_scores": [d["rel"] for d in relevant_docs]
    }

# 测试语义搜索
print("\n=== 语义搜索测试 ===")
for q in ["怎么让API更快", "数据库查询慢怎么办", "如何部署多实例"]:
    print(f"\n搜索: {q}")
    for r in search(q, 2):
        print(f"  → {r['title']} (相关度: {r['rel']:.2f})")

# 测试 RAG 问答
print("\n=== RAG 问答测试 ===")
for q in ["FastAPI依赖注入怎么用？", "如何减小Docker镜像？", "Redis缓存失效怎么办？"]:
    r = rag_answer(q)
    print(f"\nQ: {q}")
    print(f"A: {r['answer'][:150]}...")
    print(f"来源: {r['sources']}")
```

## 6.9 局限性和注意事项

- **长度限制**：512-8192 token → 需切分文档
- **语义漂移**：细粒度差异区分不好 → 混合搜索
- **领域知识**：通用模型不懂专有术语 → 领域微调
- **多语言**：跨语言效果差 → 多语言模型

```python
def chunk(text, size=500, overlap=50):
    """文档切分为重叠块"""
    chunks, start = [], 0
    while start < len(text):
        end = start + size
        if end < len(text):
            for sep in ["\n\n","\n","。",". "]:
                pos = text[start:end].rfind(sep)
                if pos > size//2: end = start+pos+len(sep); break
        c = text[start:end].strip()
        if c: chunks.append(c)
        start = end - overlap
    return chunks
```

> ⚠️ 生产推荐**混合搜索**：关键词 + 语义，合并结果。

## 6.10 实战：DocMind 索引系统

```python
import os, chromadb
from openai import OpenAI
from dotenv import load_dotenv
from dataclasses import dataclass
load_dotenv()

@dataclass
class Document:
    id:str; title:str; content:str; source:str; doc_type:str; tags:list

class DocMindIndexer:
    """DocMind 文档索引器"""
    def __init__(self, name="docmind"):
        self.client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        self.col = chromadb.Client().get_or_create_collection(name)
    
    def embed(self, text):
        return self.client.embeddings.create(
            model="text-embedding-3-small", input=text).data[0].embedding
    
    def chunk(self, text, size=500, overlap=50):
        chunks, start = [], 0
        while start < len(text):
            end = start + size
            if end < len(text):
                for sep in ["\n\n","\n","。",". "]:
                    pos = text[start:end].rfind(sep)
                    if pos > size//2: end=start+pos+len(sep); break
            c = text[start:end].strip()
            if c: chunks.append(c)
            start = end - overlap
        return chunks
    
    def add(self, doc):
        """添加文档到索引"""
        chunks = self.chunk(doc.content)
        for i, c in enumerate(chunks):
            self.col.add(ids=[f"{doc.id}_c{i}"],
                embeddings=[self.embed(f"{doc.title}\n{c}")],
                documents=[c],
                metadatas=[{"doc_id":doc.id,"title":doc.title,
                           "source":doc.source,"type":doc.doc_type,
                           "tags":",".join(doc.tags)}])
        print(f"'{doc.title}' 索引 ({len(chunks)}块)")
    
    def search(self, q, n=5, doc_type=None):
        """语义搜索"""
        r = self.col.query(query_embeddings=[self.embed(q)], n_results=n,
            where={"type":doc_type} if doc_type else None,
            include=["documents","metadatas","distances"])
        return [{"title":r["metadatas"][0][i]["title"],
                 "content":r["documents"][0][i],
                 "rel":1-r["distances"][0][i]/2}
                for i in range(len(r["ids"][0]))]
    
    def stats(self): return {"chunks":self.col.count()}

# 测试
idx = DocMindIndexer()
idx.add(Document("fa-deps","FastAPI依赖注入",
    "Depends()函数实现依赖注入..."*5,"docs/fa.md","guide",["python","fastapi"]))
idx.add(Document("dk-multi","Docker多阶段构建",
    "分离构建运行环境优化镜像..."*5,"docs/dk.md","guide",["docker","devops"]))

for r in idx.search("如何减小镜像"):
    print(f"  {r['title']} ({r['rel']:.2f})")
print(f"统计: {idx.stats()}")
```
