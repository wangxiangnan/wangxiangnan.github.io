# 第21章：Shell脚本进阶——从能用到好用

## 为什么需要进阶？

上一章你学会了Shell脚本的基础，能写出if-else和变量操作。但在真实的生产环境中，你需要更多武器：循环遍历文件列表、编写可复用的函数、正确处理错误、调试复杂问题。这就像从会写Hello World进化到会写工程项目——本章就是那个进化阶梯。

如果说第20章是学会了Shell的"语法"，那本章就是学习Shell的"工程实践"。我们会学习循环、函数、错误处理、调试技巧，最后写一个完整的自动化部署脚本。

## 21.1 循环结构——重复是计算机的超能力

循环是编程中最能体现"让机器干苦力活"的结构。Shell提供了四种循环方式，每一种都有它的用武之地。

```bash
# ============================
# for 循环——遍历列表
# ============================

# 遍历固定列表
for fruit in apple banana cherry; do
    echo "水果: $fruit"
done

# 遍历文件（通配符展开）
for file in /var/log/*.log; do
    echo "日志文件: $file ($(du -h "$file" | cut -f1))"
done

# 遍历数字序列
for i in $(seq 1 10); do
    echo "数字: $i"
done

# 遍历命令输出
for user in $(cut -d: -f1 /etc/passwd); do
    echo "系统用户: $user"
done

# 使用大括号展开
for i in {1..5}; do
    echo "编号: $i"
done

for letter in {a..z}; do
    echo "字母: $letter"
done

# ============================
# C风格 for 循环——经典三段式
# ============================
for ((i=0; i<10; i++)); do
    echo "索引: $i"
done

# 多变量
for ((i=0, j=10; i<j; i++, j--)); do
    echo "i=$i, j=$j"
done

# ============================
# while 循环——条件为真就继续
# ============================

# 基本while循环
count=1
while [ $count -le 5 ]; do
    echo "计数: $count"
    count=$((count + 1))
done

# 逐行读取文件（最常用的文件处理模式）
while IFS= read -r line; do
    echo "行内容: $line"
done < /etc/hosts
# IFS= 防止首尾空格被吃掉
# -r 防止反斜杠被解释为转义符

# 读取CSV文件（设置IFS为逗号）
while IFS=',' read -r name age email; do
    echo "姓名: $name, 年龄: $age, 邮箱: $email"
done < users.csv

# 无限循环（配合break使用）
while true; do
    read -p "请输入命令 (quit退出): " cmd
    if [ "$cmd" = "quit" ]; then
        break
    fi
    echo "执行: $cmd"
done

# ============================
# until 循环——条件为假就继续（while的反面）
# ============================
count=1
until [ $count -gt 5 ]; do
    echo "计数: $count"
    count=$((count + 1))
done

# 实用场景：等待服务启动
until curl -s http://localhost:8080/health > /dev/null 2>&1; do
    echo "等待服务启动..."
    sleep 2
done
echo "服务已就绪！"

# ============================
# 循环控制：break 和 continue
# ============================

# break: 跳出循环
for i in {1..100}; do
    if [ $i -eq 5 ]; then
        echo "到达5，退出循环"
        break
    fi
    echo $i
done

# continue: 跳过当前迭代
for i in {1..10}; do
    if [ $((i % 2)) -eq 0 ]; then
        continue    # 跳过偶数
    fi
    echo "奇数: $i"
done

# 嵌套循环中的break N / continue N
for i in {1..3}; do
    for j in {1..3}; do
        if [ $j -eq 2 ]; then
            break 2    # 跳出两层循环
        fi
        echo "i=$i, j=$j"
    done
done
```

## 21.2 case 语句——优雅的多分支选择

当你有超过3个分支要判断时，用if-elif-elif-elif就像用胶带绑电线——能用但丑。case语句是更优雅的选择，它就像其他语言中的switch/match。

```bash
# ============================
# 基本 case 结构
# ============================
cat > service_ctl.sh << 'SCRIPT'
#!/bin/bash

action="$1"

case "$action" in
    start)
        echo "启动服务..."
        systemctl start myapp
        ;;
    stop)
        echo "停止服务..."
        systemctl stop myapp
        ;;
    restart)
        echo "重启服务..."
        systemctl restart myapp
        ;;
    status)
        echo "查看状态..."
        systemctl status myapp
        ;;
    *)
        echo "用法: $0 {start|stop|restart|status}"
        exit 1
        ;;
esac
SCRIPT

# ============================
# 多模式匹配
# ============================
read -p "确认操作？(yes/no): " answer
case "$answer" in
    [Yy]|[Yy][Ee][Ss])
        echo "执行操作..."
        ;;
    [Nn]|[Nn][Oo])
        echo "操作已取消"
        ;;
    *)
        echo "无效输入"
        ;;
esac

# ============================
# 在菜单脚本中使用 case
# ============================
cat > menu.sh << 'SCRIPT'

while true; do
    show_menu

    case "$choice" in
        1)
            echo ""
            df -h
            ;;
        2)
            echo ""
            free -h
            ;;
        3)
            echo ""
            lscpu | head -15

        4)
            echo ""
            ip -brief addr

            echo ""

            exit 0
            ;;
```

## 21.3 函数——Shell脚本的子程序

函数是代码复用的基石。在Shell中，函数就像自定义命令——你可以在脚本中定义它们，然后像调用普通命令一样使用它们。

```bash
# ============================
# 函数定义（两种方式）
# ============================

# 方式一：function关键字
function greet {
    echo "Hello, $1!"
}

# 方式二：POSIX风格（更常用）
greet() {
    echo "Hello, $1!"
}

# 调用函数
greet "World"        # Hello, World!
greet "Linux"        # Hello, Linux!

# ============================
# 函数参数
# ============================
calculate() {
    local a=$1
    local b=$2
    local op=$3

    case "$op" in
        add) echo $((a + b)) ;;
        sub) echo $((a - b)) ;;
        mul) echo $((a * b)) ;;
        div) echo $((a / b)) ;;
        *)   echo "未知操作: $op" ;;
    esac
}

result=$(calculate 10 3 add)
echo "10 + 3 = $result"       # 10 + 3 = 13

# ============================
# 局部变量（local）——极其重要！
# ============================
count=0                         # 全局变量

increment() {
    local count=$1              # 局部变量，不影响全局的count
    count=$((count + 1))
    echo $count
}

result=$(increment 5)
echo "函数返回: $result"        # 6
echo "全局count: $count"        # 0（没有被修改）

# ============================
# 函数返回值
# ============================
# 方式一：通过echo返回（推荐，可以返回字符串）
get_hostname() {
    hostname | tr '[:upper:]' '[:lower:]'
}
my_host=$(get_hostname)

# 方式二：通过return返回状态码（只能返回0-255的数字）
is_admin() {
    if [ "$(id -u)" -eq 0 ]; then
        return 0    # 成功（是admin）
    else
        return 1    # 失败（不是admin）
    fi
}

if is_admin; then
    echo "你是管理员"
else
    echo "你不是管理员"
fi

# ============================
# 实用函数库模式
# ============================
cat > lib.sh << 'EOF'
#!/bin/bash
# lib.sh - 通用工具函数库

# 日志函数
log() {
    local level=$1
    shift
    local msg="$*"
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[$timestamp] [$level] $msg"
}

log_info()  { log "INFO"  "$@"; }
log_warn()  { log "WARN"  "$@"; }
log_error() { log "ERROR" "$@"; }

# 检查命令是否存在
require_cmd() {
    local cmd=$1
    if ! command -v "$cmd" &>/dev/null; then
        log_error "缺少必要命令: $cmd"
        exit 1
    fi
}

# 确认操作
confirm() {
    local prompt="${1:-确认执行？}"
    read -p "$prompt (y/n): " answer
    [[ "$answer" =~ ^[Yy] ]]
}

# 重试执行
retry() {
    local max_attempts=$1
    local delay=$2
    shift 2
    local attempt=1

    while [ $attempt -le $max_attempts ]; do
        if "$@"; then
            return 0
        fi
        log_warn "第 $attempt/$max_attempts 次尝试失败，${delay}秒后重试..."
        sleep "$delay"
        attempt=$((attempt + 1))
    done

    log_error "命令在 $max_attempts 次尝试后仍然失败: $*"
    return 1
}
EOF

# 在其他脚本中使用
cat > myscript.sh << 'EOF'
#!/bin/bash

# 加载函数库
source "$(dirname "$0")/lib.sh"

# 使用库函数
require_cmd "curl"
require_cmd "jq"

log_info "开始执行任务..."

if confirm "是否继续？"; then
    retry 3 5 curl -s https://api.example.com/data
    log_info "任务完成"
else
    log_warn "用户取消操作"
fi
EOF
```

## 21.4 命令替换与算术运算

```bash
# ============================
# 命令替换——把命令输出当字符串用
# ============================

# 推荐方式：$()
today=$(date '+%Y-%m-%d')
file_count=$(ls -1 | wc -l)
ip_address=$(hostname -I | awk '{print $1}')

echo "今天是 $today，目录中有 $file_count 个文件"

# 旧方式：反引号 ``（不推荐，难嵌套，难阅读）
today=`date '+%Y-%m-%d'`

# $() 的优势：可以嵌套
backup_name="backup_$(hostname)_$(date '+%Y%m%d').tar.gz"
echo $backup_name    # backup_myserver_20240115.tar.gz

# ============================
# 算术运算——Shell的计算器
# ============================

# 方式一：$(( ))（推荐，Bash内置）
a=10
b=3
echo $((a + b))      # 13
echo $((a - b))      # 7
echo $((a * b))      # 30
echo $((a / b))      # 3（整数除法）
echo $((a % b))      # 1（取余）
echo $((a ** b))     # 1000（幂运算）
echo $((a > b))      # 1（真为1，假为0）

# 支持位运算
echo $((a << 2))     # 40（左移2位）
echo $((a >> 1))     # 5（右移1位）
echo $((a & b))      # 2（按位与）
echo $((a | b))      # 11（按位或）

# 方式二：expr（POSIX标准，更通用但更慢）
expr 10 + 3          # 13
expr 10 \* 3         # 30（* 需要转义）
expr 10 % 3          # 1

# 方式三：bc（支持浮点运算）
echo "10 / 3" | bc -l              # 3.33333333333333
echo "scale=2; 10 / 3" | bc        # 3.33
echo "scale=4; 3.14 * 2.5" | bc    # 7.8500

# 实用示例：计算磁盘使用百分比
used=$(df / | awk 'NR==2 {print $3}')
total=$(df / | awk 'NR==2 {print $2}')
echo "scale=1; $used * 100 / $total" | bc
```

```bash
command > /dev/null 2>&1               # 丢弃所有输出
```

## 21.6 错误处理——让脚本健壮如牛

生产环境的脚本不能假设一切顺利。磁盘可能满、网络可能断、命令可能不存在。错误处理就是把"可能出问题的地方"提前准备好。

```bash
# ============================
# set 命令——Shell的安全模式
# ============================

# set -e: 遇到错误立即退出（Exit on error）
set -e
# false                # 这条命令返回非0，脚本直接退出
# echo "这行不会执行"

# set -u: 使用未定义变量时报错
set -u
# echo $UNDEFINED_VAR  # 报错: UNDEFINED_VAR: unbound variable

# set -o pipefail: 管道中任何命令失败都视为失败
set -o pipefail
# 默认行为：管道的退出状态是最后一个命令的状态
# 开启pipefail后：管道中任何一个命令失败，整条管道就算失败

# 推荐的脚本开头（黄金三件套）
set -euo pipefail

# ============================
# trap——信号捕获与清理
# ============================

# 脚本退出时清理临时文件
tmpfile=$(mktemp)
trap "rm -f '$tmpfile'" EXIT

# 捕获 Ctrl+C（SIGINT）
trap 'echo ""; echo "收到中断信号，正在清理..."; cleanup; exit 1' INT

# 实用：脚本锁文件模式
LOCKFILE="/tmp/myscript.lock"

cleanup() {
    rm -f "$LOCKFILE"
}

trap cleanup EXIT

# 检查是否已有实例在运行
if [ -f "$LOCKFILE" ]; then
    pid=$(cat "$LOCKFILE")
    if kill -0 "$pid" 2>/dev/null; then
        echo "脚本已在运行 (PID: $pid)"
        exit 1
    else
        echo "发现过期的锁文件，清理中..."
        rm -f "$LOCKFILE"
    fi
fi

echo $$ > "$LOCKFILE"
echo "脚本正在运行..."
sleep 10
echo "脚本执行完毕"

# ============================
# 完整的错误处理模式
# ============================
cat > robust_script.sh << 'EOF'
#!/usr/bin/env bash

set -euo pipefail

readonly SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
readonly SCRIPT_NAME="$(basename "$0")"
readonly LOG_FILE="/tmp/${SCRIPT_NAME%.sh}.log"

log() {
    local level="$1"; shift
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [$level] $*" | tee -a "$LOG_FILE"
}

cleanup() {
    local exit_code=$?
    log "INFO" "脚本退出，退出码: $exit_code"
    exit $exit_code
}

error_handler() {
    local line_number="$1"
    log "ERROR" "脚本在第 $line_number 行出错"
    log "ERROR" "最后执行的命令: $BASH_COMMAND"
}

trap cleanup EXIT
trap 'error_handler $LINENO' ERR

main() {
    log "INFO" "脚本开始执行"
    log "INFO" "脚本目录: $SCRIPT_DIR"
    log "INFO" "日志文件: $LOG_FILE"

    log "INFO" "执行任务1..."
    sleep 1

    log "INFO" "执行任务2..."
    sleep 1

    log "INFO" "脚本执行完成"
}

main "$@"
EOF
```

## 21.7 脚本调试——当脚本不听话时

```bash
# ============================
# bash -x: 打印每条执行的命令
# ============================
bash -x myscript.sh
# 输出类似：
# + name=Linux
# + echo 'Hello, Linux'
# Hello, Linux

# ============================
# bash -n: 语法检查（不执行）
# ============================
bash -n myscript.sh
# 如果有语法错误会报错，没有则静默

# ============================
# set -x / set +x: 在脚本内部开关调试
# ============================
cat > debug_demo.sh << 'EOF'
#!/bin/bash

echo "正常执行..."

set -x          # 开启调试
x=10
y=20
result=$((x + y))
echo "结果: $result"
set +x          # 关闭调试

echo "继续正常执行..."
EOF

# ============================
# PS4: 自定义调试前缀
# ============================
export PS4='+(${BASH_SOURCE}:${LINENO}): ${FUNCNAME[0]:+${FUNCNAME[0]}(): }'
bash -x myscript.sh
# 输出类似：
# +(myscript.sh:5): main(): name=Linux

# ============================
# 调试技巧汇总
# ============================

# 1. 在关键位置插入调试输出
echo "DEBUG: var=$var" >&2    # 输出到stderr，不干扰正常输出

# 2. 检查命令类型
type ls         # 显示ls是别名、函数还是外部命令
which curl      # 显示curl的路径

# 3. 使用 shellcheck 静态分析
shellcheck myscript.sh
```

## 21.9 实战：自动化部署脚本

让我们把所有进阶知识综合运用，编写一个生产级别的自动化部署脚本。这个脚本模拟了一个真实的CI/CD流程：拉取代码、构建、备份、部署、健康检查、通知。

```bash
#!/usr/bin/env bash
# ============================================================
# deploy.sh - 自动化部署脚本
# 功能: 拉取代码 -> 构建 -> 备份旧版本 -> 部署新版本 -> 健康检查 -> 通知
# 用法: ./deploy.sh [--branch main] [--env production] [--rollback]
# ============================================================

set -euo pipefail

# ============================
# 配置
# ============================
readonly APP_NAME="myapp"
readonly DEPLOY_DIR="/opt/${APP_NAME}"
readonly BACKUP_DIR="/opt/backups/${APP_NAME}"
readonly LOG_DIR="/var/log/${APP_NAME}"
readonly REPO_URL="https://github.com/example/myapp.git"
readonly HEALTH_CHECK_URL="http://localhost:3000/health"
readonly MAX_HEALTH_RETRIES=30
readonly HEALTH_CHECK_INTERVAL=2

# 默认值
BRANCH="main"
ENVIRONMENT="production"
ROLLBACK=false
BUILD_DIR=""

# 颜色
readonly RED='\033[0;31m'
readonly GREEN='\033[0;32m'
readonly YELLOW='\033[1;33m'
readonly BLUE='\033[0;34m'
readonly NC='\033[0m'

# ============================
# 工具函数
# ============================
log_info()  { echo -e "${GREEN}[INFO]${NC}  $(date '+%H:%M:%S') $*"; }
log_warn()  { echo -e "${YELLOW}[WARN]${NC}  $(date '+%H:%M:%S') $*"; }
log_error() { echo -e "${RED}[ERROR]${NC} $(date '+%H:%M:%S') $*"; }
log_step()  { echo -e "\n${BLUE}>>> $*${NC}"; }

error_exit() {
    log_error "$*"
    log_error "部署失败！如需回滚请运行: $0 --rollback"
    exit 1
}

cleanup() {
    local exit_code=$?
    if [ $exit_code -ne 0 ]; then
        log_warn "部署过程中断（退出码: $exit_code）"
        log_warn "清理临时文件..."
        rm -rf "/tmp/${APP_NAME}_build_$$"
    fi
}
trap cleanup EXIT

# ============================
# 参数解析
# ============================
parse_args() {
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --branch)
                BRANCH="$2"
                shift 2
                ;;
            --env)
                ENVIRONMENT="$2"
                shift 2
                ;;
            --rollback)
                ROLLBACK=true
                shift
                ;;
            --help|-h)
                echo "用法: $0 [选项]"
                echo "  --branch <name>   指定分支 (默认: main)"
                echo "  --env <name>      指定环境 (默认: production)"
                echo "  --rollback        回滚到上一个版本"
                echo "  --help            显示帮助"
                exit 0
                ;;
            *)
                error_exit "未知参数: $1"
                ;;
        esac
    done
}

# ============================
# 部署步骤
# ============================

step_preflight() {
    log_step "步骤 1/6: 预检查"

    local required_cmds=("git" "node" "npm" "tar")
    for cmd in "${required_cmds[@]}"; do
        if ! command -v "$cmd" &>/dev/null; then
            error_exit "缺少必要命令: $cmd"
        fi
    done
    log_info "所有必要命令已就绪"

    if [ ! -w "$(dirname "$DEPLOY_DIR")" ]; then
        error_exit "没有写入权限: $(dirname "$DEPLOY_DIR")"
    fi

    mkdir -p "$DEPLOY_DIR" "$BACKUP_DIR" "$LOG_DIR"
    log_info "目录结构已就绪"
}

step_pull_code() {
    log_step "步骤 2/6: 拉取代码"

    BUILD_DIR="/tmp/${APP_NAME}_build_$$"
    rm -rf "$BUILD_DIR"

    log_info "克隆分支 ${BRANCH} ..."
    git clone --depth 1 --branch "$BRANCH" "$REPO_URL" "$BUILD_DIR" 2>&1 | \
        while read -r line; do echo "    $line"; done

    local commit_hash commit_msg
    commit_hash=$(git -C "$BUILD_DIR" rev-parse --short HEAD)
    commit_msg=$(git -C "$BUILD_DIR" log -1 --format='%s')

    log_info "最新提交: ${commit_hash} - ${commit_msg}"
    echo "$commit_hash" > "${BUILD_DIR}/.deploy_commit"
}

step_build() {
    log_step "步骤 3/6: 构建项目"

    cd "$BUILD_DIR"

    log_info "安装依赖..."
    npm ci --production 2>&1 | tail -3

    log_info "执行构建..."
    if [ -f "package.json" ] && grep -q '"build"' package.json; then
        npm run build 2>&1 | tail -5
    else
        log_warn "未找到build脚本，跳过构建步骤"
    fi

    log_info "构建完成"
}

step_backup() {
    log_step "步骤 4/6: 备份当前版本"

    if [ -d "$DEPLOY_DIR" ] && [ "$(ls -A "$DEPLOY_DIR" 2>/dev/null)" ]; then
        local backup_name="${APP_NAME}_$(date '+%Y%m%d_%H%M%S').tar.gz"
        local backup_path="${BACKUP_DIR}/${backup_name}"

        log_info "备份到: ${backup_path}"
        tar -czf "$backup_path" -C "$(dirname "$DEPLOY_DIR")" "$(basename "$DEPLOY_DIR")"

        # 只保留最近5个备份
        ls -t "${BACKUP_DIR}"/${APP_NAME}_*.tar.gz 2>/dev/null | tail -n +6 | xargs -r rm -f
        log_info "备份完成（保留最近5个备份）"
    else
        log_info "首次部署，无需备份"
    fi
}

step_deploy() {
    log_step "步骤 5/6: 部署新版本"

    if systemctl is-active --quiet "$APP_NAME" 2>/dev/null; then
        log_info "停止旧服务..."
        sudo systemctl stop "$APP_NAME"
    fi

    log_info "同步文件到 ${DEPLOY_DIR} ..."
    if command -v rsync &>/dev/null; then
        rsync -a --delete \
            --exclude='.git' \
            --exclude='node_modules' \
            --exclude='.env' \
            "$BUILD_DIR/" "$DEPLOY_DIR/"
    else
        cp -r "$BUILD_DIR"/* "$DEPLOY_DIR"/
    fi

    chmod -R 755 "$DEPLOY_DIR"

    if [ -f "${DEPLOY_DIR}/.env.${ENVIRONMENT}" ]; then
        cp "${DEPLOY_DIR}/.env.${ENVIRONMENT}" "${DEPLOY_DIR}/.env"
        log_info "已加载 ${ENVIRONMENT} 环境配置"
    fi

    log_info "启动服务..."
    if [ -f "${DEPLOY_DIR}/systemd/${APP_NAME}.service" ]; then
        sudo cp "${DEPLOY_DIR}/systemd/${APP_NAME}.service" /etc/systemd/system/
        sudo systemctl daemon-reload
    fi
    sudo systemctl start "$APP_NAME" 2>/dev/null || \
        log_warn "systemd服务未配置，请手动启动"

    rm -rf "$BUILD_DIR"
    log_info "部署文件同步完成"
}

step_health_check() {
    log_step "步骤 6/6: 健康检查"

    log_info "等待服务启动..."
    local attempt=0

    while [ $attempt -lt $MAX_HEALTH_RETRIES ]; do
        if curl -sf "$HEALTH_CHECK_URL" > /dev/null 2>&1; then
            log_info "健康检查通过！服务运行正常"
            return 0
        fi
        attempt=$((attempt + 1))
        printf "."
        sleep "$HEALTH_CHECK_INTERVAL"
    done

    echo ""
    error_exit "健康检查失败！服务在 $((MAX_HEALTH_RETRIES * HEALTH_CHECK_INTERVAL))秒 内未就绪"
}

# ============================
# 回滚功能
# ============================
do_rollback() {
    log_step "执行回滚"

    local latest_backup
    latest_backup=$(ls -t "${BACKUP_DIR}"/${APP_NAME}_*.tar.gz 2>/dev/null | head -1)

    if [ -z "$latest_backup" ]; then
        error_exit "没有找到可用的备份"
    fi

    log_info "使用备份: $latest_backup"
    log_warn "这将覆盖当前部署目录！"

    sudo systemctl stop "$APP_NAME" 2>/dev/null || true

    tar -xzf "$latest_backup" -C "$(dirname "$DEPLOY_DIR")"

    sudo systemctl start "$APP_NAME" 2>/dev/null || true

    log_info "回滚完成"
}

# ============================
# 通知函数
# ============================
send_notification() {
    local status="$1"
    local message="$2"

    if [ -n "${SLACK_WEBHOOK_URL:-}" ]; then
        curl -s -X POST "$SLACK_WEBHOOK_URL" \
            -H 'Content-Type: application/json' \
            -d "{\"text\": \"[${APP_NAME}] ${status}: ${message}\"}" \
            > /dev/null 2>&1 || true
    fi

    log_info "通知已发送: [$status] $message"
}

# ============================
# 主函数
# ============================
main() {
    parse_args "$@"

    echo "============================================"
    echo "  ${APP_NAME} 自动部署"
    echo "  分支: ${BRANCH}"
    echo "  环境: ${ENVIRONMENT}"
    echo "  时间: $(date '+%Y-%m-%d %H:%M:%S')"
    echo "============================================"

    if [ "$ROLLBACK" = true ]; then
        do_rollback
        exit 0
    fi

    local start_time
    start_time=$(date +%s)

    step_preflight
    step_pull_code
    step_build
    step_backup
    step_deploy
    step_health_check

    local end_time duration
    end_time=$(date +%s)
    duration=$((end_time - start_time))

    echo ""
    echo "============================================"
    log_info "部署成功！耗时: ${duration}秒"
    echo "============================================"

    send_notification "SUCCESS" "部署成功，耗时 ${duration}秒"
}

main "$@"
```
