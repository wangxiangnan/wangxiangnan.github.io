## 24.1 Bash 脚本：批量处理利器

### 批量转码脚本

```bash
#!/bin/bash
# batch_transcode.sh - 批量转码脚本
# 用法: ./batch_transcode.sh /path/to/input /path/to/output

INPUT_DIR="${1:-.}"
OUTPUT_DIR="${2:-./output}"
PRESET="medium"
CRF=23

# 创建输出目录
mkdir -p "$OUTPUT_DIR"

# 统计
total=0
success=0
failed=0

for file in "$INPUT_DIR"/*.{mp4,mkv,avi,mov,wmv,flv}; do
    [ -f "$file" ] || continue
    total=$((total + 1))

    filename=$(basename "$file")
    name="${filename%.*}"
    output="$OUTPUT_DIR/${name}_h264.mp4"

    echo "[$total] 处理: $filename"

    if ffmpeg -y -i "$file" \
        -c:v libx264 -preset "$PRESET" -crf "$CRF" \
        -c:a aac -b:a 128k \
        -movflags +faststart \
        "$output" 2>/dev/null; then
        success=$((success + 1))
        echo "  ✓ 完成: $output"
    else
        failed=$((failed + 1))
        echo "  ✗ 失败: $filename"
    fi
done

echo ""
echo "=== 处理完成 ==="
echo "总计: $total | 成功: $success | 失败: $failed"
```

## 24.2 Python 调用 FFmpeg

### 使用 subprocess（原生方式）

```python
#!/usr/bin/env python3
"""使用 subprocess 调用 FFmpeg"""

import subprocess
import json
import os


def get_video_info(filepath):
    """获取视频信息"""
    cmd = [
        'ffprobe', '-v', 'quiet',
        '-print_format', 'json',
        '-show_format', '-show_streams',
        filepath
    ]
    result = subprocess.run(cmd, capture_output=True, text=True)
    return json.loads(result.stdout)


def transcode(input_path, output_path, video_bitrate='2M', audio_bitrate='128k',
              resolution=None, preset='medium'):
    """转码视频"""
    cmd = [
        'ffmpeg', '-y',
        '-i', input_path,
        '-c:v', 'libx264',
        '-preset', preset,
        '-b:v', video_bitrate,
    ]

    if resolution:
        cmd.extend(['-s', resolution])

    cmd.extend([
        '-c:a', 'aac',
        '-b:a', audio_bitrate,
        '-movflags', '+faststart',
        output_path
    ])

    process = subprocess.run(cmd, capture_output=True, text=True)

    if process.returncode != 0:
        raise RuntimeError(f"FFmpeg 错误: {process.stderr}")

    return output_path


def get_progress(input_path, output_path):
    """带进度回调的转码"""
    info = get_video_info(input_path)
    total_duration = float(info['format']['duration'])

    cmd = [
        'ffmpeg', '-y',
        '-i', input_path,
        '-c:v', 'libx264', '-preset', 'fast',
        '-c:a', 'aac',
        '-progress', 'pipe:1',
        '-nostats',
        output_path
    ]

    process = subprocess.Popen(
        cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
        text=True
    )

    for line in process.stdout:
        if line.startswith('out_time_ms='):
            time_us = int(line.strip().split('=')[1])
            current_sec = time_us / 1_000_000
            progress = min(100, (current_sec / total_duration) * 100)
            print(f"\r进度: {progress:.1f}%", end='', flush=True)

    process.wait()
    print()  # 换行

    if process.returncode != 0:
        raise RuntimeError("转码失败")


# 使用示例
if __name__ == '__main__':
    # 获取视频信息
    info = get_video_info('input.mp4')
    print(f"时长: {info['format']['duration']}秒")

    # 转码
    transcode('input.mp4', 'output.mp4', video_bitrate='3M', resolution='1280x720')

    # 带进度的转码
    get_progress('input.mp4', 'output_progress.mp4')
```

### 使用 ffmpeg-python 库（推荐）

```python
#!/usr/bin/env python3
"""使用 ffmpeg-python 库（pip install ffmpeg-python）"""

import ffmpeg
import sys


class VideoProcessor:
    """视频处理管线——封装常用的 FFmpeg 操作"""

    def __init__(self, input_path):
        self.input_path = input_path
        self.info = ffmpeg.probe(input_path)
        self.video_stream = next(
            (s for s in self.info['streams'] if s['codec_type'] == 'video'), None
        )
        self.audio_stream = next(
            (s for s in self.info['streams'] if s['codec_type'] == 'audio'), None
        )

    @property
    def duration(self):
        return float(self.info['format']['duration'])

    @property
    def resolution(self):
        if self.video_stream:
            return (self.video_stream['width'], self.video_stream['height'])
        return None

    @property
    def fps(self):
        if self.video_stream:
            fps_str = self.video_stream.get('r_frame_rate', '30/1')
            num, den = fps_str.split('/')
            return float(num) / float(den) if float(den) != 0 else 30.0
        return None

    def transcode(self, output_path, video_bitrate='2M', audio_bitrate='128k',
                  resolution=None, codec='libx264', preset='medium', crf=None):
        """通用转码方法"""
        stream = ffmpeg.input(self.input_path)

        video_kwargs = {'vcodec': codec, 'preset': preset}
        if crf:
            video_kwargs['crf'] = crf
        else:
            video_kwargs['video_bitrate'] = video_bitrate
        if resolution:
            video_kwargs['s'] = resolution

        video = stream.video.filter('scale', -2, 'ih' if not resolution else -2)

        (
            ffmpeg
            .output(video, stream.audio, output_path,
                    acodec='aac', audio_bitrate=audio_bitrate, **video_kwargs)
            .overwrite_output()
            .run(quiet=True)
        )
        return output_path

    def generate_hls(self, output_dir, segment_duration=10):
        """生成 HLS 切片"""
        import os
        os.makedirs(output_dir, exist_ok=True)

        (
            ffmpeg
            .input(self.input_path)
            .output(
                f'{output_dir}/stream.m3u8',
                format='hls',
                hls_time=segment_duration,
                hls_list_size=0,
                hls_segment_filename=f'{output_dir}/seg%03d.ts',
                vcodec='libx264',
                acodec='aac'
            )
            .overwrite_output()
            .run(quiet=True)
        )

    def generate_thumbnails(self, output_dir, count=10, width=320):
        """生成缩略图序列"""
        import os
        os.makedirs(output_dir, exist_ok=True)

        (
            ffmpeg
            .input(self.input_path)
            .filter('scale', width, -2)
            .filter('fps', f'{count / self.duration}')
            .output(f'{output_dir}/thumb_%04d.jpg', vsync='vfr', **{'q:v': 2})
            .overwrite_output()
            .run(quiet=True)
        )

    def generate_preview_gif(self, output_path, start=5, duration=5,
                             width=480, fps=15):
        """生成预览 GIF"""
        (
            ffmpeg
            .input(self.input_path, ss=start, t=duration)
            .filter('scale', width, -2)
            .filter('fps', fps)
            .output(output_path)
            .overwrite_output()
            .run(quiet=True)
        )

    def add_watermark(self, output_path, watermark_path, position='bottomright'):
        """添加图片水印"""
        positions = {
            'bottomright': 'W-w-10:H-h-10',
            'bottomleft': '10:H-h-10',
            'topright': 'W-w-10:10',
            'topleft': '10:10',
            'center': '(W-w)/2:(H-h)/2',
        }
        pos = positions.get(position, positions['bottomright'])

        input_video = ffmpeg.input(self.input_path)
        watermark = ffmpeg.input(watermark_path)

        (
            ffmpeg
            .filter([input_video.video, watermark.video], 'overlay',
                    x=pos.split(':')[0], y=pos.split(':')[1])
            .output(output_path, vcodec='libx264', acodec='copy')
            .overwrite_output()
            .run(quiet=True)
        )

    def extract_audio(self, output_path, format='aac'):
        """提取音频"""
        codecs = {'aac': 'aac', 'mp3': 'libmp3lame', 'wav': 'pcm_s16le', 'flac': 'flac'}
        codec = codecs.get(format, 'aac')

        (
            ffmpeg
            .input(self.input_path)
            .output(output_path, vn=None, acodec=codec)
            .overwrite_output()
            .run(quiet=True)
        )

    def trim(self, output_path, start, end):
        """精确裁剪"""
        (
            ffmpeg
            .input(self.input_path, ss=start, to=end)
            .output(output_path, vcodec='libx264', acodec='aac')
            .overwrite_output()
            .run(quiet=True)
        )


# 使用示例
if __name__ == '__main__':
    # 创建处理器
    vp = VideoProcessor('input.mp4')

    # 查看信息
    print(f"分辨率: {vp.resolution}")
    print(f"时长: {vp.duration:.1f}秒")
    print(f"帧率: {vp.fps:.1f}fps")

    # 转码
    vp.transcode('output_720p.mp4', resolution='1280x720', crf=23)

    # 生成 HLS
    vp.generate_hls('/tmp/hls_output', segment_duration=6)

    # 生成预览 GIF
    vp.generate_preview_gif('preview.gif', start=10, duration=5)

    # 添加水印
    vp.add_watermark('watermarked.mp4', 'logo.png', position='bottomright')
```

// npm install fluent-ffmpeg
const ffmpeg = require('fluent-ffmpeg');
const path = require('path');

// 简单转码
function transcode(inputPath, outputPath) {
    return new Promise((resolve, reject) => {
        ffmpeg(inputPath)
            .videoCodec('libx264')
            .audioCodec('aac')
            .videoBitrate('2000k')
            .audioBitrate('128k')
            .outputOptions(['-preset medium', '-movflags +faststart'])
            .on('progress', (progress) => {
                console.log(`进度: ${progress.percent?.toFixed(1)}%`);
            })
            .on('end', () => {
                console.log('转码完成');
                resolve(outputPath);
            })
            .on('error', (err) => {
                console.error('转码失败:', err.message);
                reject(err);
            })
            .save(outputPath);
    });
}

// 获取视频信息
function getVideoInfo(inputPath) {
    return new Promise((resolve, reject) => {
        ffmpeg.ffprobe(inputPath, (err, metadata) => {
            if (err) return reject(err);
            const video = metadata.streams.find(s => s.codec_type === 'video');
            const audio = metadata.streams.find(s => s.codec_type === 'audio');
            resolve({
                duration: metadata.format.duration,
                size: metadata.format.size,
                video: {
                    codec: video?.codec_name,
                    width: video?.width,
                    height: video?.height,
                    fps: video?.r_frame_rate,
                },
                audio: {
                    codec: audio?.codec_name,
                    sampleRate: audio?.sample_rate,
                    channels: audio?.channels,
                }
            });
        });
    });
}

// 生成 HLS
function generateHLS(inputPath, outputDir) {
    return new Promise((resolve, reject) => {
        ffmpeg(inputPath)
            .videoCodec('libx264')
            .audioCodec('aac')
            .outputOptions([
                '-hls_time 10',
                '-hls_list_size 0',
                `-hls_segment_filename ${outputDir}/seg%03d.ts`,
            ])
            .on('end', resolve)
            .on('error', reject)
            .save(`${outputDir}/stream.m3u8`);
    });
}

// 生成缩略图
function generateThumbnails(inputPath, outputDir, count = 10) {
    return new Promise((resolve, reject) => {
        ffmpeg(inputPath)
            .screenshots({
                count: count,
                folder: outputDir,
                filename: 'thumb_%i.jpg',
                size: '320x180',
            })
            .on('end', resolve)
            .on('error', reject);
    });
}

// 使用示例
async function main() {
    // 获取信息
    const info = await getVideoInfo('input.mp4');
    console.log('视频信息:', JSON.stringify(info, null, 2));

    // 转码
    await transcode('input.mp4', 'output.mp4');

    // 生成 HLS
    await generateHLS('input.mp4', './hls_output');

    // 生成缩略图
    await generateThumbnails('input.mp4', './thumbnails', 10);
}

main().catch(console.error);

## 24.6 进度监控与日志

FFmpeg 提供了多种方式来监控处理进度：

```bash
# -progress：输出机器可读的进度信息
ffmpeg -progress progress.log -i input.mp4 \
  -c:v libx264 -crf 23 output.mp4

# progress.log 内容示例：
# frame=150
# fps=30.0
# stream_0_0_q=28.0
# bitrate=2500.0kbits/s
# total_size=15000000
# out_time_us=5000000
# out_time_ms=5000000
# out_time=00:00:05.000000
# dup_frames=0
# drop_frames=0
# speed=1.00x
# progress=continue

# -stats：控制统计信息输出
ffmpeg -stats -i input.mp4 -c:v libx264 output.mp4
# 输出: frame=  150 fps= 30 q=28.0 size=   15000kB time=00:00:05.00 bitrate=2500.0kbits/s speed=1.00x

# -stats_period：控制统计信息输出频率
ffmpeg -stats_period 0.5 -i input.mp4 -c:v libx264 output.mp4

# -nostats：不输出统计信息
ffmpeg -nostats -i input.mp4 -c:v libx264 output.mp4

# -loglevel：控制日志级别
ffmpeg -loglevel warning -i input.mp4 -c:v libx264 output.mp4
# 可选: quiet, panic, fatal, error, warning, info, verbose, debug, trace
```

**实时进度监控脚本：**

```bash
#!/bin/bash
# monitor_progress.sh - 实时监控 FFmpeg 进度

INPUT="$1"
OUTPUT="$2"
DURATION=$(ffprobe -v error -show_entries format=duration \
    -of default=noprint_wrappers=1:nokey=1 "$INPUT")

ffmpeg -y -i "$INPUT" \
    -c:v libx264 -crf 23 \
    -progress pipe:1 -nostats \
    "$OUTPUT" 2>/dev/null | while IFS= read -r line; do
    if [[ "$line" == out_time_us=* ]]; then
        current_us=${line#out_time_us=}
        current_sec=$(echo "scale=2; $current_us / 1000000" | bc)
        progress=$(echo "scale=1; $current_sec * 100 / $DURATION" | bc)
        printf "\r进度: %5.1f%% (%s / %s秒)" "$progress" "$current_sec" "$DURATION"
    fi
done
echo ""
echo "处理完成！"
```

## 24.7 错误处理与日志记录

```bash
#!/bin/bash
# robust_ffmpeg.sh - 带完善错误处理的 FFmpeg 封装

LOG_FILE="/var/log/ffmpeg_jobs.log"

log() {
    local level="$1"
    shift
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [$level] $*" | tee -a "$LOG_FILE"
}

safe_ffmpeg() {
    local description="$1"
    shift

    log "INFO" "开始: $description"
    log "INFO" "命令: ffmpeg $*"

    local start_time=$(date +%s)
    local output
    output=$(ffmpeg -y "$@" 2>&1)
    local exit_code=$?
    local end_time=$(date +%s)
    local duration=$((end_time - start_time))

    if [ $exit_code -eq 0 ]; then
        log "INFO" "完成: $description (耗时 ${duration}秒)"
    else
        log "ERROR" "失败: $description (exit code: $exit_code)"
        log "ERROR" "输出: $output"
    fi

    return $exit_code
}

# 使用示例
safe_ffmpeg "转码 input.mp4 → output.mp4" \
    -i input.mp4 \
    -c:v libx264 -crf 23 \
    -c:a aac -b:a 128k \
    output.mp4

safe_ffmpeg "生成 HLS" \
    -i input.mp4 \
    -c:v libx264 -c:a aac \
    -f hls -hls_time 10 \
    output.m3u8
```

## 24.8 FFmpeg API 简介

FFmpeg 底层由一系列 C 库组成，如果你想在自己的 C/C++ 项目中使用：

| 库名 | 功能 |
| ------ | ------ |
| `libavcodec` | 编解码 |
| `libavformat` | 封装/解封装（容器格式） |
| `libavfilter` | 滤镜处理 |
| `libavutil` | 通用工具函数 |
| `libswscale` | 图像缩放和像素格式转换 |
| `libswresample` | 音频重采样和格式转换 |
| `libavdevice` | 输入输出设备（摄像头、屏幕录制等） |

一个最小化的 C 语言示例（获取视频信息）：

```c
// video_info.c
// 编译: gcc video_info.c -o video_info $(pkg-config --cflags --libs libavformat libavcodec libavutil)

#include <libavformat/avformat.h>
#include <stdio.h>

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "用法: %s <视频文件>\n", argv[0]);
        return 1;
    }

    AVFormatContext *fmt_ctx = NULL;

    if (avformat_open_input(&fmt_ctx, argv[1], NULL, NULL) < 0) {
        fprintf(stderr, "无法打开文件\n");
        return 1;
    }

    if (avformat_find_stream_info(fmt_ctx, NULL) < 0) {
        fprintf(stderr, "无法获取流信息\n");
        return 1;
    }

    printf("文件: %s\n", argv[1]);
    printf("时长: %.2f 秒\n",
           (double)fmt_ctx->duration / AV_TIME_BASE);
    printf("流数量: %d\n", fmt_ctx->nb_streams);

    for (unsigned i = 0; i < fmt_ctx->nb_streams; i++) {
        AVStream *stream = fmt_ctx->streams[i];
        AVCodecParameters *codecpar = stream->codecpar;

        printf("流 #%d: %s, %s",
               i,
               avcodec_get_name(codecpar->codec_id),
               av_get_media_type_string(codecpar->codec_type));

        if (codecpar->codec_type == AVMEDIA_TYPE_VIDEO) {
            printf(", %dx%d", codecpar->width, codecpar->height);
        } else if (codecpar->codec_type == AVMEDIA_TYPE_AUDIO) {
            printf(", %dHz, %d通道",
                   codecpar->sample_rate, codecpar->ch_layout.nb_channels);
        }
        printf("\n");
    }

    avformat_close_input(&fmt_ctx);
    return 0;
}
```

> **提示**：除非你在开发多媒体应用，否则日常使用命令行工具就够了。API 主要面向需要深度集成的场景，比如开发视频编辑器、流媒体服务器等。

### 常见错误与解决方案速查

| 错误信息 | 原因 | 解决方案 |
| --------- | ------ | --------- |
| `Protocol not found` | FFmpeg 编译时未包含该协议 | 重新编译 FFmpeg，加入对应库 |
| `Invalid data found when processing input` | 输入文件损坏或格式不支持 | 检查文件完整性，确认格式支持 |
| `Output file does not contain any stream` | 所有流都被过滤掉了 | 检查 `-map` 参数是否正确 |
| `Encoder not found` | 缺少编码器 | 安装对应的编码库（如 x264） |
| `Server returned 403 Forbidden` | 服务器拒绝访问 | 检查推流地址和密钥是否正确 |
| `Connection refused` | 服务器未运行或端口错误 | 确认服务器已启动，检查端口号 |
| `moov atom not found` | MP4 文件未正确关闭 | 使用 `ffmpeg -i broken.mp4 -c copy fixed.mp4` 修复 |
| `Non-monotonous DTS` | 时间戳不连续 | 添加 `-copyts -start_at_zero` |
| `Too many packets buffered` | 缓冲区溢出 | 添加 `-max_muxing_queue_size 9999` |
| `Conversion failed!` | 通用错误 | 检查 `-loglevel verbose` 查看详细错误 |
| `hwaccel initialisation returned error` | 硬件加速初始化失败 | 检查 GPU 驱动和 FFmpeg 编译选项 |
| 推流花屏 | 非关键帧开始或 GOP 过大 | 减小 `-g` 值，确保从关键帧开始 |
| 音画不同步 | 时间戳问题 | 使用 `-async 1` 或重新编码 |

**关于常见困惑的解答：**

> "FFmpeg 命令太多了，我记不住怎么办？"

没有人能记住所有命令。即使是 FFmpeg 的核心开发者也天天查文档。关键是你知道**能做什么**和**怎么找到方法**。速查表就是为这个目的存在的——打印出来贴在显示器旁边。

> "libx264 还是 NVENC？到底该用哪个？"

这取决于你的场景。直播推流用 NVENC（速度优先），离线转码用 libx264（质量优先）。如果不确定，先用 NVENC，等质量不满意再换。

> "HLS 还是 DASH？"

如果你的用户主要在 iOS 生态，用 HLS。如果需要跨平台且追求标准化，DASH。最聪明的做法是用 fMP4 + CMAF，一套切片同时服务 HLS 和 DASH。

> "我需要学 FFmpeg 的 C API 吗？"

除非你在开发多媒体应用（视频编辑器、流媒体服务器、播放器），否则不需要。命令行工具 + 脚本集成能覆盖 95% 的需求。

- **AI 编码**：机器学习正在重新定义视频编码。Netflix 的 Dynamic Optimizer 已经能根据内容复杂度自动调整编码参数
- **AV1 普及**：开放免费的编码标准，YouTube 和 Netflix 已经在大规模使用
- **实时 AI 处理**：超分辨率、背景虚化、实时翻译，这些曾经需要离线处理的效果现在可以实时完成
- **空间计算**：Apple Vision Pro 带来的空间视频，将催生新的多媒体处理需求
