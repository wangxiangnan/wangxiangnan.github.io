# 第23章：会话管理与限流——Web开发必备

## 引言：Web 开发的两个守门人

**"你是谁？"**——会话管理。HTTP 无状态，Session 就是用户的会员卡。

**"你太热情了！"**——限流。就像礼貌的保安："先生，您今天的访问次数已用完。"

Redis 在两个领域都是最佳选择：内存级速度保证 Session 读写，原子操作保证限流准确。

## 23.1 会话管理（Session Management）

### 为什么用 Redis 存储 Session？

```text
传统 Session：[App-1: {A,B}]  [App-2: {C,D}]  -> 不共享
Redis Session：[App-1,2,3] -> [Redis: {A,B,C,D}]  -> 共享
```

```js
  resave: false,
  saveUninitialized: false,
  name: 'app.sid',
  cookie: {
    maxAge: 86400 * 1000,
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',

  },
  rolling: true,

app.post('/api/login', express.json(), (req, res) => {
  const user = { id: 1, username: req.body.username, role: 'user' };

  req.session.regenerate((err) => {

  req.session.destroy(() => {
    res.clearCookie('app.sid');
    res.json({ message: '已登出' });
  });
});

app.listen(3000);
```

### 自定义 Session 管理器

```js
class SessionManager {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.prefix = options.prefix || 'session';
    this.ttl = options.ttl || 86400;
    this.maxDevices = options.maxDevices || 5;
  }

  async create(userId, deviceInfo = {}) {
    const sessionId = crypto.randomUUID();
    const sessionKey = `${this.prefix}:${sessionId}`;

    await this.redis.set(sessionKey, JSON.stringify({
      id: sessionId, userId, device: deviceInfo,
      createdAt: Date.now(), lastAccess: Date.now(),
    }), 'EX', this.ttl);

    await this.redis.sadd(`${this.prefix}:user:${userId}`, sessionId);

    const count = await this.redis.scard(`${this.prefix}:user:${userId}`);
    if (count > this.maxDevices) await this._evictOldest(userId);

    return sessionId;
  }

  async get(sessionId) {
    const data = await this.redis.get(`${this.prefix}:${sessionId}`);
    if (!data) return null;
    const session = JSON.parse(data);
    session.lastAccess = Date.now();
    await this.redis.set(`${this.prefix}:${sessionId}`, JSON.stringify(session), 'EX', this.ttl);
    return session;
  }

  async destroy(sessionId) {
    const session = await this.get(sessionId);
    if (!session) return;
    await this.redis.del(`${this.prefix}:${sessionId}`);
    await this.redis.srem(`${this.prefix}:user:${session.userId}`, sessionId);
  }

  async forceLogout(userId) {
    const key = `${this.prefix}:user:${userId}`;
    const sids = await this.redis.smembers(key);
    const pipeline = this.redis.pipeline();
    sids.forEach((sid) => pipeline.del(`${this.prefix}:${sid}`));
    pipeline.del(key);
    await pipeline.exec();
    return sids.length;
  }

  async getUserSessions(userId) {
    const sids = await this.redis.smembers(`${this.prefix}:user:${userId}`);
    const sessions = [];
    for (const sid of sids) {
      const s = await this.get(sid);
      if (s) sessions.push(s);
      else await this.redis.srem(`${this.prefix}:user:${userId}`, sid);
    }
    return sessions;
  }

  async _evictOldest(userId) {
    const sessions = await this.getUserSessions(userId);
    sessions.sort((a, b) => a.createdAt - b.createdAt);
    if (sessions.length > 0) await this.destroy(sessions[0].id);
  }
}

const sessionMgr = new SessionManager(redis, { ttl: 3600, maxDevices: 3 });
const sid = await sessionMgr.create(42, { device: 'iPhone 15', ip: '192.168.1.100' });
console.log('Session:', await sessionMgr.get(sid));
```

> **Warning：Session 安全清单**
>
> - `httpOnly: true` 防 XSS
> - 生产环境 `secure: true`（HTTPS）
> - 登录后 `regenerate` Session ID
> - `sameSite: 'lax'` 防 CSRF
> - 敏感操作要求二次验证

## 23.2 限流（Rate Limiting）

限流就像高速公路的限速摄像头——不是为了阻止你到达目的地，而是保证公路不因为超速车而崩溃。

### 限流算法

```text
1. 固定窗口：|--窗口1--|--窗口2--|  边界可能突发2x
2. 滑动窗口：<--滑动-->           更平滑
3. 令牌桶：固定速率加令牌，允许突发
4. 漏桶：强制均匀速率，最严格
```

### 固定窗口

```js
class FixedWindowLimiter {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.maxRequests = options.maxRequests || 100;
    this.windowSeconds = options.windowSeconds || 60;
  }

  async check(key) {
    const windowKey = `rl:${key}:${Math.floor(Date.now() / 1000 / this.windowSeconds)}`;
    const current = await this.redis.incr(windowKey);
    if (current === 1) await this.redis.expire(windowKey, this.windowSeconds + 1);

    return {
      allowed: current <= this.maxRequests,
      remaining: Math.max(0, this.maxRequests - current),
      resetAt: (Math.floor(Date.now() / 1000 / this.windowSeconds) * this.windowSeconds + this.windowSeconds) * 1000,
    };
  }
}

const limiter = new FixedWindowLimiter(redis, { maxRequests: 10, windowSeconds: 60 });
for (let i = 0; i < 15; i++) {
  const r = await limiter.check('user:42');
  console.log(`请求 ${i+1}: ${r.allowed ? '通过' : '拒绝'} (剩余 ${r.remaining})`);
}
```

### 滑动窗口（Lua）

```js
class SlidingWindowLimiter {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.maxRequests = options.maxRequests || 100;
    this.windowMs = options.windowMs || 60000;
  }

  async check(key) {
    const rlimitKey = `slimit:${key}`;
    const now = Date.now();
    const windowStart = now - this.windowMs;

    const luaScript = `
      redis.call("zremrangebyscore", KEYS[1], 0, ARGV[2])
      local current = redis.call("zcard", KEYS[1])
      if current < tonumber(ARGV[3]) then
        redis.call("zadd", KEYS[1], ARGV[1], ARGV[1]..":"..math.random(1e6))
        redis.call("pexpire", KEYS[1], ARGV[4])
        return { 1, tonumber(ARGV[3]) - current - 1 }
      else
        return { 0, 0 }
      end
    `;

    const [allowed, remaining] = await this.redis.eval(
      luaScript, 1, rlimitKey, now, windowStart, this.maxRequests, this.windowMs
    );
    return { allowed: allowed === 1, remaining, resetAt: now + this.windowMs };
  }
}
```

### 令牌桶（Lua）

```js
class TokenBucketLimiter {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.capacity = options.capacity || 100;
    this.refillRate = options.refillRate || 10;
  }

  async check(key, tokens = 1) {
    const bucketKey = `tb:${key}`;
    const luaScript = `
      local bucket = redis.call("hmget", KEYS[1], "tokens", "lastRefill")
      local currentTokens = tonumber(bucket[1]) or tonumber(ARGV[1])
      local lastRefill = tonumber(bucket[2]) or tonumber(ARGV[3])
      local elapsed = (tonumber(ARGV[3]) - lastRefill) / 1000
      local newTokens = math.min(tonumber(ARGV[1]), currentTokens + elapsed * tonumber(ARGV[2]))
      local allowed = 0
      if newTokens >= tonumber(ARGV[4]) then
        newTokens = newTokens - tonumber(ARGV[4])
        allowed = 1
      end
      redis.call("hmset", KEYS[1], "tokens", newTokens, "lastRefill", tonumber(ARGV[3]))
      redis.call("expire", KEYS[1], math.ceil(tonumber(ARGV[1]) / tonumber(ARGV[2])) + 1)
      return { allowed, math.floor(newTokens) }
    `;

    const [allowed, remaining] = await this.redis.eval(
      luaScript, 1, bucketKey, this.capacity, this.refillRate, Date.now(), tokens
    );
    return { allowed: allowed === 1, remaining };
  }
}

const tb = new TokenBucketLimiter(redis, { capacity: 10, refillRate: 2 });
for (let i = 0; i < 12; i++) {
  const r = await tb.check('api:heavy');
  console.log(`请求 ${i+1}: ${r.allowed ? '通过' : '拒绝'} (剩余 ${r.remaining})`);
}
```

### Express 限流中间件

```js
function createRateLimiter(options = {}) {
  const { maxRequests = 100, windowSeconds = 60, keyGenerator = (req) => req.ip, message = '请求过于频繁' } = options;

  return async (req, res, next) => {
    const key = keyGenerator(req);
    const windowKey = `rl:${key}:${Math.floor(Date.now() / 1000 / windowSeconds)}`;

    try {
      const current = await redis.incr(windowKey);
      if (current === 1) await redis.expire(windowKey, windowSeconds + 1);

      const remaining = Math.max(0, maxRequests - current);
      const resetAt = (Math.floor(Date.now() / 1000 / windowSeconds) + 1) * windowSeconds;

      res.set({
        'X-RateLimit-Limit': maxRequests,
        'X-RateLimit-Remaining': remaining,
        'X-RateLimit-Reset': Math.ceil(resetAt),
      });

      if (current > maxRequests) {
        res.set('Retry-After', Math.ceil(resetAt - Date.now() / 1000));
        return res.status(429).json({ error: message });
      }
      next();
    } catch (err) {
      next(); // fail open
    }
  };
}
```

### 多维度限流

```js
class MultiDimensionRateLimiter {
  constructor(redisClient) {
    this.redis = redisClient;
    this.rules = [
      { dimension: 'ip', maxRequests: 200, windowSeconds: 60 },
      { dimension: 'user', maxRequests: 100, windowSeconds: 60 },
      { dimension: 'api', maxRequests: 30, windowSeconds: 60 },
    ];
  }

  async check(req) {
    for (const rule of this.rules) {
      let key;
      if (rule.dimension === 'ip') key = `rl:ip:${req.ip}`;
      else if (rule.dimension === 'user') key = req.session?.userId ? `rl:user:${req.session.userId}` : null;
      else key = `rl:api:${req.method}:${req.path}`;

      if (!key) continue;

      const windowKey = `${key}:${Math.floor(Date.now() / 1000 / rule.windowSeconds)}`;
      const current = await this.redis.incr(windowKey);
      if (current === 1) await this.redis.expire(windowKey, rule.windowSeconds + 1);

      if (current > rule.maxRequests) {
        return { allowed: false, dimension: rule.dimension, limit: rule.maxRequests, current };
      }
    }
    return { allowed: true };
  }
}
```

## 23.3 实战：API 网关限流模块

```js
class APIGatewayRateLimiter {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.defaultLimits = options.defaultLimits || { maxRequests: 100, windowSeconds: 60 };
    this.customLimits = new Map();
    this.whitelist = new Set();
  }

  addRule(pathPattern, limits) {
    this.customLimits.set(pathPattern, limits);
  }

  addToWhitelist(ip) { this.whitelist.add(ip); }

  async addToBlacklist(ip, durationSeconds = 3600) {
    await this.redis.set(`blacklist:${ip}`, '1', 'EX', durationSeconds);
  }

  async check(req) {
    const ip = req.ip || 'unknown';

    // 黑名单
    if (await this.redis.exists(`blacklist:${ip}`)) {
      return { allowed: false, status: 403, message: 'IP 已被封禁' };
    }

    // 白名单
    if (this.whitelist.has(ip)) return { allowed: true };

    // 获取规则
    const rule = this._getRule(req.path) || this.defaultLimits;
    const windowKey = `gw:${ip}:${req.path}:${Math.floor(Date.now() / 1000 / rule.windowSeconds)}`;
    const current = await this.redis.incr(windowKey);
    if (current === 1) await this.redis.expire(windowKey, rule.windowSeconds + 1);

    // 频繁超限自动封禁
    if (current > rule.maxRequests * 2) {
      await this.addToBlacklist(ip, 600);
      return { allowed: false, status: 403, message: '频繁超限，IP 已临时封禁' };
    }

    if (current > rule.maxRequests) {
      return { allowed: false, status: 429, message: '请求频率超限', remaining: 0 };
    }

    return { allowed: true, remaining: rule.maxRequests - current };
  }

  _getRule(path) {
    for (const [pattern, limits] of this.customLimits) {
      if (path.startsWith(pattern)) return limits;
    }
    return null;
  }
}

const gateway = new APIGatewayRateLimiter(redis);
gateway.addRule('/api/search', { maxRequests: 20, windowSeconds: 60 });
gateway.addRule('/api/upload', { maxRequests: 5, windowSeconds: 60 });
gateway.addRule('/api/login', { maxRequests: 10, windowSeconds: 60 });
gateway.addToWhitelist('10.0.0.1');
```

## 小结

**会话管理**：Redis Session 解决分布式共享问题。安全要点：httpOnly、secure、sameSite、Session 固定防护。

**限流**：固定窗口简单但有边界问题；滑动窗口更平滑；令牌桶允许突发；多维度组合限流。

**记住**：限流不是为了拒绝用户，而是为了保护系统。好的限流策略像弹簧——有弹性、可调节、能快速恢复。
