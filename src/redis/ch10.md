# 第10章：Lua 脚本——在 Redis 里写代码

## 引言：给 Redis 装上一个大脑

到目前为止，我们和 Redis 的交互都是"一问一答"式的。**Lua 脚本**让你能把一系列指令打包成一个"剧本"，让 Redis 自己演完。一气呵成，中间不会被打断。

更妙的是，Lua 脚本的执行是**原子性的**——脚本执行期间不会有其他客户端的命令插进来。

## 10.1 为什么需要 Lua 脚本？

**1. 原子性** —— "检查库存 → 扣减库存 → 创建订单"必须一气呵成。

**2. 减少网络开销** —— 5 次 Redis 调用变成 1 次网络往返。

```
不用 Lua（5次往返）：
Client --GET--> Redis --OK--> Client
Client --SET--> Redis --OK--> Client
Client --INCR--> Redis --OK--> Client
Client --SET--> Redis --OK--> Client
Client --EXPIRE--> Redis --OK--> Client

用 Lua（1次往返）：
Client --EVAL(script)--> Redis --result--> Client
```

## 10.2 Lua 速成（5分钟）

```lua
-- 变量
local name = "Redis"
local count = 42

-- 条件
if count > 10 then
    return "大于10"
end

-- 循环
for i = 1, 10 do
    -- do something
end

-- 字符串拼接用 ".."
local msg = "count = " .. tostring(count)

-- 表（Table）
local arr = {1, 2, 3}
local dict = {name = "Alice", age = 28}
```

## 10.3 Redis 中的 Lua：核心 API

```lua
-- redis.call()：失败时抛出错误（中断脚本）
local val = redis.call('GET', 'mykey')

-- redis.pcall()：失败时返回错误对象（脚本继续）
local result = redis.pcall('GET', 'mykey')
```

参数通过 `KEYS` 和 `ARGV` 传入：

```lua
local key = KEYS[1]
local value = ARGV[1]
return redis.call('SET', key, value)
```

## 10.4 EVAL / EVALSHA / SCRIPT LOAD

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function luaBasics() {
  // EVAL：直接执行
  const r1 = await redis.eval(
    'return redis.call("SET", KEYS[1], ARGV[1])',
    1, 'greeting', 'Hello from Lua!'
  );
  console.log('EVAL SET:', r1);

  const r2 = await redis.eval('return redis.call("GET", KEYS[1])', 1, 'greeting');
  console.log('EVAL GET:', r2);

  // SCRIPT LOAD + EVALSHA
  const script = `
    local key = KEYS[1]
    local increment = tonumber(ARGV[1])
    local max = tonumber(ARGV[2])
    local current = tonumber(redis.call('GET', key) or '0')
    if current + increment > max then return -1 end
    return redis.call('INCRBY', key, increment)
  `;

  const sha = await redis.script('LOAD', script);
  console.log('\n脚本 SHA:', sha);

  const r3 = await redis.evalsha(sha, 1, 'counter', '5', '100');
  console.log('EVALSHA:', r3);

  const exists = await redis.script('EXISTS', sha, 'nonexistent_sha');
  console.log('脚本存在:', exists);

  await redis.quit();
}

luaBasics().catch(console.error);
```

> [!TIP] 提示
> 生产环境推荐 `SCRIPT LOAD` + `EVALSHA`，避免每次传输完整脚本。但 Redis 重启后脚本缓存会丢失，需要做 fallback。

## 10.5 常用 Lua 脚本示例

### 10.5.1 原子性 CAS（Compare-And-Swap）

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

const casScript = `
  local current = redis.call('GET', KEYS[1])
  if current == ARGV[1] then
    redis.call('SET', KEYS[1], ARGV[2])
    return 1
  else
    return 0
  end
`;

async function casDemo() {
  await redis.set('config:version', '1.0');

  const r1 = await redis.eval(casScript, 1, 'config:version', '1.0', '2.0');
  console.log('CAS 1.0->2.0:', r1 === 1 ? '成功' : '失败');

  const r2 = await redis.eval(casScript, 1, 'config:version', '1.0', '3.0');
  console.log('CAS 1.0->3.0:', r2 === 1 ? '成功' : '失败（版本不匹配）');

  console.log('最终值:', await redis.get('config:version'));
  await redis.quit();
}

casDemo().catch(console.error);
```

### 10.5.2 分布式锁

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

const acquireLockScript = `
  local lockKey = KEYS[1]
  local lockValue = ARGV[1]
  local ttl = tonumber(ARGV[2])
  if redis.call('SET', lockKey, lockValue, 'NX', 'EX', ttl) then return 1 end
  if redis.call('GET', lockKey) == lockValue then
    redis.call('EXPIRE', lockKey, ttl)
    return 1
  end
  return 0
`;

const releaseLockScript = `
  local lockKey = KEYS[1]
  local lockValue = ARGV[1]
  if redis.call('GET', lockKey) == lockValue then
    return redis.call('DEL', lockKey)
  end
  return 0
`;

class DistributedLock {
  constructor(redisClient) { this.redis = redisClient; }

  async acquire(key, value, ttlSeconds = 30) {
    const result = await this.redis.eval(
      acquireLockScript, 1, `lock:${key}`, value, ttlSeconds
    );
    return result === 1;
  }

  async release(key, value) {
    const result = await this.redis.eval(
      releaseLockScript, 1, `lock:${key}`, value
    );
    return result === 1;
  }
}

async function lockDemo() {
  const lock = new DistributedLock(redis);
  const myId = `worker_${Date.now()}`;

  const acquired = await lock.acquire('order_processing', myId, 30);
  console.log(`获取锁: ${acquired ? '成功' : '失败'}`);

  if (acquired) {
    console.log('执行临界区代码...');
    await new Promise((r) => setTimeout(r, 1000));
    const released = await lock.release('order_processing', myId);
    console.log(`释放锁: ${released ? '成功' : '失败'}`);
  }

  await redis.quit();
}

lockDemo().catch(console.error);
```

### 10.5.3 滑动窗口限流器

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

const slidingWindowScript = `
  local key = KEYS[1]
  local now = tonumber(ARGV[1])
  local window = tonumber(ARGV[2])
  local max_requests = tonumber(ARGV[3])
  redis.call('ZREMRANGEBYSCORE', key, '-inf', now - window)
  local count = redis.call('ZCARD', key)
  if count < max_requests then
    redis.call('ZADD', key, now, now .. '-' .. math.random(1000000))
    redis.call('PEXPIRE', key, window)
    return {1, max_requests - count - 1}
  else
    return {0, 0}
  end
`;

class RateLimiter {
  constructor(redisClient) { this.redis = redisClient; }

  async check(identifier, windowMs, maxRequests) {
    const result = await this.redis.eval(
      slidingWindowScript, 1, `ratelimit:${identifier}`,
      Date.now(), windowMs, maxRequests
    );
    return { allowed: result[0] === 1, remaining: result[1] };
  }
}

async function rateLimiterDemo() {
  const limiter = new RateLimiter(redis);
  console.log('滑动窗口限流器（5秒内最多3次）\n');

  for (let i = 1; i <= 6; i++) {
    const r = await limiter.check('user:42', 5000, 3);
    console.log(`  #${i}: ${r.allowed ? '允许' : '拒绝'} (剩余:${r.remaining})`);
    await new Promise((r) => setTimeout(r, 200));
  }

  console.log('\n等待窗口过期...');
  await new Promise((r) => setTimeout(r, 5500));

  const r = await limiter.check('user:42', 5000, 3);
  console.log(`  窗口过期后: ${r.allowed ? '允许' : '拒绝'} (剩余:${r.remaining})`);

  await redis.quit();
}

rateLimiterDemo().catch(console.error);
```

### 10.5.4 令牌桶限流器

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

const tokenBucketScript = `
  local key = KEYS[1]
  local capacity = tonumber(ARGV[1])
  local rate = tonumber(ARGV[2])
  local now = tonumber(ARGV[3])
  local requested = tonumber(ARGV[4])
  local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
  local tokens = tonumber(bucket[1]) or capacity
  local last_refill = tonumber(bucket[2]) or now
  local elapsed = now - last_refill
  local new_tokens = math.min(capacity, tokens + elapsed * rate)
  if new_tokens >= requested then
    new_tokens = new_tokens - requested
    redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
    redis.call('EXPIRE', key, math.ceil(capacity / rate) + 10)
    return {1, math.floor(new_tokens)}
  else
    redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
    redis.call('EXPIRE', key, math.ceil(capacity / rate) + 10)
    return {0, math.floor(new_tokens)}
  end
`;

class TokenBucket {
  constructor(redisClient) { this.redis = redisClient; }

  async consume(id, capacity, rate, tokens = 1) {
    const result = await this.redis.eval(
      tokenBucketScript, 1, `tokenbucket:${id}`,
      capacity, rate, Date.now() / 1000, tokens
    );
    return { allowed: result[0] === 1, remainingTokens: result[1] };
  }
}

async function tokenBucketDemo() {
  const bucket = new TokenBucket(redis);
  console.log('令牌桶（容量5，每秒补1个）\n');

  for (let i = 1; i <= 7; i++) {
    const r = await bucket.consume('api:user42', 5, 1);
    console.log(`  #${i}: ${r.allowed ? '允许' : '拒绝'} (剩余:${r.remainingTokens})`);
  }

  console.log('\n等待3秒...');
  await new Promise((r) => setTimeout(r, 3000));

  for (let i = 1; i <= 4; i++) {
    const r = await bucket.consume('api:user42', 5, 1);
    console.log(`  #${i}: ${r.allowed ? '允许' : '拒绝'} (剩余:${r.remainingTokens})`);
  }

  await redis.quit();
}

tokenBucketDemo().catch(console.error);
```

## 10.6 Node.js 中优雅地使用 Lua 脚本

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

redis.defineCommand('rateLimit', {
  numberOfKeys: 1,
  lua: `
    local key = KEYS[1]
    local limit = tonumber(ARGV[1])
    local window = tonumber(ARGV[2])
    local current = tonumber(redis.call('GET', key) or '0')
    if current >= limit then return {0, redis.call('TTL', key)} end
    local result = redis.call('INCR', key)
    if result == 1 then redis.call('EXPIRE', key, window) end
    return {1, limit - result}
  `,
});

redis.defineCommand('cas', {
  numberOfKeys: 1,
  lua: `
    if redis.call('GET', KEYS[1]) == ARGV[1] then
      redis.call('SET', KEYS[1], ARGV[2])
      return 1
    end
    return 0
  `,
});

async function customCommandsDemo() {
  const rateResult = await redis.rateLimit('api:user42', 5, 60);
  console.log('限流结果:', rateResult);

  await redis.set('mykey', 'v1');
  const casResult = await redis.cas('mykey', 'v1', 'v2');
  console.log('CAS 结果:', casResult);
  console.log('新值:', await redis.get('mykey'));

  await redis.quit();
}

customCommandsDemo().catch(console.error);
```

## 10.7 Lua 脚本的限制和注意事项

```
Lua 脚本的限制：

1. 原子性 = 阻塞
   脚本执行期间 Redis 不处理其他命令。保持脚本简短高效！

2. 确定性要求
   math.random() 用固定种子初始化，TIME 返回脚本开始时间。

3. 超时限制
   默认 lua-time-limit = 5000ms。

4. Cluster 模式下键名必须在 KEYS 参数中声明。

5. 不能调用外部系统（网络请求、文件读写）。
```

> [!WARNING] 注意
> 最常见的坑是在 Lua 脚本里用 `KEYS` 命令扫描键。在 Cluster 模式下这会直接报错。

## 小结

Lua 脚本就像 Redis 的**存储过程**——把业务逻辑推到数据所在的位置执行。这种"计算向数据移动"的思想，是高性能系统的核心哲学。
