# 第13章：RedisGraph 与 RedisTimeSeries——特殊场景利器

## 引言：瑞士军刀的最后几把刀片

- **RedisGraph**：让 Redis 变成图数据库（像 Neo4j）
- **RedisTimeSeries**：让 Redis 变成时序数据库（像 InfluxDB）

## 13.1 RedisGraph：图数据库模块

> **注意：** RedisGraph 已于 2024 年进入 EOL 阶段。生产环境建议使用 Neo4j 或 Apache AGE。但学习图数据库概念仍然很有价值。

### 什么是图数据库？

传统数据库处理"朋友的朋友的朋友"需要多次 JOIN。图数据库专门为**关系**而生：

```
      [Alice]--认识-->[Bob]
         |                |
       认识              认识
         |                |
      [Charlie]--认识-->[Diana]
         |
       认识
         |
      [Eve]
```

### 13.1.1 Cypher 查询语言

```bash
docker run -d --name redis-stack -p 6379:6379 redis/redis-stack:latest
```

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

function printResults(result) {
  if (!result || result.length === 0) { console.log('  (无结果)'); return; }
  const rows = result.slice(1, -1);
  rows.forEach((row) => {
    console.log(`  ${Array.isArray(row) ? row.join(' | ') : row}`);
  });
}

async function graphBasics() {
  const g = 'social_network';

  await redis.call('GRAPH.QUERY', g, `CREATE
    (alice:Person {name: 'Alice', age: 28, city: 'Beijing'}),
    (bob:Person {name: 'Bob', age: 32, city: 'Shanghai'}),
    (charlie:Person {name: 'Charlie', age: 25, city: 'Beijing'}),
    (diana:Person {name: 'Diana', age: 30, city: 'Shenzhen'}),
    (eve:Person {name: 'Eve', age: 27, city: 'Beijing'}),
    (alice)-[:KNOWS {since: 2020}]->(bob),
    (alice)-[:KNOWS {since: 2021}]->(charlie),
    (bob)-[:KNOWS {since: 2019}]->(diana),
    (charlie)-[:KNOWS {since: 2022}]->(diana),
    (charlie)-[:KNOWS {since: 2023}]->(eve),
    (diana)-[:WORKS_WITH]->(eve)
    RETURN alice, bob, charlie, diana, eve`);
  console.log('社交网络已创建\n');

  console.log('所有人员:');
  printResults(await redis.call('GRAPH.QUERY', g,
    'MATCH (p:Person) RETURN p.name, p.age, p.city ORDER BY p.age'));

  console.log('\nAlice 认识的人:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH (a:Person {name: 'Alice'})-[:KNOWS]->(f) RETURN f.name, f.city`));

  console.log('\n朋友的朋友:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH (a:Person {name: 'Alice'})-[:KNOWS]->(f)-[:KNOWS]->(fof)
     WHERE a <> fof RETURN DISTINCT fof.name, fof.city`));

  console.log('\n最短路径:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH path = shortestPath((a:Person {name: 'Alice'})-[*]->(e:Person {name: 'Eve'}))
     RETURN length(path)`));

  console.log('\n城市统计:');
  printResults(await redis.call('GRAPH.QUERY', g,
    'MATCH (p:Person) RETURN p.city, count(p) as cnt ORDER BY cnt DESC'));

  await redis.quit();
}

graphBasics().catch(console.error);
```

```js
async function recommendationGraph() {
  const g = 'movie_recs';

  await redis.call('GRAPH.QUERY', g, `CREATE
    (u1:User {name: 'Alice'}), (u2:User {name: 'Bob'}), (u3:User {name: 'Charlie'}),
    (m1:Movie {title: 'Inception', genre: 'Sci-Fi', year: 2010}),
    (m2:Movie {title: 'The Matrix', genre: 'Sci-Fi', year: 1999}),
    (m3:Movie {title: 'Interstellar', genre: 'Sci-Fi', year: 2014}),
    (m4:Movie {title: 'The Godfather', genre: 'Drama', year: 1972}),
    (m5:Movie {title: 'Pulp Fiction', genre: 'Crime', year: 1994}),
    (u1)-[:WATCHED {rating: 5}]->(m1), (u1)-[:WATCHED {rating: 4}]->(m2),
    (u1)-[:WATCHED {rating: 5}]->(m3), (u2)-[:WATCHED {rating: 4}]->(m1),
    (u2)-[:WATCHED {rating: 5}]->(m4), (u3)-[:WATCHED {rating: 3}]->(m2),
    (u3)-[:WATCHED {rating: 5}]->(m5),
    (m1)-[:SIMILAR_TO {score: 0.85}]->(m2),
    (m1)-[:SIMILAR_TO {score: 0.90}]->(m3),
    (m4)-[:SIMILAR_TO {score: 0.75}]->(m5)`);
  console.log('推荐系统图已创建\n');

  console.log('为 Alice 推荐:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH (a:User {name: 'Alice'})-[:WATCHED]->(w)-[:SIMILAR_TO]->(r)
     WHERE NOT (a)-[:WATCHED]->(r) RETURN r.title, r.genre ORDER BY r.year DESC`));

  console.log('\n品味相似用户:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH (a:User {name: 'Alice'})-[:WATCHED]->(m)<-[:WATCHED]-(o)
     WHERE a <> o RETURN o.name, count(m) as common ORDER BY common DESC`));

  console.log('\n所有科幻电影:');
  printResults(await redis.call('GRAPH.QUERY', g,
    `MATCH (m:Movie {genre: 'Sci-Fi'}) RETURN m.title, m.year ORDER BY m.year DESC`));

  await redis.quit();
}
```

## 13.2 RedisTimeSeries：时序数据模块

### 什么是时序数据？

按时间顺序排列的数据点。你每天都在产生海量时序数据：CPU 使用率、股票价格、IoT 传感器读数、网站访问量、应用性能指标。

共同特点：**写入量大、查询多为时间范围查询**。

### 13.2.1 核心操作

```bash
docker run -d --name redis-stack -p 6379:6379 redis/redis-stack:latest
```

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function timeSeriesBasics() {
  // TS.CREATE
  await redis.call('TS.CREATE', 'sensor:temp:room1',
    'RETENTION', 86400000, 'LABELS', 'sensor_id', 'temp_001', 'location', 'room1', 'type', 'temperature');
  await redis.call('TS.CREATE', 'sensor:temp:room2',
    'RETENTION', 86400000, 'LABELS', 'sensor_id', 'temp_002', 'location', 'room2', 'type', 'temperature');
  await redis.call('TS.CREATE', 'sensor:humidity:room1',
    'RETENTION', 86400000, 'LABELS', 'sensor_id', 'hum_001', 'location', 'room1', 'type', 'humidity');
  console.log('时序键已创建');

  // TS.ADD
  const now = Date.now();
  for (let i = 0; i < 60; i++) {
    const ts = now - (60 - i) * 60000;
    const temp1 = 22 + Math.sin(i * 0.1) * 3 + Math.random() * 2;
    const temp2 = 20 + Math.cos(i * 0.15) * 2 + Math.random() * 1.5;
    const hum = 45 + Math.sin(i * 0.08) * 10 + Math.random() * 3;
    await redis.call('TS.ADD', 'sensor:temp:room1', ts, temp1.toFixed(2));
    await redis.call('TS.ADD', 'sensor:temp:room2', ts, temp2.toFixed(2));
    await redis.call('TS.ADD', 'sensor:humidity:room1', ts, hum.toFixed(2));
  }
  console.log('已添加60个数据点\n');

  // TS.GET
  const latest = await redis.call('TS.GET', 'sensor:temp:room1');
  console.log('Room1 最新温度:', latest);

  // TS.RANGE（带聚合）
  const startTime = now - 30 * 60000;
  const range = await redis.call('TS.RANGE', 'sensor:temp:room1', startTime, '+',
    'AGGREGATION', 'avg', 300000);
  console.log('\nRoom1 最近30分钟（5分钟平均）:');
  range.forEach(([ts, val]) => {
    console.log(`  ${new Date(ts).toLocaleTimeString()}: ${val}`);
  });

  // TS.MRANGE（多键 + 标签过滤）
  const multi = await redis.call('TS.MRANGE', '-', '+',
    'FILTER', 'type=temperature',
    'GROUPBY', 'location', 'REDUCE', 'avg',
    'AGGREGATION', 'avg', 600000);
  console.log('\n所有温度（按位置分组，10分钟平均）:');
  multi.forEach(([key, labels, data]) => {
    console.log(`  位置: ${key}`);
    data.forEach(([ts, val]) => console.log(`    ${new Date(ts).toLocaleTimeString()}: ${val}`));
  });

  // TS.INFO
  const info = await redis.call('TS.INFO', 'sensor:temp:room1');
  console.log('\n传感器信息:');
  for (let i = 0; i < info.length; i += 2) {
    if (typeof info[i + 1] !== 'object') console.log(`  ${info[i]}: ${info[i + 1]}`);
  }

  await redis.quit();
}

timeSeriesBasics().catch(console.error);
```

### 13.2.2 降采样和聚合规则

降采样是时序数据库的核心——将高频数据自动汇总为低频数据。

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function downsamplingDemo() {
  // 原始数据（保留1天）
  await redis.call('TS.CREATE', 'cpu:server1:raw',
    'RETENTION', 86400000, 'LABELS', 'server', 'server1', 'metric', 'cpu', 'resolution', 'raw');

  // 1分钟平均（保留7天）
  await redis.call('TS.CREATE', 'cpu:server1:avg_1m',
    'RETENTION', 604800000, 'LABELS', 'server', 'server1', 'metric', 'cpu', 'resolution', '1min');

  // 1小时平均（保留30天）
  await redis.call('TS.CREATE', 'cpu:server1:avg_1h',
    'RETENTION', 2592000000, 'LABELS', 'server', 'server1', 'metric', 'cpu', 'resolution', '1hour');

  // 设置聚合规则
  await redis.call('TS.CREATERULE', 'cpu:server1:raw', 'cpu:server1:avg_1m', 'AGGREGATION', 'avg', 60000);
  await redis.call('TS.CREATERULE', 'cpu:server1:raw', 'cpu:server1:avg_1h', 'AGGREGATION', 'avg', 3600000);

  console.log('聚合规则:');
  console.log('  raw -> avg_1m (1分钟平均)');
  console.log('  raw -> avg_1h (1小时平均)');

  // 模拟数据
  const now = Date.now();
  for (let i = 0; i < 120; i++) {
    const ts = now - (120 - i) * 1000;
    const cpu = 30 + Math.sin(i * 0.2) * 20 + Math.random() * 10;
    await redis.call('TS.ADD', 'cpu:server1:raw', ts, cpu.toFixed(2));
  }
  console.log('\n已写入120秒的 CPU 数据');

  // 清理
  await redis.call('TS.DELETERULE', 'cpu:server1:raw', 'cpu:server1:avg_1m');
  await redis.call('TS.DELETERULE', 'cpu:server1:raw', 'cpu:server1:avg_1h');

  await redis.quit();
}

downsamplingDemo().catch(console.error);
```

### 13.2.3 标签过滤和 MGET

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function labelFiltering() {
  // TS.MGET：匹配标签的所有键的最新值
  const latest = await redis.call('TS.MGET', 'FILTER', 'type=temperature');
  console.log('所有温度传感器最新值:');
  latest.forEach(([key, labels, [ts, val]]) => {
    console.log(`  ${key}: ${val}`);
  });

  // TS.MGET + WITHLABELS
  const withLabels = await redis.call('TS.MGET', 'WITHLABELS', 'FILTER', 'location=room1');
  console.log('\nRoom1 所有传感器:');
  withLabels.forEach(([key, labels, [ts, val]]) => {
    const labelObj = {};
    for (let i = 0; i < labels.length; i += 2) labelObj[labels[i]] = labels[i + 1];
    console.log(`  ${key}: ${val} (标签: ${JSON.stringify(labelObj)})`);
  });

  // TS.QUERYINDEX：查找匹配的键名
  const keys = await redis.call('TS.QUERYINDEX', 'type=temperature');
  console.log('\n温度传感器键:', keys);

  const room1Keys = await redis.call('TS.QUERYINDEX', 'location=room1');
  console.log('Room1 所有键:', room1Keys);

  await redis.quit();
}

labelFiltering().catch(console.error);
```

### 13.2.4 实战：IoT 监控仪表盘

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

class IoTMonitor {
  constructor(redisClient) { this.redis = redisClient; }

  async setupSensor(deviceId, location, type) {
    const key = `iot:${type}:${deviceId}`;
    await this.redis.call('TS.CREATE', key,
      'RETENTION', 604800000,
      'LABELS', 'device', deviceId, 'location', location, 'type', type);
    return key;
  }

  async recordReading(key, value, timestamp) {
    await this.redis.call('TS.ADD', key, timestamp || '*', value);
  }

  async getLatestAll(type) {
    return await this.redis.call('TS.MGET', 'WITHLABELS', 'FILTER', `type=${type}`);
  }

  async getRange(key, fromTime, toTime, aggregation, bucketMs) {
    const args = [key, fromTime, toTime];
    if (aggregation && bucketMs) {
      args.push('AGGREGATION', aggregation, bucketMs);
    }
    return await this.redis.call('TS.RANGE', ...args);
  }

  async getStats(key, fromTime, toTime) {
    const [avg, min, max] = await Promise.all([
      this.redis.call('TS.RANGE', key, fromTime, toTime, 'AGGREGATION', 'avg', toTime - fromTime),
      this.redis.call('TS.RANGE', key, fromTime, toTime, 'AGGREGATION', 'min', toTime - fromTime),
      this.redis.call('TS.RANGE', key, fromTime, toTime, 'AGGREGATION', 'max', toTime - fromTime),
    ]);
    return {
      avg: avg.length > 0 ? parseFloat(avg[0][1]) : null,
      min: min.length > 0 ? parseFloat(min[0][1]) : null,
      max: max.length > 0 ? parseFloat(max[0][1]) : null,
    };
  }

  async detectAnomaly(key, threshold) {
    const latest = await this.redis.call('TS.GET', key);
    if (!latest || latest.length === 0) return null;
    const [ts, val] = latest;
    const value = parseFloat(val);
    return {
      timestamp: ts, value,
      isAnomaly: value > threshold,
      message: value > threshold
        ? `异常！当前值 ${value} 超过阈值 ${threshold}`
        : `正常: ${value}`,
    };
  }
}

async function main() {
  const monitor = new IoTMonitor(redis);

  // 设置传感器
  const tempSensor1 = await monitor.setupSensor('dev_001', 'warehouse_A', 'temperature');
  const tempSensor2 = await monitor.setupSensor('dev_002', 'warehouse_B', 'temperature');
  const humSensor1 = await monitor.setupSensor('dev_003', 'warehouse_A', 'humidity');
  console.log('传感器已配置\n');

  // 模拟数据
  const now = Date.now();
  for (let i = 0; i < 60; i++) {
    const ts = now - (60 - i) * 60000;
    await monitor.recordReading(tempSensor1, (22 + Math.sin(i * 0.1) * 5 + Math.random() * 2).toFixed(2), ts);
    await monitor.recordReading(tempSensor2, (18 + Math.cos(i * 0.15) * 3 + Math.random()).toFixed(2), ts);
    await monitor.recordReading(humSensor1, (45 + Math.sin(i * 0.08) * 15).toFixed(2), ts);
  }
  console.log('已记录60分钟数据\n');

  // 查看所有温度传感器最新值
  const latest = await monitor.getLatestAll('temperature');
  console.log('温度传感器最新值:');
  latest.forEach(([key, labels, [ts, val]]) => {
    console.log(`  ${key}: ${val}`);
  });

  // 获取统计信息
  const hourAgo = now - 3600000;
  const stats1 = await monitor.getStats(tempSensor1, hourAgo, now);
  console.log('\nWarehouse A 温度统计:');
  console.log(`  平均: ${stats1.avg.toFixed(2)}`);
  console.log(`  最低: ${stats1.min}`);
  console.log(`  最高: ${stats1.max}`);

  // 异常检测
  const anomaly = await monitor.detectAnomaly(tempSensor1, 30);
  console.log('\n异常检测:');
  console.log(`  ${anomaly.message}`);

  await redis.quit();
}

main().catch(console.error);
```

## 13.3 Redis Modules 生态概览

| 模块 | 类比 | 核心能力 | 适用场景 |
|------|------|----------|----------|
| **RediSearch** | Elasticsearch | 全文搜索、聚合 | 商品搜索、日志搜索 |
| **RedisJSON** | MongoDB | JSON 文档存储 | 配置管理、用户资料 |
| **RedisGraph** | Neo4j | 图数据库 | 社交网络、推荐系统 |
| **RedisTimeSeries** | InfluxDB | 时序数据 | IoT、监控、金融 |
| **RedisBloom** | - | 概率数据结构 | 去重、频率估计 |
| **RedisAI** | TensorFlow Serving | 模型推理 | 实时推荐、异常检测 |

## 小结

- **RedisGraph** 是 Redis 里的 Neo4j——用 Cypher 查询关系网络（注意：已 EOL）
- **RedisTimeSeries** 是 Redis 里的 InfluxDB——高效处理时序数据
- Redis 的模块生态让它从"缓存工具"进化为"全能数据平台"
- 选择合适的模块，就像选择合适的瑞士军刀刀片——在特定场景下事半功倍
