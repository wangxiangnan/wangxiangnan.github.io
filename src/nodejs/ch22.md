## 22.1 ORM vs 原生驱动：选择策略

```
┌──────────────┬────────────────────┬──────────────────────┐
│   维度        │    ORM             │    原生驱动           │
├──────────────┼────────────────────┼──────────────────────┤
│ 开发效率      │ 高（自动映射）      │ 低（手写SQL）         │
│ 性能         │ 略低（抽象层开销）   │ 最高                 │
│ 类型安全      │ 优秀（Prisma等）    │ 需手动定义           │
│ 学习曲线      │ 需学ORM API        │ 需精通SQL            │
│ 灵活性       │ 受ORM能力限制       │ 完全自由             │
│ 迁移管理      │ 内置               │ 需额外工具           │
│ 适合场景      │ CRUD为主的业务      │ 复杂查询、高性能要求  │
└──────────────┴────────────────────┴──────────────────────┘
```

**我的建议**：

- 新项目/CRUD为主 → **Prisma** 或 **Drizzle**
- 复杂查询/高性能 → **原生驱动** + 手写SQL
- 中间路线 → **Drizzle**（既有类型安全，又接近SQL）

```typescript
// 连接池配置
const pool: Pool = mysql.createPool({

  user: process.env.DB_USER || 'root',
  password: process.env.DB_PASSWORD || '',
  database: process.env.DB_NAME || 'myapp',
  
  // 连接池配置
  waitForConnections: true,   // 无可用连接时等待
  connectionLimit: 20,        // 最大连接数
  maxIdle: 10,               // 最大空闲连接数
  idleTimeout: 60000,        // 空闲连接超时(ms)
  queueLimit: 0,             // 等待队列无限制
  enableKeepAlive: true,     // TCP keep-alive
  keepAliveInitialDelay: 30000,
  
  // 性能优化
  namedPlaceholders: true,   // 支持 :name 占位符
  decimalNumbers: true,      // DECIMAL返回number而非string
  dateStrings: false,        // 返回Date对象
  typeCast: true,            // 自动类型转换
  
  // 字符集
  charset: 'utf8mb4',
});

// 健康检查
async function healthCheck(): Promise<boolean> {
  try {
    const [rows] = await pool.query('SELECT 1');
    return true;
  } catch (error) {
    console.error('Database health check failed:', error);
    return false;
  }
}

// ===== 用户CRUD操作 =====

interface User {
  id: number;
  name: string;
  email: string;
  age: number;
  created_at: Date;
  updated_at: Date;
}

// 查询 - 使用预处理语句防止SQL注入
async function getUserById(id: number): Promise<User | null> {
  const [rows] = await pool.query<(User & RowDataPacket)[]>(
    'SELECT * FROM users WHERE id = ? LIMIT 1',
    [id]
  );
  return rows[0] || null;
}

// 查询列表 - 分页
async function getUsers(page: number = 1, limit: number = 20): Promise<{
  users: User[];
  total: number;
}> {
  const offset = (page - 1) * limit;
  
  // 并行执行查询和计数
  const [usersResult, countResult] = await Promise.all([
    pool.query<(User & RowDataPacket)[]>(
      'SELECT * FROM users ORDER BY created_at DESC LIMIT ? OFFSET ?',
      [limit, offset]
    ),
    pool.query<(RowDataPacket & { total: number })[]>(
      'SELECT COUNT(*) as total FROM users'
    ),
  ]);

  return {
    users: usersResult[0],
    total: countResult[0][0].total,
  };
}

// 插入
async function createUser(data: Omit<User, 'id' | 'created_at' | 'updated_at'>): Promise<number> {
  const [result] = await pool.query<ResultSetHeader>(
    'INSERT INTO users (name, email, age) VALUES (?, ?, ?)',
    [data.name, data.email, data.age]
  );
  return result.insertId;
}

// 更新
async function updateUser(id: number, data: Partial<Pick<User, 'name' | 'email' | 'age'>>): Promise<boolean> {
  const fields: string[] = [];
  const values: unknown[] = [];
  
  if (data.name !== undefined) { fields.push('name = ?'); values.push(data.name); }
  if (data.email !== undefined) { fields.push('email = ?'); values.push(data.email); }
  if (data.age !== undefined) { fields.push('age = ?'); values.push(data.age); }
  
  if (fields.length === 0) return false;
  
  values.push(id);
  const [result] = await pool.query<ResultSetHeader>(
    `UPDATE users SET ${fields.join(', ')}, updated_at = NOW() WHERE id = ?`,
    values
  );
  
  return result.affectedRows > 0;
}

export { pool, healthCheck, getUserById, getUsers, createUser, updateUser };
```

### 22.2.2 事务处理

```typescript
// transactions.ts
import { pool } from './mysql.js';
import type { PoolConnection, ResultSetHeader } from 'mysql2/promise';

// 转账示例 - 经典事务场景
async function transferMoney(
  fromUserId: number, 
  toUserId: number, 
  amount: number
): Promise<{ success: boolean; message: string }> {
  const connection: PoolConnection = await pool.getConnection();
  
  try {
    // 开始事务
    await connection.beginTransaction();

    // 1. 检查发送方余额
    const [senderRows] = await connection.query<any[]>(
      'SELECT balance FROM accounts WHERE user_id = ? FOR UPDATE', // FOR UPDATE加行锁
      [fromUserId]
    );
    
    if (senderRows.length === 0) {
      throw new Error('Sender account not found');
    }
    
    if (senderRows[0].balance < amount) {
      throw new Error('Insufficient balance');
    }

    // 2. 扣款
    await connection.query(
      'UPDATE accounts SET balance = balance - ? WHERE user_id = ?',
      [amount, fromUserId]
    );

    // 3. 加款
    const [receiverResult] = await connection.query<ResultSetHeader>(
      'UPDATE accounts SET balance = balance + ? WHERE user_id = ?',
      [amount, toUserId]
    );
    
    if (receiverResult.affectedRows === 0) {
      throw new Error('Receiver account not found');
    }

    // 4. 记录交易日志
    await connection.query(
      'INSERT INTO transactions (from_user, to_user, amount, type) VALUES (?, ?, ?, ?)',
      [fromUserId, toUserId, amount, 'transfer']
    );

    // 提交事务
    await connection.commit();
    
    return { success: true, message: `Transferred ${amount} successfully` };
  } catch (error) {
    // 回滚事务
    await connection.rollback();
    return { 
      success: false, 
      message: error instanceof Error ? error.message : 'Unknown error' 
    };
  } finally {
    // 释放连接回连接池（重要！否则连接泄漏）
    connection.release();
  }
}

// 通用事务包装器
async function withTransaction<T>(
  fn: (connection: PoolConnection) => Promise<T>
): Promise<T> {
  const connection = await pool.getConnection();
  
  try {
    await connection.beginTransaction();
    const result = await fn(connection);
    await connection.commit();
    return result;
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}

// 使用通用事务包装器
async function batchInsertUsers(users: Array<{ name: string; email: string }>): Promise<number[]> {
  return withTransaction(async (conn) => {
    const ids: number[] = [];
    
    for (const user of users) {
      const [result] = await conn.query<ResultSetHeader>(
        'INSERT INTO users (name, email) VALUES (?, ?)',
        [user.name, user.email]
      );
      ids.push(result.insertId);
    }
    
    return ids;
  });
}

export { transferMoney, withTransaction, batchInsertUsers };
```

```typescript
const globalForPrisma = globalThis as unknown as {

// ===== Prisma CRUD操作 =====

// 创建用户（带关联数据）
async function createUserWithProfile(data: {
  email: string;
  name: string;
  bio?: string;
}) {
  return prisma.user.create({
    data: {
      email: data.email,
      name: data.name,
      profile: data.bio ? {
        create: { bio: data.bio },
      } : undefined,
    },
    include: {
      profile: true,
    },
  });
}

// 复杂查询：分页 + 过滤 + 排序
async function searchUsers(params: {
  keyword?: string;
  role?: 'USER' | 'ADMIN';
  page?: number;
  limit?: number;
  orderBy?: 'name' | 'createdAt';
  order?: 'asc' | 'desc';
}) {
  const { keyword, role, page = 1, limit = 20, orderBy = 'createdAt', order = 'desc' } = params;
  
  const where: Prisma.UserWhereInput = {
    ...(keyword && {
      OR: [
        { name: { contains: keyword } },
        { email: { contains: keyword } },
      ],
    }),
    ...(role && { role }),
  };

  const [users, total] = await Promise.all([
    prisma.user.findMany({
      where,
      skip: (page - 1) * limit,
      take: limit,
      orderBy: { [orderBy]: order },
      include: {
        profile: true,
        _count: { select: { posts: true, orders: true } },
      },
    }),
    prisma.user.count({ where }),
  ]);

  return {
    users,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit),
    },
  };
}

// 事务操作
async function createOrderWithItems(userId: number, items: Array<{
  productId: number;
  quantity: number;
  price: number;
}>) {
  return prisma.$transaction(async (tx) => {
    // 计算总金额
    const totalAmount = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
    
    // 创建订单
    const order = await tx.order.create({
      data: {
        userId,
        amount: totalAmount,
        items: {
          create: items.map(item => ({
            productId: item.productId,
            quantity: item.quantity,
            price: item.price,
          })),
        },
      },
      include: { items: true },
    });

    // 这里可以做其他关联操作，如扣减库存等
    // 如果任何操作失败，整个事务自动回滚

    return order;
  });
}

// 原始SQL（需要时可以用）
async function getTopSpenders(limit: number = 10) {
  return prisma.$queryRaw<Array<{ userId: number; name: string; totalSpent: number }>>`
    SELECT u.id as userId, u.name, SUM(o.amount) as totalSpent
    FROM users u
    JOIN orders o ON u.id = o.user_id
    WHERE o.status = 'PAID'
    GROUP BY u.id, u.name
    ORDER BY totalSpent DESC
    LIMIT ${limit}
  `;
}
```

### 22.2.4 Drizzle ORM：轻量级类型安全方案

```typescript
// db/drizzle-schema.ts
import { 
  mysqlTable, int, varchar, text, boolean, 
  decimal, timestamp, mysqlEnum, index, uniqueIndex 
} from 'drizzle-orm/mysql-core';
import { relations } from 'drizzle-orm';

// 表定义（比Prisma更接近SQL思维）
export const users = mysqlTable('users', {
  id: int('id').primaryKey().autoincrement(),
  name: varchar('name', { length: 255 }).notNull(),
  email: varchar('email', { length: 255 }).notNull(),
  age: int('age'),
  role: mysqlEnum('role', ['user', 'admin', 'moderator']).default('user'),
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow().onUpdateNow(),
}, (table) => ({
  emailIdx: uniqueIndex('email_idx').on(table.email),
}));

export const posts = mysqlTable('posts', {
  id: int('id').primaryKey().autoincrement(),
  title: varchar('title', { length: 500 }).notNull(),
  content: text('content'),
  published: boolean('published').default(false),
  authorId: int('author_id').notNull().references(() => users.id),
  createdAt: timestamp('created_at').defaultNow(),
}, (table) => ({
  authorIdx: index('author_idx').on(table.authorId),
}));

// 关系定义
export const usersRelations = relations(users, ({ many, one }) => ({
  posts: many(posts),
}));

export const postsRelations = relations(posts, ({ one }) => ({
  author: one(users, {
    fields: [posts.authorId],
    references: [users.id],
  }),
}));
```

```typescript
async function getUserPosts(userId: number) {
  return db.query.users.findFirst({
    where: eq(schema.users.id, userId),
    with: {
      posts: {
        where: eq(schema.posts.published, true),
        orderBy: [desc(schema.posts.createdAt)],
        limit: 10,
      },
    },
  });
}

// Select Builder（更接近SQL的写法）
async function searchPosts(keyword: string, page: number = 1, limit: number = 20) {
  const offset = (page - 1) * limit;
  
  const results = await db
    .select({
      id: schema.posts.id,
      title: schema.posts.title,
      authorName: schema.users.name,
      createdAt: schema.posts.createdAt,
    })
    .from(schema.posts)
    .innerJoin(schema.users, eq(schema.posts.authorId, schema.users.id))
    .where(
      and(
        eq(schema.posts.published, true),
        or(
          like(schema.posts.title, `%${keyword}%`),
          like(schema.posts.content, `%${keyword}%`),
        ),
      )
    )
    .orderBy(desc(schema.posts.createdAt))
    .limit(limit)
    .offset(offset);

  const [{ total }] = await db
    .select({ total: count() })
    .from(schema.posts)
    .where(eq(schema.posts.published, true));

  return { results, total, page, limit };
}

      .update(schema.posts)
      .set({ published: true })
      .where(eq(schema.posts.id, postId));

    // 这里可以插入通知记录等
    // 任何错误会自动回滚
  });
}
```

```typescript
async function cacheSet(key: string, value: unknown, ttlSeconds: number = 3600): Promise<void> {
  const serialized = JSON.stringify(value);
  await redis.set(key, serialized, 'EX', ttlSeconds);
}

async function cacheGet<T>(key: string): Promise<T | null> {
  const data = await redis.get(key);
  if (!data) return null;
  return JSON.parse(data) as T;
}

// Hash操作（适合存储对象）
async function setUserSession(userId: string, session: Record<string, string>): Promise<void> {
  const key = `session:${userId}`;
  await redis.hmset(key, session);
  await redis.expire(key, 86400); // 24小时过期
}

async function getUserSession(userId: string): Promise<Record<string, string> | null> {
  const key = `session:${userId}`;
  const data = await redis.hgetall(key);
  return Object.keys(data).length > 0 ? data : null;
}

// 有序集合（排行榜）
async function updateLeaderboard(userId: string, score: number): Promise<void> {
  await redis.zadd('leaderboard', score, userId);
}

async function getTopPlayers(limit: number = 10): Promise<Array<{ userId: string; score: number }>> {
  const results = await redis.zrevrange('leaderboard', 0, limit - 1, 'WITHSCORES');
  const players: Array<{ userId: string; score: number }> = [];
  
  for (let i = 0; i < results.length; i += 2) {
    players.push({
      userId: results[i],
      score: Number(results[i + 1]),
    });
  }
  
  return players;
}

export { redis, cacheSet, cacheGet, setUserSession, getUserSession, updateLeaderboard, getTopPlayers };
```

### 22.3.2 Pipeline：批量操作优化

```typescript
// redis-pipeline.ts
import { redis } from './redis.js';

// 没有Pipeline：每个命令都是独立的网络请求
// 100个命令 = 100次网络往返 = 慢！
async function withoutPipeline(keys: string[]): Promise<(string | null)[]> {
  const results: (string | null)[] = [];
  for (const key of keys) {
    results.push(await redis.get(key)); // 每次都等待网络响应
  }
  return results;
}

// 使用Pipeline：所有命令打包发送，一次往返
// 100个命令 = 1次网络往返 = 快！
async function withPipeline(keys: string[]): Promise<(string | null)[]> {
  const pipeline = redis.pipeline();
  
  for (const key of keys) {
    pipeline.get(key); // 只是添加到队列，不立即发送
  }
  
  const results = await pipeline.exec(); // 一次性发送所有命令
  return results!.map(([err, val]) => {
    if (err) throw err;
    return val as string | null;
  });
}

// 性能对比
async function benchmarkPipeline(): Promise<void> {
  const keys = Array.from({ length: 1000 }, (_, i) => `bench:key:${i}`);
  
  // 先设置一些数据
  const setPipeline = redis.pipeline();
  keys.forEach((key, i) => setPipeline.set(key, `value-${i}`, 'EX', 60));
  await setPipeline.exec();

  // 对比
  console.time('Without Pipeline');
  await withoutPipeline(keys.slice(0, 100));
  console.timeEnd('Without Pipeline');
```

### 22.3.3 缓存策略

```typescript
// cache-strategies.ts
import { redis } from './redis.js';

// ===== Cache-Aside（旁路缓存）=====
// 读：先查缓存 → 命中则返回 → 未命中则查DB → 写缓存 → 返回
// 写：更新DB → 删除缓存（注意顺序！）

async function getUserWithCache(userId: number): Promise<User | null> {
  const cacheKey = `user:${userId}`;
  
  // 1. 查缓存
  const cached = await redis.get(cacheKey);
  if (cached) {
    return JSON.parse(cached);
  }
  
  // 2. 缓存未命中，查数据库
  const user = await db.users.findUnique({ where: { id: userId } });
  
  if (user) {
    // 3. 写入缓存（带TTL）
    await redis.set(cacheKey, JSON.stringify(user), 'EX', 3600);
  }
  
  return user;
}

async function updateUserWithCache(userId: number, data: Partial<User>): Promise<void> {
  // 1. 先更新数据库
  await db.users.update({ where: { id: userId }, data });
  
  // 2. 再删除缓存（不是更新！避免并发问题）
  await redis.del(`user:${userId}`);
}

// ===== 缓存穿透防护 =====
// 问题：恶意请求不存在的key，每次都打到DB
// 方案：缓存空值 + 布隆过滤器

async function getUserSafe(userId: number): Promise<User | null> {
  const cacheKey = `user:${userId}`;
  
  const cached = await redis.get(cacheKey);
  
  // 缓存命中（包括空值标记）
  if (cached !== null) {
    if (cached === '__NULL__') return null; // 空值标记
    return JSON.parse(cached);
  }
  
  // 查DB
  const user = await db.users.findUnique({ where: { id: userId } });
  
  if (user) {
    await redis.set(cacheKey, JSON.stringify(user), 'EX', 3600);
  } else {
    // 缓存空值，TTL短一些
    await redis.set(cacheKey, '__NULL__', 'EX', 300); // 5分钟
  }
  
  return user;
}

// ===== 缓存雪崩防护 =====
// 问题：大量缓存同时过期，瞬间全打到DB
// 方案：TTL加随机偏移

function randomTTL(baseTTL: number): number {
  // 在基础TTL上加减20%的随机值
  const jitter = baseTTL * 0.2;
  return baseTTL + Math.floor(Math.random() * jitter * 2 - jitter);
}

// ===== 缓存击穿防护（互斥锁） =====
// 问题：热点key过期瞬间，大量并发请求同时查DB
// 方案：分布式锁，只让一个请求去查DB

async function getHotDataWithLock(key: string, fetchFn: () => Promise<unknown>): Promise<unknown> {
  const cacheKey = `data:${key}`;
  const lockKey = `lock:${key}`;
  
  // 查缓存
  const cached = await redis.get(cacheKey);
  if (cached) return JSON.parse(cached);
  
  // 尝试获取锁
  const acquired = await redis.set(lockKey, '1', 'EX', 10, 'NX'); // NX=不存在才设置
  
  if (acquired) {
    try {
      // 获取锁成功，去查DB
      const data = await fetchFn();
      await redis.set(cacheKey, JSON.stringify(data), 'EX', randomTTL(3600));
      return data;
    } finally {
      await redis.del(lockKey); // 释放锁
    }
  } else {
    // 没获取到锁，等待后重试
    await new Promise(r => setTimeout(r, 100));
    return getHotDataWithLock(key, fetchFn); // 递归重试
  }
}

// 这里假设db和User类型已在上文定义
interface User { id: number; name: string; email: string; }
const db = { users: { findUnique: async (_: any): Promise<User | null> => null, update: async (_: any) => {} } };
```

### 22.3.4 分布式锁实现

```typescript
// distributed-lock.ts
import { redis } from './redis.js';
import crypto from 'node:crypto';

class DistributedLock {
  private redis;
  private lockValue: string;
  
  constructor(redisClient: typeof redis) {
    this.redis = redisClient;
    this.lockValue = crypto.randomUUID(); // 每个锁实例唯一标识
  }

  /**
   * 获取锁
   * @param key 锁的key
   * @param ttl 锁超时时间(秒)，防止死锁
   * @param retries 重试次数
   * @param retryDelay 重试间隔(ms)
   */
  async acquire(
    key: string, 
    ttl: number = 30, 
    retries: number = 10, 
    retryDelay: number = 200
  ): Promise<boolean> {
    const lockKey = `lock:${key}`;
    
    for (let i = 0; i < retries; i++) {
      // SET key value EX ttl NX
      // NX: 只在key不存在时设置（原子操作）
      const result = await this.redis.set(lockKey, this.lockValue, 'EX', ttl, 'NX');
      
      if (result === 'OK') {
        return true; // 获取锁成功
      }
      
      // 获取失败，等待后重试
      await new Promise(resolve => setTimeout(resolve, retryDelay));
    }
    
    return false; // 重试耗尽，获取失败
  }

  /**
   * 释放锁（使用Lua脚本保证原子性）
   * 只有锁的持有者才能释放（通过lockValue验证）
   */
  async release(key: string): Promise<boolean> {
    const lockKey = `lock:${key}`;
    
    // Lua脚本：检查value是否匹配，匹配则删除
    // 这是原子操作，避免误删别人的锁
    const script = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("del", KEYS[1])
      else
        return 0
      end
    `;
    
    const result = await this.redis.eval(script, 1, lockKey, this.lockValue);
    return result === 1;
  }

  /**
   * 续期（看门狗机制）
   * 防止业务执行时间超过锁TTL导致锁自动释放
   */
  async extend(key: string, ttl: number): Promise<boolean> {
    const lockKey = `lock:${key}`;
    
    const script = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("expire", KEYS[1], ARGV[2])
      else
        return 0
      end
    `;
    
    const result = await this.redis.eval(script, 1, lockKey, this.lockValue, ttl);
    return result === 1;
  }
}

// 使用分布式锁的高级包装
async function withLock<T>(
  key: string,
  fn: () => Promise<T>,
  options: { ttl?: number; retries?: number } = {}
): Promise<T> {
  const lock = new DistributedLock(redis);
  const { ttl = 30, retries = 10 } = options;
  
  const acquired = await lock.acquire(key, ttl, retries);
  if (!acquired) {
    throw new Error(`Failed to acquire lock: ${key}`);
  }
  
  // 启动看门狗自动续期
  const watchdog = setInterval(async () => {
    await lock.extend(key, ttl);
  }, (ttl * 1000) / 3); // 每1/3 TTL续期一次
  
  try {
    return await fn();
  } finally {
    clearInterval(watchdog);
    await lock.release(key);
  }
}

// 实际应用：防止重复扣款
async function deductBalance(userId: number, amount: number): Promise<boolean> {
  return withLock(`deduct:${userId}`, async () => {
    // 临界区：同一用户同一时刻只有一个请求能执行
    const balance = await getBalance(userId);
    if (balance < amount) return false;
    
    await updateBalance(userId, balance - amount);
    return true;
  });
}

async function updateBalance(userId: number, newBalance: number): Promise<void> {}

export { DistributedLock, withLock };
```

```typescript
// redis-pubsub.ts
import Redis from 'ioredis';

// Pub/Sub需要独立的连接（订阅后连接进入订阅模式，不能执行其他命令）
const subscriber = new Redis({ host: 'localhost', port: 6379 });
const publisher = new Redis({ host: 'localhost', port: 6379 });

// 定义事件类型
interface AppEvents {
  'user:registered': { userId: number; email: string };
  'order:created': { orderId: number; userId: number; amount: number };
  'order:paid': { orderId: number; paymentId: string };
  'notification:send': { userId: number; title: string; body: string };
}

type EventName = keyof AppEvents;

// 类型安全的发布
async function publish<E extends EventName>(event: E, data: AppEvents[E]): Promise<void> {
  const message = JSON.stringify({ event, data, timestamp: Date.now() });
  await publisher.publish(`app:${event}`, message);
}

// 类型安全的订阅
function subscribe<E extends EventName>(
  event: E, 
  handler: (data: AppEvents[E]) => void | Promise<void>
): void {
  const channel = `app:${event}`;
  
  subscriber.subscribe(channel, (err) => {
    if (err) console.error(`Failed to subscribe to ${channel}:`, err);
    else console.log(`Subscribed to ${channel}`);
  });

  subscriber.on('message', (ch, message) => {
    if (ch === channel) {
      const parsed = JSON.parse(message);
      handler(parsed.data);
    }
  });
}

// 使用示例

  console.log(`New user registered: ${data.email}`);
  // 发送欢迎邮件
  // 初始化用户配置

  console.log(`Order ${data.orderId} paid: ${data.paymentId}`);
  // 更新订单状态
  // 通知仓库发货
});
```

```typescript
// db/connection-manager.ts
import { EventEmitter } from 'node:events';

interface DatabaseConfig {
  mysql: { host: string; port: number; database: string };
  redis: { host: string; port: number };
  mongodb: { uri: string };
}

class ConnectionManager extends EventEmitter {
  private connections: Map<string, unknown> = new Map();
  private healthChecks: Map<string, NodeJS.Timeout> = new Map();

  async connectAll(config: DatabaseConfig): Promise<void> {
    const results = await Promise.allSettled([
      this.connectMySQL(config.mysql),
      this.connectRedis(config.redis),
      this.connectMongoDB(config.mongodb),
    ]);

    results.forEach((result, index) => {
      const name = ['MySQL', 'Redis', 'MongoDB'][index];
      if (result.status === 'rejected') {
        console.error(`Failed to connect ${name}:`, result.reason);
      }
    });
  }

  async disconnectAll(): Promise<void> {
    // 停止所有健康检查
    for (const [name, timer] of this.healthChecks) {
      clearInterval(timer);
    }
    this.healthChecks.clear();

    // 关闭所有连接
    const disconnections = [...this.connections.entries()].map(
      async ([name, conn]) => {
        try {
          // 根据类型调用对应的关闭方法
          if (name === 'mysql') await (conn as any).end();
          if (name === 'redis') await (conn as any).quit();
          if (name === 'mongodb') await (conn as any).disconnect();
          console.log(`${name} disconnected`);
        } catch (err) {
          console.error(`Error disconnecting ${name}:`, err);
        }
      }
    );

    await Promise.all(disconnections);
    this.connections.clear();
  }

  // 优雅关闭
  setupGracefulShutdown(): void {
    const shutdown = async (signal: string) => {
      console.log(`\nReceived ${signal}, shutting down gracefully...`);
      await this.disconnectAll();

    };

    process.on('SIGTERM', () => shutdown('SIGTERM'));

  private async connectMongoDB(config: any): Promise<void> { /* ... */ }
```

## 22.6 常见坑点

### 坑点1：连接泄漏

