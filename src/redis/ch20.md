## 20.1 缓存的基本概念

在正式进入各种模式之前，我们先搞懂两个核心术语：

- **Cache Hit（缓存命中）**：你要的数据在小纸条上！直接从 Redis 返回，速度飞快，通常在 1ms 以内。
- **Cache Miss（缓存未命中）**：小纸条上没有这个数据。只能去数据库"翻书"了，可能需要 10ms-100ms 甚至更久。

衡量缓存效果的核心指标是 **缓存命中率**：

```text
缓存命中率 = Cache Hit 次数 / 总请求次数 * 100%
```

一个好的缓存系统，命中率通常在 95% 以上。如果你的命中率低于 80%，那就要反思一下缓存策略了——就像你抄了一堆小纸条，结果考试一题都没考到，那还不如不抄。

## 20.2 Cache-Aside（旁路缓存）模式

Cache-Aside 是最经典、最常用的缓存模式，也是大多数团队的第一选择。它之所以叫"旁路"，是因为缓存和数据库是"并行"存在的，应用程序需要自己协调两者的关系。

### 20.2.1 读流程

```
客户端请求 -> 查Redis缓存 -> 命中？
  |-- 是 -> 直接返回缓存数据
  +-- 否 -> 查数据库 -> 写入Redis缓存 -> 返回数据
```

### 20.2.2 写流程

```text
客户端请求更新 -> 更新数据库 -> 删除Redis缓存（注意：是删除，不是更新！）
```

> **为什么写操作是删除缓存而不是更新缓存？**
>
> 想象一个场景：线程A更新数据库为"值1"，线程B紧接着更新数据库为"值2"。如果是更新缓存，由于并发和延迟，可能出现缓存里留下"值1"而数据库里是"值2"的不一致情况。但如果是删除缓存，下次读取时会从数据库加载最新的"值2"，一致性就有了保证。

### 20.2.3 Node.js 完整实现

```js
const Redis = require('ioredis');

// 创建 Redis 客户端
const redis = new Redis({
  host: '127.0.0.1',
  port: 6379,
  password: 'your_password',
  db: 0,
});

// 模拟数据库
const fakeDB = {
  users: {
    '1': { id: 1, name: '张三', age: 28, email: 'zhangsan@example.com' },
    '2': { id: 2, name: '李四', age: 32, email: 'lisi@example.com' },
    '3': { id: 3, name: '王五', age: 25, email: 'wangwu@example.com' },
  },
};

// 模拟数据库查询（带延迟）
function queryDB(table, id) {
  return new Promise((resolve) => {
    setTimeout(() => {
      const data = fakeDB[table]?.[id] || null;
      console.log(`  [DB] 查询 ${table}:${id}，结果: ${data ? '找到' : '未找到'}`);
      resolve(data);
    }, 50);
  });
}

// 模拟数据库更新
function updateDB(table, id, data) {
  return new Promise((resolve) => {
    setTimeout(() => {
      if (fakeDB[table]) {
        fakeDB[table][id] = { ...fakeDB[table][id], ...data };
      }
      console.log(`  [DB] 更新 ${table}:${id}`);
      resolve(fakeDB[table]?.[id]);
    }, 30);
  });
}

/**
 * Cache-Aside 模式的缓存管理器
 */
class CacheAsideManager {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.defaultTTL = options.defaultTTL || 3600;
    this.keyPrefix = options.keyPrefix || 'cache';
  }

  _buildKey(table, id) {
    return `${this.keyPrefix}:${table}:${id}`;
  }

  /**
   * 读取数据：先查缓存，未命中则查数据库
   */
  async get(table, id) {
    const cacheKey = this._buildKey(table, id);

    const cached = await this.redis.get(cacheKey);
    if (cached) {
      console.log(`  [Cache] 命中 ${cacheKey}`);
      return JSON.parse(cached);
    }

    console.log(`  [Cache] 未命中 ${cacheKey}`);
    const data = await queryDB(table, id);
    if (data) {
      await this.redis.set(cacheKey, JSON.stringify(data), 'EX', this.defaultTTL);
      console.log(`  [Cache] 已写入 ${cacheKey}，TTL=${this.defaultTTL}s`);
    }
    return data;
  }

  /**
   * 更新数据：更新数据库 -> 删除缓存
   */
  async update(table, id, data) {
    const cacheKey = this._buildKey(table, id);
    const updated = await updateDB(table, id, data);
    await this.redis.del(cacheKey);
    console.log(`  [Cache] 已删除 ${cacheKey}`);
    return updated;
  }

  /**
   * 批量获取
   */
  async mget(table, ids) {
    const cacheKeys = ids.map((id) => this._buildKey(table, id));
    const pipeline = this.redis.pipeline();
    cacheKeys.forEach((key) => pipeline.get(key));
    const results = await pipeline.exec();

    const data = {};
    const missedIds = [];

    results.forEach(([err, value], index) => {
      if (value) {
        data[ids[index]] = JSON.parse(value);
      } else {
        missedIds.push(ids[index]);
      }
    });

    if (missedIds.length > 0) {
      console.log(`  [Cache] 批量未命中: ${missedIds.join(', ')}`);
      for (const id of missedIds) {
        const record = await queryDB(table, id);
        if (record) {
          data[id] = record;
          await this.redis.set(
            this._buildKey(table, id),
            JSON.stringify(record),
            'EX',
            this.defaultTTL
          );
        }
      }
    }
    return data;
  }
}

// 使用示例
async function main() {
  const cache = new CacheAsideManager(redis, { defaultTTL: 300, keyPrefix: 'app' });

  console.log('=== 第一次读取（缓存未命中） ===');
  const user1 = await cache.get('users', '1');
  console.log('结果:', user1);

  console.log('\n=== 第二次读取（缓存命中） ===');
  const user1Again = await cache.get('users', '1');
  console.log('结果:', user1Again);

  console.log('\n=== 更新用户（删除缓存） ===');
  const updated = await cache.update('users', '1', { age: 29 });
  console.log('更新后:', updated);

  console.log('\n=== 更新后再次读取（缓存未命中，重新加载） ===');
  const user1Updated = await cache.get('users', '1');
  console.log('结果:', user1Updated);

  console.log('\n=== 批量获取 ===');
  const users = await cache.mget('users', ['1', '2', '3']);
  console.log('结果:', users);

  redis.disconnect();
}

main().catch(console.error);
```

> **Tip：Pipeline 批量操作**
>
> 注意上面 `mget` 方法中使用了 `pipeline`。当你需要一次性执行多个 Redis 命令时，pipeline 可以将它们打包发送，减少网络往返次数。就像你把10封信装进一个信封邮寄，而不是分10次邮寄——网络延迟从 10RTT 降低到 1RTT。

## 20.3 Read-Through / Write-Through 模式

Cache-Aside 模式需要应用程序自己管理缓存和数据库的交互。而 Read-Through 和 Write-Through 模式则是把这个"协调"的工作交给了缓存层本身。

**Read-Through**：应用程序只和缓存层交互。缓存未命中时，缓存层自己去数据库加载数据。

**Write-Through**：应用程序只和缓存层交互。写操作时，缓存层同步写入数据库和缓存。

```text
Read-Through:
  App -> Cache Layer -> (miss?) -> Database
                     -> (hit!)  -> 返回

Write-Through:
  App -> Cache Layer -> Database (同步写入)
                     -> Cache (同步更新)
```

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

/**
 * Read-Through 缓存层
 */
class ReadThroughCache {
  constructor(redisClient, loaderFn, options = {}) {
    this.redis = redisClient;
    this.loader = loaderFn;
    this.ttl = options.ttl || 3600;
    this.prefix = options.prefix || 'rt';
  }

  async get(key) {
    const cacheKey = `${this.prefix}:${key}`;
    const cached = await this.redis.get(cacheKey);

    if (cached !== null) {
      return JSON.parse(cached);
    }

    console.log(`  [ReadThrough] Cache miss for "${key}", loading...`);
    const data = await this.loader(key);

    if (data !== null && data !== undefined) {
      await this.redis.set(cacheKey, JSON.stringify(data), 'EX', this.ttl);
    }
    return data;
  }

  async set(key, value) {
    const cacheKey = `${this.prefix}:${key}`;
    await this.redis.set(cacheKey, JSON.stringify(value), 'EX', this.ttl);
  }
}

// 使用示例
const userCache = new ReadThroughCache(
  redis,
  async (key) => {
    const dbResult = await queryDB('users', key);
    return dbResult;
  },
  { ttl: 600, prefix: 'user' }
);

const user = await userCache.get('42');
```

> **Warning：Write-Through 的陷阱**
>
> Write-Through 模式虽然保证了强一致性，但每次写操作都要同时写缓存和数据库，会增加写操作的延迟。如果你的应用是写多读少的场景，Write-Through 可能不是最佳选择。

## 20.4 Write-Behind（异步写回）模式

Write-Behind 是 Write-Through 的"懒人版本"——写操作只更新缓存，然后异步批量同步到数据库。

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

class WriteBehindCache {
  constructor(redisClient, dbWriterFn, options = {}) {
    this.redis = redisClient;
    this.dbWriter = dbWriterFn;
    this.flushInterval = options.flushInterval || 5000;
    this.prefix = options.prefix || 'wb';
    this.dirtyKeys = new Map();
    this._timer = setInterval(() => this._flush(), this.flushInterval);
  }

  async set(key, value) {
    const cacheKey = `${this.prefix}:${key}`;
    await this.redis.set(cacheKey, JSON.stringify(value));
    this.dirtyKeys.set(key, value);
    console.log(`  [WriteBehind] 缓存已写入: ${key}`);
    return value;
  }

  async get(key) {
    const cacheKey = `${this.prefix}:${key}`;
    const cached = await this.redis.get(cacheKey);
    if (cached) return JSON.parse(cached);
    if (this.dirtyKeys.has(key)) return this.dirtyKeys.get(key);
    return null;
  }

  async _flush() {
    if (this.dirtyKeys.size === 0) return;
    const batch = new Map(this.dirtyKeys);
    this.dirtyKeys.clear();
    console.log(`  [WriteBehind] 刷盘: ${batch.size} 条脏数据`);
    try {
      await this.dbWriter(batch);
      console.log(`  [WriteBehind] 刷盘成功`);
    } catch (err) {
      console.error(`  [WriteBehind] 刷盘失败:`, err.message);
      for (const [k, v] of batch) this.dirtyKeys.set(k, v);
    }
  }

  destroy() {
    clearInterval(this._timer);
  }
}

const productCache = new WriteBehindCache(
  redis,
  async (batch) => {
    for (const [key, value] of batch) {
      console.log(`    [DB Batch] 写入 product:${key}`);
    }
  },
  { flushInterval: 3000 }
);

await productCache.set('sku001', { name: 'iPhone 16', price: 7999 });
await productCache.set('sku002', { name: 'Pixel 9', price: 5999 });
```

> **Warning：Write-Behind 的数据丢失风险**
>
> 如果缓存节点在数据刷盘之前宕机，脏数据就永远丢失了。适合对一致性要求不高但写入性能要求极高的场景，比如游戏积分、点赞数。

## 20.5 缓存的三大问题

面试高频考点来了！穿透、击穿、雪崩——缓存世界的三座大山。

### 20.5.1 缓存穿透（Cache Penetration）

**问题描述**：大量请求查询根本不存在的 key。缓存查不到 -> 去数据库查 -> 数据库也查不到 -> 不写缓存 -> 下次还来查数据库。

**解决方案一：缓存空值**

```js
async function getUserWithNullCache(userId) {
  const cacheKey = `user:${userId}`;
  const cached = await redis.get(cacheKey);

  if (cached !== null) {
    return JSON.parse(cached);
  }

  const user = await queryDB('users', userId);

  if (user) {
    await redis.set(cacheKey, JSON.stringify(user), 'EX', 3600);
  } else {
    // 空值也缓存，但只缓存5分钟
    await redis.set(cacheKey, JSON.stringify(null), 'EX', 300);
    console.log(`  [穿透防护] 缓存空值: ${cacheKey}，TTL=300s`);
  }

  return user;
}
```

**解决方案二：布隆过滤器**

```js
class SimpleBloomFilter {
  constructor(size = 1000000, hashCount = 3) {
    this.size = size;
    this.hashCount = hashCount;
    this.bitArray = Buffer.alloc(Math.ceil(size / 8));
  }

  _hashes(item) {
    const hashes = [];
    for (let i = 0; i < this.hashCount; i++) {
      let hash = 0;
      const str = `${item}:${i}`;
      for (let j = 0; j < str.length; j++) {
        hash = ((hash << 5) - hash + str.charCodeAt(j)) | 0;
      }
      hashes.push(Math.abs(hash) % this.size);
    }
    return hashes;
  }

  add(item) {
    const positions = this._hashes(item);
    for (const pos of positions) {
      const byteIndex = Math.floor(pos / 8);
      const bitIndex = pos % 8;
      this.bitArray[byteIndex] |= 1 << bitIndex;
    }
  }

  mightExist(item) {
    const positions = this._hashes(item);
    for (const pos of positions) {
      const byteIndex = Math.floor(pos / 8);
      const bitIndex = pos % 8;
      if (!(this.bitArray[byteIndex] & (1 << bitIndex))) {
        return false;
      }
    }
    return true;
  }
}

const bloomFilter = new SimpleBloomFilter(100000, 3);
['1', '2', '3', '42', '100'].forEach((id) => bloomFilter.add(id));

async function getUserWithBloomFilter(userId) {
  if (!bloomFilter.mightExist(userId)) {
    console.log(`  [布隆过滤器] ${userId} 一定不存在，直接拒绝`);
    return null;
  }
  return getUserWithNullCache(userId);
}
```

### 20.5.2 缓存击穿（Cache Breakdown）

**问题描述**：热点 key 过期瞬间，大量并发请求同时打到数据库。

**解决方案：互斥锁**

```js
async function getWithMutex(key, dbQueryFn, ttl = 3600) {
  const cacheKey = `data:${key}`;
  const lockKey = `lock:${key}`;

  const cached = await redis.get(cacheKey);
  if (cached) return JSON.parse(cached);

  const lockValue = `${Date.now()}:${Math.random()}`;
  const locked = await redis.set(lockKey, lockValue, 'NX', 'EX', 10);

  if (locked) {
    try {
      const doubleCheck = await redis.get(cacheKey);
      if (doubleCheck) return JSON.parse(doubleCheck);

      console.log(`  [互斥锁] 获取锁成功，查询数据库: ${key}`);
      const data = await dbQueryFn(key);

      if (data) {
        await redis.set(cacheKey, JSON.stringify(data), 'EX', ttl);
      }
      return data;
    } finally {
      const luaScript = `
        if redis.call("get", KEYS[1]) == ARGV[1] then
          return redis.call("del", KEYS[1])
        else
          return 0
        end
      `;
      await redis.eval(luaScript, 1, lockKey, lockValue);
    }
  } else {
    await new Promise((r) => setTimeout(r, 50));
    return getWithMutex(key, dbQueryFn, ttl);
  }
}
```

**解决方案二：永不过期 + 异步刷新**

```js
async function getWithAsyncRefresh(key, dbQueryFn, logicalTTL = 3600) {
  const cacheKey = `data:${key}`;
  const metaKey = `meta:${key}`;

  const [cached, meta] = await Promise.all([
    redis.get(cacheKey),
    redis.get(metaKey),
  ]);

  if (cached) {
    const now = Date.now();
    const expireAt = meta ? parseInt(meta) : 0;

    if (now < expireAt) return JSON.parse(cached);

    // 逻辑过期，先返回旧数据，后台异步刷新
    const refreshLock = await redis.set(`refresh:${key}`, '1', 'NX', 'EX', 60);
    if (refreshLock) {
      (async () => {
        try {
          const freshData = await dbQueryFn(key);
          if (freshData) {
            await redis.set(cacheKey, JSON.stringify(freshData));
            await redis.set(metaKey, String(Date.now() + logicalTTL * 1000));
          }
        } catch (err) {
          console.error(`  [异步刷新] 刷新失败: ${key}`, err.message);
        }
      })();
    }
    return JSON.parse(cached);
  }

  const data = await dbQueryFn(key);
  if (data) {
    await redis.set(cacheKey, JSON.stringify(data));
    await redis.set(metaKey, String(Date.now() + logicalTTL * 1000));
  }
  return data;
}
```

### 20.5.3 缓存雪崩（Cache Avalanche）

**问题描述**：大量 key 同时过期，所有请求涌入数据库。

**解决方案：随机 TTL + 熔断降级**

```js
function randomTTL(baseTTL, variancePercent = 20) {
  const variance = baseTTL * (variancePercent / 100);
  const randomOffset = Math.random() * variance * 2 - variance;
  return Math.floor(baseTTL + randomOffset);
}

class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.resetTimeout = options.resetTimeout || 30000;
    this.failures = 0;
    this.state = 'CLOSED';
    this.lastFailureTime = 0;
  }

  async execute(fn) {
    if (this.state === 'OPEN') {
      if (Date.now() - this.lastFailureTime > this.resetTimeout) {
        this.state = 'HALF_OPEN';
      } else {
        return { degraded: true, message: '服务暂时不可用' };
      }
    }

    try {
      const result = await fn();
      if (this.state === 'HALF_OPEN') {
        this.state = 'CLOSED';
        this.failures = 0;
      }
      return result;
    } catch (err) {
      this.failures++;
      this.lastFailureTime = Date.now();
      if (this.failures >= this.failureThreshold) {
        this.state = 'OPEN';
        console.log(`  [熔断器] 触发熔断！连续失败 ${this.failures} 次`);
      }
      throw err;
    }
  }
}
```

> **Tip：三大问题记忆口诀**
>
> - **穿透**：查不到 -> 缓存空值 or 布隆过滤器
> - **击穿**：热点 key 过期 -> 互斥锁 or 永不过期
> - **雪崩**：集体过期 -> 随机 TTL + 多级缓存 + 熔断

## 20.7 多级缓存设计

就像 CPU 的 L1/L2/L3 Cache：

```
请求 -> L1（本地内存缓存） -> L2（Redis） -> L3（数据库）
       命中延迟 ~0.01ms       命中延迟 ~1ms    查询延迟 ~50ms
```

```js
const Redis = require('ioredis');
const NodeCache = require('node-cache');

const redis = new Redis({ host: '127.0.0.1', port: 6379 });
const localCache = new NodeCache({ stdTTL: 60, checkperiod: 30 });

class MultiLevelCache {
  constructor(localCache, redisClient, options = {}) {
    this.l1 = localCache;
    this.l2 = redisClient;
    this.l2TTL = options.l2TTL || 3600;
  }

  async get(key, dbQueryFn) {
    // L1: 查本地缓存
    let data = this.l1.get(key);
    if (data !== undefined) {
      console.log(`  [L1 本地缓存] 命中: ${key}`);
      return data;
    }

    // L2: 查 Redis
    const cached = await this.l2.get(`data:${key}`);
    if (cached) {
      data = JSON.parse(cached);
      console.log(`  [L2 Redis] 命中: ${key}`);
      this.l1.set(key, data);
      return data;
    }

    // L3: 查数据库
    console.log(`  [L3 数据库] 查询: ${key}`);
    data = await dbQueryFn(key);

    if (data) {
      await this.l2.set(`data:${key}`, JSON.stringify(data), 'EX', this.l2TTL);
      this.l1.set(key, data);
    }

    return data;
  }

  async invalidate(key) {
    this.l1.del(key);
    await this.l2.del(`data:${key}`);
  }
}
```

## 20.8 实战：构建完整的 API 缓存层

```js
const Redis = require('ioredis');
const crypto = require('crypto');

const redis = new Redis({
  host: '127.0.0.1',
  port: 6379,
  maxRetriesPerRequest: 3,
  retryStrategy(times) {
    return Math.min(times * 100, 3000);
  },
});

class APICacheLayer {
  constructor(redisClient, options = {}) {
    this.redis = redisClient;
    this.prefix = options.prefix || 'api';
    this.defaultTTL = options.defaultTTL || 300;
    this.nullTTL = options.nullTTL || 60;
    this.lockTimeout = options.lockTimeout || 5000;
    this.stats = { hits: 0, misses: 0, locks: 0 };
  }

  buildCacheKey(method, path, query = {}) {
    const raw = `${method}:${path}:${JSON.stringify(query)}`;
    const hash = crypto.createHash('md5').update(raw).digest('hex').slice(0, 12);
    return `${this.prefix}:${method}:${path}:${hash}`;
  }

  _randomTTL(base) {
    const jitter = base * 0.2;
    return Math.floor(base + (Math.random() * jitter * 2 - jitter));
  }

  async cachedFetch(method, path, query, fetchFn) {
    const cacheKey = this.buildCacheKey(method, path, query);

    const cached = await this.redis.get(cacheKey);
    if (cached) {
      this.stats.hits++;
      const parsed = JSON.parse(cached);
      if (parsed.__null) return null;
      return parsed;
    }

    this.stats.misses++;

    const lockKey = `${cacheKey}:lock`;
    const lockValue = `${Date.now()}`;
    const locked = await this.redis.set(lockKey, lockValue, 'NX', 'PX', this.lockTimeout);

    if (!locked) {
      await new Promise((r) => setTimeout(r, 30));
      const retryCached = await this.redis.get(cacheKey);
      if (retryCached) {
        const parsed = JSON.parse(retryCached);
        return parsed.__null ? null : parsed;
      }
      this.stats.locks++;
    }

    try {
      const doubleCheck = await this.redis.get(cacheKey);
      if (doubleCheck) {
        const parsed = JSON.parse(doubleCheck);
        return parsed.__null ? null : parsed;
      }

      const data = await fetchFn();

      if (data !== null && data !== undefined) {
        const ttl = this._randomTTL(this.defaultTTL);
        await this.redis.set(cacheKey, JSON.stringify(data), 'EX', ttl);
      } else {
        await this.redis.set(cacheKey, JSON.stringify({ __null: true }), 'EX', this.nullTTL);
      }

      return data;
    } finally {
      if (locked) {
        const lua = `
          if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
          end
          return 0
        `;
        await this.redis.eval(lua, 1, lockKey, lockValue);
      }
    }
  }

  getStats() {
    const total = this.stats.hits + this.stats.misses;
    return {
      ...this.stats,
      total,
      hitRate: total > 0 ? ((this.stats.hits / total) * 100).toFixed(2) + '%' : 'N/A',
    };
  }
}

const apiCache = new APICacheLayer(redis, { prefix: 'myapi', defaultTTL: 600 });

const result = await apiCache.cachedFetch(
  'GET', '/api/users/42', { include: 'posts' },
  async () => {
    await new Promise((r) => setTimeout(r, 100));
    return { id: 42, name: '张三', posts: 15 };
  }
);

console.log('API 结果:', result);
console.log('缓存统计:', apiCache.getStats());
```

## 小结

| 模式 | 特点 | 适用场景 |
|------|------|----------|
| Cache-Aside | 应用自己管理缓存 | 通用场景，最常用 |
| Read-Through | 缓存层代理读取 | 简化应用代码 |
| Write-Through | 缓存层同步写入DB | 强一致性要求 |
| Write-Behind | 缓存层异步写入DB | 写多读少，允许短暂不一致 |

三大问题与解决方案：穿透 -> 缓存空值+布隆过滤器；击穿 -> 互斥锁+逻辑过期；雪崩 -> 随机TTL+多级缓存+熔断。
