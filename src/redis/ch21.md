# 第21章：分布式锁——并发控制的守护者

## 引言：从单机锁到分布式锁

还记得操作系统课上的 mutex 吗？在单个进程里，mutex 保证同一时刻只有一个线程能进入临界区。这就像公厕的门锁——一个人进去锁上门，其他人只能在外面等。

但在分布式系统中，你的应用跑在 10 台服务器上。单机锁就像只在自家门上锁——隔壁老王照样能进。我们需要一把"全局锁"，让所有机器、所有进程都认。

Redis 凭借单线程执行模型和超快的响应速度，成为了实现分布式锁的绝佳选择。本章从最基础的 SETNX 开始，逐步升级到 Redlock 算法。

---

## 21.1 为什么需要分布式锁？

```
传统单机架构：
  [App进程] -> [单机锁] -> [共享资源]

分布式架构：
  [App-1] --+
  [App-2] --+--> [共享资源（数据库、Redis等）]
  [App-3] --+
  多台机器，各自有锁，互不认识
```

典型场景：库存扣减、定时任务去重、分布式 ID 生成、资源互斥访问。

## 21.2 Redis 分布式锁的原理

### 基本思路的陷阱：SETNX + EXPIRE

```
1. SETNX lock_key my_value  -> 不存在就设置（成功=拿到锁）
2. EXPIRE lock_key 30       -> 设置过期时间（防止死锁）
```

**问题**：SETNX 和 EXPIRE 不是原子操作。如果程序在两步之间崩溃，锁永远不过期——死锁！

### 正确做法：SET key value NX PX（原子操作）

```
SET lock_key unique_value NX PX 30000
```

- `NX`：只在 key 不存在时设置（互斥性）
- `PX 30000`：30000 毫秒过期（防止死锁）
- `unique_value`：UUID（确保只有持有者能释放锁）

> **为什么 value 需要唯一？** 线程A拿到锁但处理太慢导致锁过期，线程B拿到锁。此时线程A去释放锁——如果没有唯一标识，A就会删掉B的锁！

## 21.3 基础实现

```js
const Redis = require('ioredis');
const crypto = require('crypto');

const redis = new Redis({ host: '127.0.0.1', port: 6379, maxRetriesPerRequest: 3 });

class RedisLock {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  async acquire(key, ttlMs = 30000, retryCount = 3, retryDelay = 100) {
    const lockKey = `lock:${key}`;
    const token = crypto.randomUUID();

    for (let i = 0; i <= retryCount; i++) {
      const result = await this.redis.set(lockKey, token, 'NX', 'PX', ttlMs);

      if (result === 'OK') {
        console.log(`  [Lock] 获取锁成功: ${key}, token=${token.slice(0, 8)}...`);
        return token;
      }

      if (i < retryCount) {
        console.log(`  [Lock] 获取锁失败，${retryDelay}ms 后重试 (${i + 1}/${retryCount})`);
        await new Promise((r) => setTimeout(r, retryDelay));
      }
    }

    console.log(`  [Lock] 获取锁最终失败: ${key}`);
    return null;
  }

  async release(key, token) {
    const lockKey = `lock:${key}`;

    const luaScript = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("del", KEYS[1])
      else
        return 0
      end
    `;

    const result = await this.redis.eval(luaScript, 1, lockKey, token);

    if (result === 1) {
      console.log(`  [Lock] 释放锁成功: ${key}`);
      return true;
    } else {
      console.log(`  [Lock] 释放锁失败: ${key}`);
      return false;
    }
  }
}

async function basicLockDemo() {
  const lock = new RedisLock(redis);

  const token = await lock.acquire('order:12345', 10000);
  if (token) {
    try {
      console.log('  [业务] 正在处理订单 12345...');
      await new Promise((r) => setTimeout(r, 2000));
      console.log('  [业务] 订单 12345 处理完成');
    } finally {
      await lock.release('order:12345', token);
    }
  }

  // 模拟并发获取
  const [t1, t2] = await Promise.all([
    lock.acquire('resource:shared', 5000),
    lock.acquire('resource:shared', 5000),
  ]);
  console.log(`进程1: ${t1 ? '获取成功' : '获取失败'}`);
  console.log(`进程2: ${t2 ? '获取成功' : '获取失败'}`);

  if (t1) await lock.release('resource:shared', t1);
  if (t2) await lock.release('resource:shared', t2);
}
```

## 21.4 锁的续期（Watchdog 机制）

业务处理时间可能超过锁的 TTL。解决方案：Watchdog 在锁过期前自动续期。

```js
class WatchdogLock {
  constructor(redisClient) {
    this.redis = redisClient;
    this.watchdogs = new Map();
  }

  async acquire(key, ttlMs = 30000) {
    const lockKey = `lock:${key}`;
    const token = crypto.randomUUID();

    const result = await this.redis.set(lockKey, token, 'NX', 'PX', ttlMs);
    if (result !== 'OK') return null;

    console.log(`  [WatchdogLock] 获取锁成功: ${key}`);

    const renewalInterval = Math.floor(ttlMs / 3);
    const timer = setInterval(async () => {
      const renewed = await this._renew(lockKey, token, ttlMs);
      if (!renewed) {
        clearInterval(timer);
        this.watchdogs.delete(key);
      } else {
        console.log(`  [Watchdog] 续期成功: ${key}`);
      }
    }, renewalInterval);

    this.watchdogs.set(key, timer);
    return token;
  }

  async _renew(lockKey, token, ttlMs) {
    const luaScript = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("pexpire", KEYS[1], ARGV[2])
      else
        return 0
      end
    `;
    return (await this.redis.eval(luaScript, 1, lockKey, token, ttlMs)) === 1;
  }

  async release(key, token) {
    const timer = this.watchdogs.get(key);
    if (timer) {
      clearInterval(timer);
      this.watchdogs.delete(key);
    }

    const lockKey = `lock:${key}`;
    const luaScript = `
      if redis.call("get", KEYS[1]) == ARGV[1] then
        return redis.call("del", KEYS[1])
      else
        return 0
      end
    `;
    return (await this.redis.eval(luaScript, 1, lockKey, token)) === 1;
  }

  destroy() {
    for (const [, timer] of this.watchdogs) clearInterval(timer);
    this.watchdogs.clear();
  }
}

// 使用：长任务场景
async function watchdogDemo() {
  const lock = new WatchdogLock(redis);
  const token = await lock.acquire('long-task', 9000);

  if (token) {
    try {
      for (let i = 1; i <= 5; i++) {
        await new Promise((r) => setTimeout(r, 3000));
        console.log(`  [业务] 任务进行中... ${i * 3}秒`);
      }
      console.log('  [业务] 长任务完成！');
    } finally {
      await lock.release('long-task', token);
    }
  }
  lock.destroy();
}
```

## 21.5 可重入锁实现

可重入锁允许同一线程多次获取同一把锁。

```js
class ReentrantLock {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  async acquire(key, threadId, ttlMs = 30000) {
    const lockKey = `lock:${key}`;

    const luaScript = `
      local lockKey = KEYS[1]
      local threadId = ARGV[1]
      local ttlMs = tonumber(ARGV[2])

      if redis.call("exists", lockKey) == 0 then
        redis.call("hset", lockKey, threadId, 1)
        redis.call("pexpire", lockKey, ttlMs)
        return 1
      end

      if redis.call("hexists", lockKey, threadId) == 1 then
        redis.call("hincrby", lockKey, threadId, 1)
        redis.call("pexpire", lockKey, ttlMs)
        return 1
      end

      return 0
    `;

    return (await this.redis.eval(luaScript, 1, lockKey, threadId, ttlMs)) === 1;
  }

  async release(key, threadId) {
    const lockKey = `lock:${key}`;

    const luaScript = `
      local lockKey = KEYS[1]
      local threadId = ARGV[1]

      if redis.call("hexists", lockKey, threadId) == 0 then
        return -1
      end

      local count = redis.call("hincrby", lockKey, threadId, -1)

      if count > 0 then
        redis.call("pexpire", lockKey, redis.call("pttl", lockKey))
        return 0
      else
        redis.call("del", lockKey)
        return 1
      end
    `;

    return await this.redis.eval(luaScript, 1, lockKey, threadId);
    // 1 = 完全释放, 0 = 减少重入, -1 = 不属于当前线程
  }
}

async function reentrantDemo() {
  const lock = new ReentrantLock(redis);
  const threadId = `process-${process.pid}-${crypto.randomUUID().slice(0, 8)}`;

  console.log(`线程ID: ${threadId}`);
  const r1 = await lock.acquire('resource', threadId);
  console.log(`第1次获取: ${r1 ? '成功' : '失败'}`);
  const r2 = await lock.acquire('resource', threadId);
  console.log(`第2次获取（重入）: ${r2 ? '成功' : '失败'}`);

  console.log(`释放1: ${await lock.release('resource', threadId)}`);
  console.log(`释放2: ${await lock.release('resource', threadId)}`);
}
```

## 21.6 Redlock 算法

单节点锁的问题：Master 宕机时锁数据可能丢失，导致两个客户端同时持有同一把锁。

Redlock 使用 N 个独立 Redis 节点（通常 N=5），需要在 **多数节点（N/2+1）** 上成功获取锁才算成功。

```js
class Redlock {
  constructor(nodes, options = {}) {
    this.quorum = Math.floor(nodes.length / 2) + 1;
    this.retryCount = options.retryCount || 3;
    this.retryDelay = options.retryDelay || 200;
    this.clockDriftFactor = options.clockDriftFactor || 0.01;

    this.clients = nodes.map((addr) => {
      const [host, port] = addr.split(':');
      return new Redis({ host, port: parseInt(port) || 6379, connectTimeout: 5000, maxRetriesPerRequest: 1 });
    });
  }

  async acquire(key, ttlMs = 30000) {
    const lockKey = `redlock:${key}`;
    const token = crypto.randomUUID();

    for (let attempt = 0; attempt <= this.retryCount; attempt++) {
      const startTime = Date.now();

      const promises = this.clients.map(async (client) => {
        try {
          const result = await client.set(lockKey, token, 'NX', 'PX', ttlMs);
          return result === 'OK';
        } catch {
          return false;
        }
      });

      const results = await Promise.all(promises);
      const successCount = results.filter(Boolean).length;
      const elapsed = Date.now() - startTime;
      const drift = Math.floor(ttlMs * this.clockDriftFactor) + 2;
      const validTTL = ttlMs - elapsed - drift;

      if (successCount >= this.quorum && validTTL > 0) {
        console.log(`  [Redlock] 成功: ${successCount}/${this.clients.length}, 有效TTL=${validTTL}ms`);
        return { token, validTTL };
      }

      // 释放已获取的锁
      const luaScript = `if redis.call("get",KEYS[1])==ARGV[1] then return redis.call("del",KEYS[1]) else return 0 end`;
      await Promise.all(this.clients.map((c) => c.eval(luaScript, 1, lockKey, token).catch(() => 0)));

      if (attempt < this.retryCount) {
        const delay = this.retryDelay + Math.random() * this.retryDelay;
        await new Promise((r) => setTimeout(r, delay));
      }
    }

    return null;
  }

  async release(key, token) {
    const lockKey = `redlock:${key}`;
    const luaScript = `if redis.call("get",KEYS[1])==ARGV[1] then return redis.call("del",KEYS[1]) else return 0 end`;
    await Promise.all(this.clients.map((c) => c.eval(luaScript, 1, lockKey, token).catch(() => 0)));
  }
}
```

> **Warning：Redlock 的争议**
>
> Martin Kleppmann（《DDIA》作者）指出 Redlock 在时钟跳跃、GC 暂停等场景下可能不安全。金融级场景建议用 Zookeeper 或 etcd。

## 21.7 分布式锁的常见问题

**时钟跳跃**：Redis 服务器时钟跳跃导致锁提前/延后过期。防护：NTP slew 模式 + Redlock 多节点。

**GC 暂停**：长时间 GC 导致锁在业务未完成时过期。防护：Watchdog 续期 + 合理 TTL。

**网络分区**：客户端与 Redis 断连导致锁失控。防护：fencing token + 幂等操作。

## 21.8 Redis 分布式锁 vs Zookeeper vs 数据库锁

| 对比维度 | Redis | Zookeeper | 数据库 |
|----------|-------|-----------|--------|
| 性能 | 极高（~10万 QPS） | 中等 | 低 |
| 一致性 | 最终一致性 | 强一致性 | 强一致性 |
| 实现复杂度 | 低 | 高 | 极低 |
| 可靠性 | 中（依赖时钟） | 高（ZAB 协议） | 高（ACID） |
| 适用场景 | 高性能、允许小概率失败 | 高可靠、金融级 | 低并发 |

## 21.9 实战：用分布式锁实现库存扣减

```js
class InventoryService {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  async initStock(productId, quantity) {
    await this.redis.set(`stock:${productId}`, quantity);
  }

  // 方案一：分布式锁
  async deductStock(productId, quantity) {
    const token = crypto.randomUUID();
    const locked = await this.redis.set(`lock:stock:${productId}`, token, 'NX', 'PX', 5000);
    if (!locked) return { success: false, message: '系统繁忙' };

    try {
      const currentStock = parseInt(await this.redis.get(`stock:${productId}`));
      if (currentStock == null) return { success: false, message: '商品不存在' };
      if (currentStock < quantity) return { success: false, message: '库存不足' };

      const newStock = currentStock - quantity;
      await this.redis.set(`stock:${productId}`, newStock);
      return { success: true, newStock };
    } finally {
      const lua = `if redis.call("get",KEYS[1])==ARGV[1] then return redis.call("del",KEYS[1]) else return 0 end`;
      await this.redis.eval(lua, 1, `lock:stock:${productId}`, token);
    }
  }

  // 方案二：Lua 脚本原子操作（更优）
  async deductStockAtomic(productId, quantity) {
    const luaScript = `
      local stock = tonumber(redis.call("get", KEYS[1]))
      if stock == nil then return -1 end
      if stock < tonumber(ARGV[1]) then return -2 end
      return redis.call("decrby", KEYS[1], ARGV[1])
    `;
    const result = await this.redis.eval(luaScript, 1, `stock:${productId}`, quantity);
    if (result === -1) return { success: false, message: '商品不存在' };
    if (result === -2) return { success: false, message: '库存不足' };
    return { success: true, newStock: result };
  }
}

// 并发测试
async function inventoryTest() {
  const service = new InventoryService(redis);
  await service.initStock('SKU-001', 100);

  const requests = Array.from({ length: 50 }, () =>
    service.deductStockAtomic('SKU-001', 1)
  );

  const results = await Promise.all(requests);
  const success = results.filter((r) => r.success).length;
  const finalStock = await redis.get('stock:SKU-001');

  console.log(`成功: ${success}, 最终库存: ${finalStock}`);
  console.log(`准确性: ${finalStock === '50' ? '正确' : '有误'}`);
}
```

> **Tip：Lua 脚本 vs 分布式锁**
>
> 操作单个 key 时用 Lua 脚本更简洁高效。**能用 Lua 脚本解决的，就不要用分布式锁。**

## 小结

1. **基本实现**：`SET NX PX` + Lua 释放
2. **Watchdog**：自动续期防过期
3. **可重入**：Hash 结构记重入次数
4. **Redlock**：多节点多数派
5. **黄金法则**：能用 Lua 就别用锁，能用单节点就别用 Redlock
