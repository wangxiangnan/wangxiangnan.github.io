## 17.4 状态管理设计（Zustand Store）

我们先把状态管理层搭好，这是整个应用的"大脑"。

### 任务 Store

```jsx
// src/store/useTaskStore.js
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

// 列的定义（常量）
export const COLUMNS = [
  { id: 'todo', title: '待办', color: 'bg-gray-400' },
  { id: 'in-progress', title: '进行中', color: 'bg-blue-500' },
  { id: 'done', title: '已完成', color: 'bg-green-500' },
];

const useTaskStore = create(
  persist(
    (set, get) => ({
      // === 状态 ===
      tasks: [],
      searchQuery: '',
      selectedTag: 'all',

      // === 任务 CRUD ===

      // 创建任务
      addTask: (taskData) => {
        const newTask = {
          id: crypto.randomUUID(), // 生成唯一 ID
          title: taskData.title,
          description: taskData.description || '',
          status: taskData.status || 'todo',
          priority: taskData.priority || 'medium',
          tags: taskData.tags || [],
          createdAt: new Date().toISOString(),
          updatedAt: new Date().toISOString(),
        };
        set((state) => ({
          tasks: [...state.tasks, newTask],
        }));
      },

      // 更新任务
      updateTask: (id, updates) => {
        set((state) => ({
          tasks: state.tasks.map((task) =>
            task.id === id
              ? { ...task, ...updates, updatedAt: new Date().toISOString() }
              : task
          ),
        }));
      },

      // 删除任务
      deleteTask: (id) => {
        set((state) => ({
          tasks: state.tasks.filter((task) => task.id !== id),
        }));
      },

      // 移动任务到指定列（拖拽时调用）
      moveTask: (taskId, newStatus) => {
        set((state) => ({
          tasks: state.tasks.map((task) =>
            task.id === taskId
              ? { ...task, status: newStatus, updatedAt: new Date().toISOString() }
              : task
          ),
        }));
      },

      // === 搜索与筛选 ===

      setSearchQuery: (query) => set({ searchQuery: query }),
      setSelectedTag: (tag) => set({ selectedTag: tag }),

      // === 派生数据 ===

      // 获取某一列的任务（经过搜索和标签过滤）
      getTasksByStatus: (status) => {
        const { tasks, searchQuery, selectedTag } = get();
        let filtered = tasks.filter((task) => task.status === status);

        // 按标签筛选
        if (selectedTag !== 'all') {
          filtered = filtered.filter((task) => task.tags.includes(selectedTag));
        }

        // 按搜索词筛选
        if (searchQuery.trim()) {
          const query = searchQuery.toLowerCase().trim();
          filtered = filtered.filter(
            (task) =>
              task.title.toLowerCase().includes(query) ||
              task.description.toLowerCase().includes(query) ||
              task.tags.some((tag) => tag.toLowerCase().includes(query))
          );
        }

        return filtered;
      },

      // 获取所有标签（去重）
      getAllTags: () => {
        const { tasks } = get();
        const tagSet = new Set();
        tasks.forEach((task) => task.tags.forEach((tag) => tagSet.add(tag)));
        return ['all', ...Array.from(tagSet).sort()];
      },

      // 获取统计信息
      getStats: () => {
        const { tasks } = get();
        return {
          total: tasks.length,
          todo: tasks.filter((t) => t.status === 'todo').length,
          inProgress: tasks.filter((t) => t.status === 'in-progress').length,
          done: tasks.filter((t) => t.status === 'done').length,
        };
      },
    }),
    {
      name: 'taskboard-storage', // localStorage key
      partialize: (state) => ({
        // 只持久化任务数据，不持久化 UI 状态
        tasks: state.tasks,
      }),
    }
  )
);

export default useTaskStore;
```

### 主题 Store

```jsx
// src/store/useThemeStore.js
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

const useThemeStore = create(
  persist(
    (set) => ({
      theme: 'light',

      toggleTheme: () =>
        set((state) => ({
          theme: state.theme === 'light' ? 'dark' : 'light',
        })),

      // 在 <html> 元素上切换 dark class
      applyTheme: () => {
        const { theme } = useThemeStore.getState();
        if (theme === 'dark') {
          document.documentElement.classList.add('dark');
        } else {
          document.documentElement.classList.remove('dark');
        }
      },
    }),
    {
      name: 'taskboard-theme',
    }
  )
);

// 应用启动时立即应用主题
useThemeStore.getState().applyTheme();

export default useThemeStore;
```

```jsx
// src/App.jsx
import { useEffect } from 'react';
import useThemeStore from './store/useThemeStore';
import useTaskStore from './store/useTaskStore';
import initialTasks from './data/initialTasks';
import Layout from './components/Layout';
import Header from './components/Header';
import KanbanBoard from './components/KanbanBoard';

function App() {
  const applyTheme = useThemeStore((state) => state.applyTheme);
  const tasks = useTaskStore((state) => state.tasks);
  const addTask = useTaskStore((state) => state.addTask);

  // 主题变化时应用到 DOM
  useEffect(() => {
    applyTheme();
  }, [useThemeStore((state) => state.theme), applyTheme]);

  // 首次加载时，如果任务为空，填充初始数据
  useEffect(() => {
    if (tasks.length === 0) {
      initialTasks.forEach((task) => {
        addTask(task);
      });
    }
  }, []); // 只在挂载时执行一次

  return (
    <Layout>
      <Header />
      <KanbanBoard />
    </Layout>
  );
}

export default App;
```

```jsx
import useTaskStore from '../store/useTaskStore';

function TagFilter() {
  const selectedTag = useTaskStore((state) => state.selectedTag);
  const setSelectedTag = useTaskStore((state) => state.setSelectedTag);
  const allTags = useTaskStore((state) => state.getAllTags());

  if (allTags.length <= 1) return null; // 没有标签时不显示

  return (
    <div className="flex items-center gap-2 flex-wrap">
      <span className="text-sm text-gray-500 dark:text-gray-400 whitespace-nowrap">标签:</span>
      {allTags.map((tag) => (

          key={tag}
          onClick={() => setSelectedTag(tag)}
          className={`px-3 py-1 rounded-full text-sm transition-colors ${
            selectedTag === tag
              ? 'bg-blue-500 text-white'
              : 'bg-gray-200 dark:bg-gray-700 text-gray-600 dark:text-gray-300 hover:bg-gray-300 dark:hover:bg-gray-600'
          }`}

          {tag === 'all' ? '全部' : tag}
        </button>

}
```

这是我们的核心组件！它负责渲染三列，并管理"新增任务"按钮。

```jsx
// src/components/KanbanBoard.jsx
import { useState } from 'react';
import useTaskStore, { COLUMNS } from '../store/useTaskStore';
import Column from './Column';
import TaskModal from './TaskModal';

function KanbanBoard() {
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [modalMode, setModalMode] = useState('add'); // 'add' | 'edit'
  const [editingTask, setEditingTask] = useState(null);
  const [defaultStatus, setDefaultStatus] = useState('todo');

  // 打开新增对话框
  const openAddModal = (status = 'todo') => {
    setModalMode('add');
    setEditingTask(null);
    setDefaultStatus(status);
    setIsModalOpen(true);
  };

  // 打开编辑对话框
  const openEditModal = (task) => {
    setModalMode('edit');
    setEditingTask(task);
    setIsModalOpen(true);
  };

  return (
    <div className="flex-1 p-6">
      {/* 看板列 */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6 h-full">
        {COLUMNS.map((column) => (
          <Column
            key={column.id}
            column={column}
            onAddTask={() => openAddModal(column.id)}
            onEditTask={openEditModal}
          />
        ))}
      </div>

      {/* 新增/编辑任务对话框 */}
      {isModalOpen && (
        <TaskModal
          mode={modalMode}
          task={editingTask}
          defaultStatus={defaultStatus}
          onClose={() => setIsModalOpen(false)}
        />
      )}
    </div>
  );
}

export default KanbanBoard;
```

### Column — 单个列

```jsx
// src/components/Column.jsx
import { useState } from 'react';
import useTaskStore from '../store/useTaskStore';
import TaskCard from './TaskCard';

function Column({ column, onAddTask, onEditTask }) {
  const [isDragOver, setIsDragOver] = useState(false);
  const getTasksByStatus = useTaskStore((state) => state.getTasksByStatus);
  const moveTask = useTaskStore((state) => state.moveTask);

  // 获取当前列的任务
  const tasks = getTasksByStatus(column.id);

  // 拖拽进入列区域
  const handleDragOver = (e) => {
    e.preventDefault(); // 必须阻止默认行为才能 drop
    setIsDragOver(true);
  };

  // 拖拽离开列区域
  const handleDragLeave = () => {
    setIsDragOver(false);
  };

  // 放置任务
  const handleDrop = (e) => {
    e.preventDefault();
    setIsDragOver(false);

    const taskId = e.dataTransfer.getData('taskId');
    if (taskId) {
      moveTask(taskId, column.id);
    }
  };

  return (
    <div
      className={`flex flex-col bg-gray-100 dark:bg-gray-800 rounded-xl p-4 min-h-[400px] transition-colors ${
        isDragOver ? 'ring-2 ring-blue-400 bg-blue-50 dark:bg-blue-900/20' : ''
      }`}
      onDragOver={handleDragOver}
      onDragLeave={handleDragLeave}
      onDrop={handleDrop}
    >
      {/* 列头 */}
      <div className="flex items-center justify-between mb-4">
        <div className="flex items-center gap-2">
          <div className={`w-3 h-3 rounded-full ${column.color}`}></div>
          <h2 className="font-semibold text-gray-800 dark:text-gray-200">
            {column.title}
          </h2>
          <span className="bg-gray-300 dark:bg-gray-600 text-gray-600 dark:text-gray-300 rounded-full px-2 py-0.5 text-xs">
            {tasks.length}
          </span>
        </div>

        {/* 新增按钮 */}
        <button
          onClick={onAddTask}
          className="p-1 rounded hover:bg-gray-200 dark:hover:bg-gray-700 text-gray-500 dark:text-gray-400"
          title="新增任务"
        >
          <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4" />
          </svg>
        </button>
      </div>

      {/* 任务卡片列表 */}
      <div className="flex-1 space-y-3 overflow-y-auto">
        {tasks.length === 0 ? (
          <div className="flex items-center justify-center h-32 border-2 border-dashed border-gray-300 dark:border-gray-600 rounded-lg">
            <p className="text-gray-400 dark:text-gray-500 text-sm">
              拖拽任务到这里
            </p>
          </div>
        ) : (
          tasks.map((task) => (
            <TaskCard key={task.id} task={task} onEdit={onEditTask} />
          ))
        )}
      </div>
    </div>
  );
}
```

### TaskModal — 新增/编辑任务对话框

这是整个应用中最复杂的组件 — 一个表单对话框，支持新增和编辑两种模式。

```jsx
// src/components/TaskModal.jsx
import { useState, useEffect, useRef } from 'react';
import useTaskStore, { COLUMNS } from '../store/useTaskStore';

function TaskModal({ mode, task, defaultStatus, onClose }) {
  const addTask = useTaskStore((state) => state.addTask);
  const updateTask = useTaskStore((state) => state.updateTask);

  // 表单状态
  const [formData, setFormData] = useState({
    title: '',
    description: '',
    status: defaultStatus,
    priority: 'medium',
    tags: [],
  });

  // 标签输入
  const [tagInput, setTagInput] = useState('');
  const modalRef = useRef(null);
  const titleInputRef = useRef(null);

  // 编辑模式时，填充表单数据
  useEffect(() => {
    if (mode === 'edit' && task) {
      setFormData({
        title: task.title,
        description: task.description,
        status: task.status,
        priority: task.priority,
        tags: [...task.tags],
      });
    }
  }, [mode, task]);

  // 打开时自动聚焦标题输入框
  useEffect(() => {
    titleInputRef.current?.focus();
  }, []);

  // 点击遮罩层关闭
  const handleBackdropClick = (e) => {
    if (e.target === e.currentTarget) {
      onClose();
    }
  };

  // ESC 键关闭
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'Escape') onClose();
    };
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [onClose]);

  // 提交表单
  const handleSubmit = (e) => {
    e.preventDefault();

    if (!formData.title.trim()) {
      titleInputRef.current?.focus();
      return;
    }

    if (mode === 'add') {
      addTask(formData);
    } else {
      updateTask(task.id, formData);
    }

    onClose();
  };

  // 添加标签
  const addTag = () => {
    const tag = tagInput.trim();
    if (tag && !formData.tags.includes(tag)) {
      setFormData((prev) => ({
        ...prev,
        tags: [...prev.tags, tag],
      }));
    }
    setTagInput('');
  };

  // 删除标签
  const removeTag = (tagToRemove) => {
    setFormData((prev) => ({
      ...prev,
      tags: prev.tags.filter((t) => t !== tagToRemove),
    }));
  };

  // 标签输入按键处理
  const handleTagKeyDown = (e) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      addTag();
    }
  };

  return (
    <div
      className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4"
      onClick={handleBackdropClick}
    >
      <div
        ref={modalRef}
        className="bg-white dark:bg-gray-800 rounded-xl shadow-xl w-full max-w-md max-h-[90vh] overflow-y-auto"
      >
        {/* 头部 */}
        <div className="flex items-center justify-between p-6 pb-0">
          <h2 className="text-lg font-bold text-gray-900 dark:text-white">
            {mode === 'add' ? '新增任务' : '编辑任务'}
          </h2>
          <button
            onClick={onClose}
            className="p-1 rounded hover:bg-gray-100 dark:hover:bg-gray-700 text-gray-400"
          >
            <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        </div>

        {/* 表单 */}
        <form onSubmit={handleSubmit} className="p-6 space-y-4">
          {/* 标题 */}
          <div>
            <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
              标题 <span className="text-red-500">*</span>
            </label>
            <input
              ref={titleInputRef}
              type="text"
              value={formData.title}
              onChange={(e) => setFormData((prev) => ({ ...prev, title: e.target.value }))}
              placeholder="输入任务标题..."
              className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-white placeholder-gray-400 focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
          </div>

          {/* 描述 */}
          <div>
            <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
              描述
            </label>
            <textarea
              value={formData.description}
              onChange={(e) => setFormData((prev) => ({ ...prev, description: e.target.value }))}
              placeholder="描述这个任务..."
              rows={3}
              className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-white placeholder-gray-400 focus:outline-none focus:ring-2 focus:ring-blue-500 resize-none"
            />
          </div>

          {/* 状态和优先级 */}
          <div className="grid grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
                状态
              </label>
              <select
                value={formData.status}
                onChange={(e) => setFormData((prev) => ({ ...prev, status: e.target.value }))}
                className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-white focus:outline-none focus:ring-2 focus:ring-blue-500"
              >
                {COLUMNS.map((col) => (
                  <option key={col.id} value={col.id}>
                    {col.title}
                  </option>
                ))}
              </select>
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
                优先级
              </label>
              <select
                value={formData.priority}
                onChange={(e) => setFormData((prev) => ({ ...prev, priority: e.target.value }))}
                className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-white focus:outline-none focus:ring-2 focus:ring-blue-500"
              >
                <option value="low">低</option>
                <option value="medium">中</option>
                <option value="high">高</option>
              </select>
            </div>
          </div>

          {/* 标签 */}
          <div>
            <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
              标签
            </label>
            <div className="flex gap-2">
              <input
                type="text"
                value={tagInput}
                onChange={(e) => setTagInput(e.target.value)}
                onKeyDown={handleTagKeyDown}
                placeholder="输入标签后按 Enter"
                className="flex-1 px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-white placeholder-gray-400 focus:outline-none focus:ring-2 focus:ring-blue-500"
              />
              <button
                type="button"
                onClick={addTag}
                className="px-3 py-2 bg-gray-200 dark:bg-gray-600 rounded-lg hover:bg-gray-300 dark:hover:bg-gray-500 text-gray-700 dark:text-gray-200"
              >
                添加
              </button>
            </div>
            {formData.tags.length > 0 && (
              <div className="flex flex-wrap gap-1 mt-2">
                {formData.tags.map((tag) => (
                  <span
                    key={tag}
                    className="inline-flex items-center gap-1 bg-blue-50 dark:bg-blue-900/40 text-blue-600 dark:text-blue-300 px-2 py-0.5 rounded text-xs"
                  >
                    {tag}
                    <button
                      type="button"
                      onClick={() => removeTag(tag)}
                      className="hover:text-red-500"
                    >
                      ✕
                    </button>
                  </span>
                ))}
              </div>
            )}
          </div>

          {/* 操作按钮 */}
          <div className="flex gap-3 pt-4">
            <button
              type="button"
              onClick={onClose}
              className="flex-1 px-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700"
            >
              取消
            </button>
            <button
              type="submit"
              className="flex-1 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2"
            >
              {mode === 'add' ? '创建任务' : '保存修改'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

export default TaskModal;
```

## 17.8 拖拽功能的实现细节

我们在上面使用了 HTML5 原生的 Drag and Drop API。让我详细解释一下它是如何工作的：

### 拖拽流程图

```
用户鼠标按下卡片
    ↓
TaskCard: onDragStart
  → 设置 isDragging = true（视觉反馈）
  → e.dataTransfer.setData('taskId', task.id)（携带数据）
    ↓
用户拖拽到目标列
    ↓
Column: onDragOver
  → e.preventDefault()（必须！否则不能 drop）
  → setIsDragOver(true)（高亮目标列）
    ↓
用户松手
    ↓
Column: onDrop
  → e.dataTransfer.getData('taskId')（取出任务 ID）
  → moveTask(taskId, column.id)（更新状态）
    ↓
Column: onDragEnd（在 TaskCard 上）
  → setIsDragging(false)（恢复视觉）
```

> ⚠️ **Warning**：`onDragOver` 中的 `e.preventDefault()` 是必须的！没有它，浏览器不允许在该元素上 drop。这是一个非常容易踩的坑。

> 💡 **Tip**：HTML5 Drag and Drop API 在移动端的支持不太好。如果你需要支持移动端拖拽，可以考虑使用 `@dnd-kit/core` 或 `react-beautiful-dnd` 库。但对于桌面应用，原生 API 完全够用。

## 17.9 暗色模式的实现

暗色模式的实现依赖于三个部分的配合：

### 1. Tailwind 配置

在 `tailwind.config.js` 中设置 `darkMode: 'class'`，这样 Tailwind 会根据 `<html>` 元素上是否有 `dark` class 来决定是否应用暗色样式。

### 2. Zustand Theme Store

`useThemeStore` 管理主题状态，并通过 `applyTheme` 方法在 `<html>` 上添加/移除 `dark` class。

### 3. 组件中使用 `dark:` 前缀

```jsx
// 亮色模式：bg-white text-gray-900
// 暗色模式：bg-gray-800 text-white
<div className="bg-white dark:bg-gray-800 text-gray-900 dark:text-white">
  内容
</div>
```

> 💡 **Tip**：Tailwind 的暗色模式非常优雅。你不需要写任何媒体查询或 JavaScript 条件判断，只需要在每个需要暗色适配的样式前加 `dark:` 前缀就行了。

## 17.10 数据持久化

数据持久化通过 Zustand 的 `persist` 中间件实现。它会自动：

1. 每次状态变化时，将指定的状态序列化后保存到 `localStorage`
2. 页面加载时，从 `localStorage` 读取数据并恢复到 store 中

```jsx
// 关键配置
persist(
  (set, get) => ({
    // ... store 定义
  }),
  {
    name: 'taskboard-storage', // localStorage 的 key
    partialize: (state) => ({
      // 只持久化 tasks，不持久化 searchQuery 等临时状态
      tasks: state.tasks,
    }),
  }
)
```

这样，用户刷新页面后，所有任务数据都会保留。但搜索框的内容、选中的标签等 UI 状态会被重置 — 这正是我们想要的！

## 17.12 代码回顾与最佳实践

让我们回顾一下在这个项目中运用的最佳实践：

### 1. 组件设计原则

- **单一职责**：每个组件只做一件事（`SearchBar` 只管搜索，`ThemeToggle` 只管主题）
- **组合优于继承**：通过组件嵌套构建复杂 UI（`Layout` > `Header` + `KanbanBoard` > `Column` > `TaskCard`）
- **Props 向下，事件向上**：子组件通过回调函数通知父组件

### 2. 状态管理原则

- **最小化状态**：能算出来的就不存（如 `getTasksByStatus`、`getStats`）
- **状态下放**：只在需要的地方管理状态（如 `TaskModal` 的表单状态）
- **全局状态用 Zustand**：任务数据、主题设置是全局共享的

### 3. 性能考量

- **选择器精确订阅**：组件只订阅它需要的状态，避免不必要的重渲染
- **memo 优化**：`TaskCard` 可以用 `React.memo` 包裹
- **事件委托**：列级别的 drag 事件处理，而非每个卡片独立处理

### 4. 用户体验

- **键盘支持**：ESC 关闭对话框，Enter 添加标签
- **视觉反馈**：拖拽时的透明度变化、目标列的高亮
- **确认操作**：删除任务前的 `confirm` 对话框
- **自动聚焦**：打开对话框后自动聚焦到标题输入框
