# 第8章 LCEL —— LangChain的声明式编排核心

## 8.1 开篇：LCEL是什么？

LCEL（LangChain Expression Language）是 LangChain 的"管道式"编排语言。如果你对 Unix 管道不陌生，你会秒懂：

```bash
# Unix管道
cat file.txt | grep "error" | sort | uniq -c

# LCEL
prompt | llm | output_parser
```

核心理念：**每个 LangChain 组件都是 Runnable**，统一接口，管道串联。

为什么要学 LCEL？它解决了几个痛点：代码冗长、流式输出困难、错误处理繁琐、异步支持不一致。

## 8.2 核心概念：Runnable接口

```python
class Runnable:
    """所有LangChain组件的统一接口"""
    def invoke(self, input): ...     # 同步调用
    def batch(self, inputs): ...     # 批量调用
    def stream(self, input): ...     # 流式调用
    async def ainvoke(self, input): ...   # 异步
    async def astream(self, input): ...   # 异步流式
```

精妙之处：组合后的链仍是 Runnable。像乐高——不管怎么拼，最终还是一个整体。

## 8.3 管道操作符

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatOpenAI(model="gpt-4o")
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是技术文档助手。"),
    ("human", "{question}")
])
parser = StrOutputParser()

chain = prompt | llm | parser

# 同步调用
result = chain.invoke({"question": "什么是LCEL？"})

# 流式调用
for chunk in chain.stream({"question": "解释Python的GIL"}):
    print(chunk, end="", flush=True)
```

数据流动：

```
{"question": "..."} → prompt → ChatPromptValue → llm → AIMessage → parser → "纯文本"
```

## 8.4 关键Runnable组件

### 8.4.1 RunnablePassthrough：透传数据

```python
from langchain_core.runnables import RunnablePassthrough

chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | prompt | llm | parser
)
# question原样传递，context由retriever获取
```

### 8.4.2 RunnableLambda：自定义转换

```python
from langchain_core.runnables import RunnableLambda


def format_docs(docs: list) -> str:
    """格式化检索文档"""
    return "\n\n".join([
        f"[文档{i+1}] {d.metadata.get('source','?')}\n{d.page_content}"
        for i, d in enumerate(docs)
    ])


chain = (
    {"context": retriever | RunnableLambda(format_docs), 
     "question": RunnablePassthrough()}
    | prompt | llm | parser
)
```

### 8.4.3 RunnableParallel：并行执行

```python
from langchain_core.runnables import RunnableParallel, RunnablePassthrough

parallel = RunnableParallel(
    vector_search=retriever,
    keyword_search=keyword_retriever,
    original_query=RunnablePassthrough()
)

chain = parallel | merge_results | prompt | llm | parser
```

### 8.4.4 RunnableBranch：条件分支

```python
from langchain_core.runnables import RunnableBranch

branch = RunnableBranch(
    (lambda x: x.get("type") == "technical", technical_chain),
    (lambda x: x.get("type") == "general", general_chain),
    fallback_chain
)
full_chain = classify | branch | format_output
```

### 8.4.5 RunnableWithFallbacks：错误回退

```python
primary = prompt | ChatOpenAI(model="gpt-4o") | parser
fallback = prompt | ChatOpenAI(model="gpt-4o-mini") | parser
robust = primary.with_fallbacks([fallback])
# gpt-4o失败时自动切到gpt-4o-mini
```

## 8.5 流式传播

LCEL最强大的特性——LLM每个token实时流经整个链：

```python
llm = ChatOpenAI(model="gpt-4o", streaming=True)
chain = prompt | llm | parser

print("回答：", end="", flush=True)
for chunk in chain.stream({"question": "详细解释Python装饰器"}):
    print(chunk, end="", flush=True)
print()
```

用户在LLM还在生成时就能看到前几段，体验好太多。

## 8.6 异步支持

```python
import asyncio

async def run_async():
    chain = prompt | llm | parser
    
    # 异步调用
    result = await chain.ainvoke({"question": "什么是LCEL？"})
    
    # 异步流式
    async for chunk in chain.astream({"question": "Python异步编程"}):
        print(chunk, end="", flush=True)
    
    # 批量异步（并发）
    questions = [{"question": "什么是Agent？"}, {"question": "什么是RAG？"}]
    results = await chain.abatch(questions)

asyncio.run(run_async())
```

## 8.7.1 高级LCEL模式

### 复杂数据流控制

```python
from langchain_core.runnables import (
    RunnablePassthrough, RunnableLambda, RunnableParallel, RunnableBranch
)


def build_smart_rag_chain(retriever, llm):
    """智能RAG链：根据查询复杂度选择不同策略"""
    
    def preprocess(query: str) -> dict:
        """预处理：分析查询特征"""
        words = query.split()
        complexity = min(1.0, len(words) / 15)  # 越长越复杂
        has_code = any(kw in query for kw in ["代码", "函数", "class", "def", "import"])
        return {"query": query, "complexity": complexity, "has_code": has_code}
    
    def smart_retrieve(data: dict) -> dict:
        """智能检索：简单查询少检索，复杂查询多检索"""
        query = data["query"]
        k = 5 if data["complexity"] > 0.5 else 3
        docs = retriever.invoke(query)[:k]
        context = "\n\n".join([d.page_content for d in docs])
        return {**data, "context": context, "docs_count": len(docs)}
    
    answer_prompt = ChatPromptTemplate.from_messages([
        ("system", "基于上下文回答问题。\n上下文：{context}"),
        ("human", "{query}")
    ])
    
    chain = (
        RunnableLambda(preprocess)
        | RunnableLambda(smart_retrieve)
        | RunnableLambda(lambda d: {"query": d["query"], "context": d["context"]})
        | answer_prompt
        | llm
        | StrOutputParser()
    )
    return chain
```

### LCEL与LangGraph结合

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict


class QualityState(TypedDict):
    query: str
    context: str
    answer: str
    quality: float


def build_quality_loop(retriever, llm):
    """带质量检查的循环管线"""
    
    retrieve_chain = retriever | RunnableLambda(lambda docs: "\n".join(d.page_content for d in docs))
    answer_chain = answer_prompt | llm | StrOutputParser()
    
    def retrieve_node(state):
        ctx = retrieve_chain.invoke(state["query"])
        return {"context": ctx}
    
    def answer_node(state):
        ans = answer_chain.invoke({"query": state["query"], "context": state["context"]})
        return {"answer": ans}
    
    def quality_node(state):
        score = llm.invoke(f"评分(0-10)：Q:{state['query']}\nA:{state['answer']}").content
        try:
            return {"quality": int(score.strip()) / 10}
        except:
            return {"quality": 0.5}
    
    def route(state):
        return "done" if state["quality"] >= 0.7 else "retry"
    
    def retry_node(state):
        # 用增强策略重新检索
        new_ctx = retriever.invoke(state["query"] + " 详细解释")
        new_ctx = "\n".join(d.page_content for d in new_ctx[:7])
        return {"context": new_ctx}
    
    builder = StateGraph(QualityState)
    builder.add_node("retrieve", retrieve_node)
    builder.add_node("answer", answer_node)
    builder.add_node("quality", quality_node)
    builder.add_node("retry", retry_node)
    builder.set_entry_point("retrieve")
    builder.add_edge("retrieve", "answer")
    builder.add_edge("answer", "quality")
    builder.add_conditional_edges("quality", route, {"done": END, "retry": "retry"})
    builder.add_edge("retry", "answer")
    return builder.compile()
```

### 流式处理完整示例

```python
import asyncio
import time


async def stream_with_metrics(chain, input_data):
    """带性能指标的流式输出"""
    start = time.time()
    tokens = 0
    chunks = 0
    
    async for chunk in chain.astream(input_data):
        tokens += len(chunk.split())
        chunks += 1
        elapsed = time.time() - start
        
        yield {
            "content": chunk,
            "tokens": tokens,
            "chunks": chunks,
            "elapsed_ms": round(elapsed * 1000, 1),
            "tokens_per_sec": round(tokens / max(elapsed, 0.01), 1)
        }


async def demo():
    from langchain_openai import ChatOpenAI
    chain = prompt | ChatOpenAI(model="gpt-4o", streaming=True) | StrOutputParser()
    
    last = None
    async for item in stream_with_metrics(chain, {"question": "解释量子计算"}):
        print(item["content"], end="", flush=True)
        last = item
    
    print(f"\n\nTokens: {last['tokens']}, 耗时: {last['elapsed_ms']}ms")
    print(f"速度: {last['tokens_per_sec']} tokens/sec")
```

## 8.8 实操：用LCEL重构DocMind

```python
"""DocMind RAG管线 — LCEL版本"""

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import (
    RunnablePassthrough, RunnableLambda, RunnableParallel
)


class DocMindLCEL:
    def __init__(self, vectorstore: Chroma):
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0, streaming=True)
        self.fast_llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
        self.retriever = vectorstore.as_retriever(search_kwargs={"k": 5})
        self._build_chains()
    
    def _build_chains(self):
        self.format_docs = RunnableLambda(self._format_docs)
        self.answer_prompt = ChatPromptTemplate.from_messages([
            ("system", "你是DocMind。基于上下文回答问题。不足则说明。\n上下文：{context}"),
            ("human", "{question}")
        ])
        
        # 主链
        self.main_chain = (
            RunnableParallel(
                context=(self.retriever | self.format_docs),
                question=RunnablePassthrough()
            ) | self.answer_prompt | self.llm | StrOutputParser()
        )
        
        # 回退链
        fallback = (
            RunnableParallel(
                context=(self.retriever | self.format_docs),
                question=RunnablePassthrough()
            ) | self.answer_prompt | self.fast_llm | StrOutputParser()
        )
        self.robust_chain = self.main_chain.with_retry(
            stop_after_attempt=3
        ).with_fallbacks([fallback])
    
    def _format_docs(self, docs: list) -> str:
        if not docs:
            return "（无相关文档）"
        return "\n---\n".join([
            f"[{i+1}] {d.metadata.get('source','?')}\n{d.page_content}"
            for i, d in enumerate(docs)
        ])
    
    def ask(self, question: str) -> str:
        return self.robust_chain.invoke(question)
    
    def ask_stream(self, question: str):
        for chunk in self.main_chain.stream(question):
            yield chunk
    
    async def ask_async(self, question: str) -> str:
        return await self.robust_chain.ainvoke(question)
    
    def ask_batch(self, questions: list) -> list:
        return self.robust_chain.batch(questions)
    
    def ask_with_sources(self, question: str) -> dict:
        chain = RunnableParallel(answer=self.robust_chain, docs=self.retriever)
        result = chain.invoke(question)
        return {
            "answer": result["answer"],
            "sources": [{"source": d.metadata.get("source",""), "preview": d.page_content[:100]}
                        for d in result["docs"]]
        }
```

## 8.9 本章小结

| 组件 | 作用 | Unix类比 |
| ------ | ------ | --------- |
| 管道 \| | 串联Runnable | \| |
| RunnablePassthrough | 透传 | cat |
| RunnableLambda | 自定义转换 | awk/sed |
| RunnableParallel | 并行 | & |
| RunnableBranch | 条件分支 | if-else |
| .with_retry() | 重试 | while循环 |
| .with_fallbacks() | 回退 | \|\| |
| .stream() | 流式 | 实时管道 |
