# 第12章：RedisJSON——文档型存储

## 引言：Redis 里的 MongoDB

想用 Redis Hash 存嵌套对象？你得把对象序列化成 JSON 字符串，每次修改都要读出整个对象再写回去。这就像为了换一双袜子，得先把整个衣柜搬出来。

**RedisJSON** 让你在 Redis 中直接存储和操作 JSON 文档，支持 JSONPath 语法精确定位嵌套数据。

## 12.2 基础操作

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function jsonBasics() {
  await redis.call('JSON.SET', 'user:1', '$', JSON.stringify({
    name: 'Alice', age: 28, email: 'alice@example.com',
    address: { city: 'Beijing', district: 'Chaoyang', zipCode: '100020' },
    hobbies: ['reading', 'coding', 'traveling'],
    preferences: {
      theme: 'dark', language: 'zh-CN',
      notifications: { email: true, sms: false, push: true },
    },
  }));
  console.log('JSON 文档已存储\n');

  const fullDoc = await redis.call('JSON.GET', 'user:1');
  console.log('完整文档:', JSON.parse(fullDoc));

  const nameAndAge = await redis.call('JSON.GET', 'user:1', '$.name', '$.age');
  console.log('\nname 和 age:', JSON.parse(nameAndAge));

  const city = await redis.call('JSON.GET', 'user:1', '$.address.city');
  console.log('\n城市:', JSON.parse(city));

  const firstHobby = await redis.call('JSON.GET', 'user:1', '$.hobbies[0]');
  console.log('\n第一个爱好:', JSON.parse(firstHobby));

  const types = await Promise.all([
    redis.call('JSON.TYPE', 'user:1', '$'),
    redis.call('JSON.TYPE', 'user:1', '$.name'),
    redis.call('JSON.TYPE', 'user:1', '$.age'),
    redis.call('JSON.TYPE', 'user:1', '$.hobbies'),
  ]);
  console.log('\n类型:', types);

  await redis.call('JSON.DEL', 'user:1', '$.preferences.theme');
  console.log('\n已删除 preferences.theme');

  const nameLen = await redis.call('JSON.STRLEN', 'user:1', '$.name');
  console.log('\nname 长度:', nameLen);

  const hobbiesLen = await redis.call('JSON.ARRLEN', 'user:1', '$.hobbies');
  console.log('hobbies 长度:', hobbiesLen);

  await redis.quit();
}

jsonBasics().catch(console.error);
```

## 12.3 JSONPath 语法

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function jsonPathDemo() {
  await redis.call('JSON.SET', 'team:1', '$', JSON.stringify({
    name: 'Engineering Team',
    members: [
      { name: 'Alice', role: 'lead', skills: ['Node.js', 'Python', 'Go'], salary: 45000 },
      { name: 'Bob', role: 'senior', skills: ['Java', 'Kotlin'], salary: 38000 },
      { name: 'Charlie', role: 'junior', skills: ['JavaScript', 'React'], salary: 20000 },
      { name: 'Diana', role: 'senior', skills: ['Python', 'ML', 'Go'], salary: 42000 },
    ],
    projects: [
      { name: 'Alpha', status: 'active', priority: 1 },
      { name: 'Beta', status: 'completed', priority: 2 },
    ],
  }));

  console.log('$ (根):', typeof JSON.parse(await redis.call('JSON.GET', 'team:1', '$')));

  const members = JSON.parse(await redis.call('JSON.GET', 'team:1', '$.members'));
  console.log('\n$.members:', members.length, '个');

  const names = JSON.parse(await redis.call('JSON.GET', 'team:1', '$.members[*].name'));
  console.log('\n$.members[*].name:', names);

  const first = JSON.parse(await redis.call('JSON.GET', 'team:1', '$.members[0]'));
  console.log('\n$.members[0]:', first);

  const last = JSON.parse(await redis.call('JSON.GET', 'team:1', '$.members[-1]'));
  console.log('\n$.members[-1]:', last);

  const allSkills = JSON.parse(await redis.call('JSON.GET', 'team:1', '$.members[*].skills[*]'));
  console.log('\n所有技能(扁平):', allSkills);

  const allNames = JSON.parse(await redis.call('JSON.GET', 'team:1', '$..name'));
  console.log('\n$..name (递归):', allNames);

  await redis.quit();
}

jsonPathDemo().catch(console.error);
```

## 12.4 高级操作

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function advancedOps() {
  await redis.call('JSON.SET', 'user:42', '$', JSON.stringify({
    name: 'Alice', age: 28, balance: 1000,
    scores: [85, 92, 78], settings: { theme: 'dark' },
  }));

  console.log('余额+500:', await redis.call('JSON.NUMINCRBY', 'user:42', '$.balance', 500));
  console.log('年龄+1:', await redis.call('JSON.NUMINCRBY', 'user:42', '$.age', 1));
  console.log('余额x1.1:', await redis.call('JSON.NUMMULTBY', 'user:42', '$.balance', 1.1));

  await redis.call('JSON.SET', 'user:42', '$.name', '"Alice"');
  console.log('\n名字追加:', await redis.call('JSON.STRAPPEND', 'user:42', '$.name', '" Smith"'));

  console.log('\n分数追加:', await redis.call('JSON.ARRAPPEND', 'user:42', '$.scores', '95', '88'));
  console.log('开头插入100:', await redis.call('JSON.ARRINSERT', 'user:42', '$.scores', 0, '100'));
  console.log('弹出最后:', await redis.call('JSON.ARRPOP', 'user:42', '$.scores', -1));
  console.log('85的位置:', await redis.call('JSON.ARRINDEX', 'user:42', '$.scores', '85'));
  console.log('裁剪到前3:', await redis.call('JSON.ARRTRIM', 'user:42', '$.scores', 0, 2));
  console.log('\n根对象的键:', await redis.call('JSON.OBJKEYS', 'user:42', '$'));

  console.log('\n最终:', JSON.parse(await redis.call('JSON.GET', 'user:42')));
  await redis.quit();
}

advancedOps().catch(console.error);
```

## 12.5 与 RediSearch 结合

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function jsonSearchDemo() {
  const users = [
    { name: 'Alice', age: 28, city: 'Beijing', skills: ['Node.js', 'Python'] },
    { name: 'Bob', age: 35, city: 'Shanghai', skills: ['Java', 'Go'] },
    { name: 'Charlie', age: 22, city: 'Beijing', skills: ['React', 'TypeScript'] },
    { name: 'Diana', age: 30, city: 'Shenzhen', skills: ['Python', 'ML'] },
  ];

  for (const u of users) {
    await redis.call('JSON.SET', `juser:${u.name.toLowerCase()}`, '$', JSON.stringify(u));
  }

  try {
    await redis.call(
      'FT.CREATE', 'idx:json_users', 'ON', 'JSON', 'PREFIX', '1', 'juser:',
      'SCHEMA',
        '$.name', 'AS', 'name', 'TEXT',
        '$.age', 'AS', 'age', 'NUMERIC', 'SORTABLE',
        '$.city', 'AS', 'city', 'TAG',
        '$.skills[*]', 'AS', 'skills', 'TAG'
    );
  } catch (err) {
    if (!err.message.includes('already exists')) throw err;
  }

  await new Promise((r) => setTimeout(r, 500));

  const result = await redis.call('FT.SEARCH', 'idx:json_users', '@city:{Beijing}', 'SORTBY', 'age', 'ASC');
  console.log('北京的用户:');
  for (let i = 1; i < result.length; i += 2) {
    const doc = JSON.parse(result[i + 1][1]);
    console.log(`  ${doc.name}, ${doc.age}岁`);
  }

  await redis.quit();
}

jsonSearchDemo().catch(console.error);
```

## 12.6 RedisJSON vs Hash

| 特性 | Hash | RedisJSON |
|------|------|-----------|
| 数据结构 | 扁平键值对 | 嵌套 JSON 文档 |
| 嵌套支持 | 否 | 是 |
| 部分读取 | HGET/HMGET | JSON.GET with path |
| 部分更新 | HSET | JSON.SET with path |
| 原子操作 | HINCRBY | JSON.NUMINCRBY |
| 搜索索引 | ON HASH | ON JSON |
| 适用场景 | 简单对象 | 嵌套文档 |

## 12.7 实战：用户配置存储系统

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

class UserConfigStore {
  constructor(redisClient) { this.redis = redisClient; }

  async createConfig(userId, config = {}) {
    const defaults = {
      userId,
      profile: { displayName: '', avatar: '', bio: '' },
      preferences: { theme: 'light', language: 'zh-CN', timezone: 'Asia/Shanghai' },
      notifications: { email: true, push: true, sms: false, channels: ['system', 'marketing'] },
      privacy: { profileVisible: true, showEmail: false, showActivity: true },
      createdAt: Date.now(), updatedAt: Date.now(),
    };
    const merged = this._deepMerge(defaults, config);
    await this.redis.call('JSON.SET', `config:${userId}`, '$', JSON.stringify(merged));
    return merged;
  }

  async getConfig(userId, path = '$') {
    const result = await this.redis.call('JSON.GET', `config:${userId}`, path);
    if (!result) return null;
    const parsed = JSON.parse(result);
    return path === '$' ? parsed[0] : parsed;
  }

  async updateConfig(userId, path, value) {
    await this.redis.call('JSON.SET', `config:${userId}`, path, JSON.stringify(value));
    await this.redis.call('JSON.SET', `config:${userId}`, '$.updatedAt', JSON.stringify(Date.now()));
  }

  async toggleChannel(userId, channel) {
    const channels = await this.redis.call('JSON.GET', `config:${userId}`, '$.notifications.channels');
    if (!channels) return;
    const list = JSON.parse(channels)[0] || [];
    const idx = list.indexOf(channel);
    if (idx >= 0) {
      await this.redis.call('JSON.ARRPOP', `config:${userId}`, '$.notifications.channels', String(idx));
    } else {
      await this.redis.call('JSON.ARRAPPEND', `config:${userId}`, '$.notifications.channels', JSON.stringify(channel));
    }
  }

  _deepMerge(target, source) {
    const result = { ...target };
    for (const key of Object.keys(source)) {
      if (source[key] && typeof source[key] === 'object' && !Array.isArray(source[key])
          && target[key] && typeof target[key] === 'object') {
        result[key] = this._deepMerge(target[key], source[key]);
      } else {
        result[key] = source[key];
      }
    }
    return result;
  }
}

async function main() {
  const store = new UserConfigStore(redis);

  const config = await store.createConfig('user_42', {
    profile: { displayName: 'Alice', bio: 'Full-stack developer' },
    preferences: { theme: 'dark' },
  });
  console.log('主题:', config.preferences.theme);
  console.log('语言:', config.preferences.language);

  console.log('\n当前主题:', await store.getConfig('user_42', '$.preferences.theme'));

  await store.updateConfig('user_42', '$.preferences.theme', 'solarized');
  console.log('更新后:', await store.getConfig('user_42', '$.preferences.theme'));

  await store.toggleChannel('user_42', 'security');
  console.log('通知渠道:', await store.getConfig('user_42', '$.notifications.channels'));

  await redis.quit();
}

main().catch(console.error);
```

## 小结

- **RedisJSON** 让 Redis 支持原生 JSON 文档存储和操作
- JSONPath 语法精确地定位和修改嵌套数据
- 与 RediSearch 结合可直接对 JSON 建立搜索索引
- 就像给 Redis 装了一个 MongoDB——但速度快几个数量级
