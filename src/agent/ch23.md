# 第23章 评估与测试 — 如何衡量Agent的表现

## 23.1 开篇：Agent评估的特殊挑战

测试传统软件：输入 → 预期输出 → assert。但Agent不一样：

1. **非确定性**：同样的输入可能产生不同但都正确的回答
2. **多步骤推理**：中间步骤可能偏差但最终结果对
3. **主观质量**："好"的回答没有唯一标准

    """评估结果"""
    score: float = Field(ge=0, le=1, description="0-1评分")

# 六个核心维度
DIMENSIONS = {
    "correctness": "正确性：答案是否事实正确",
    "faithfulness": "忠实度：答案是否忠于检索上下文",
    "relevance": "相关性：答案是否回答了用户问题",
    "completeness": "完整性：答案是否覆盖各方面",

```

不同场景重点维度：

| 场景 | 重点 | 原因 |
| ------ | ------ | ------ |
| 医疗问答 | 正确性+安全性 | 危及生命 |
| 法律助手 | 忠实度+正确性 | 忠于法律条文 |
| 技术文档 | 相关性+完整性 | 完整准确 |
| 客服 | 相关性+连贯性 | 切题易懂 |

## 23.4 自动化评估方法

### 23.4.1 LLM-as-Judge

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate


class LLMJudge:
    def __init__(self, model="gpt-4o"):
        self.llm = ChatOpenAI(model=model, temperature=0)
    
    def evaluate_correctness(self, question: str, answer: str, reference: str = "") -> EvaluationResult:
        prompt = ChatPromptTemplate.from_messages([
            ("system", "评估正确性。1.0=完全正确|0.7-0.9=基本正确|0.4-0.6=部分正确|0-0.3=不正确"),
            ("human", "问题：{question}\n回答：{answer}\n参考：{reference}\nJSON:{{score,reasoning,dimension}}")
        ])
        return (prompt | self.llm.with_structured_output(EvaluationResult)).invoke({
            "question": question, "answer": answer, "reference": reference or "无"
        })
    
    def evaluate_faithfulness(self, question: str, answer: str, context: str) -> EvaluationResult:
        prompt = ChatPromptTemplate.from_messages([
            ("system", "评估回答是否忠于上下文。每个声明都应有上下文依据。"),
            ("human", "问题：{question}\n上下文：{context}\n回答：{answer}\nJSON:{{score,reasoning,dimension}}")
        ])
        return (prompt | self.llm.with_structured_output(EvaluationResult)).invoke({
            "question": question, "answer": answer, "context": context
        })
    
    def evaluate_relevance(self, question: str, answer: str) -> EvaluationResult:
        prompt = ChatPromptTemplate.from_messages([
            ("system", "评估回答是否切题，是否直接回答了问题。"),
            ("human", "问题：{question}\n回答：{answer}\nJSON:{{score,reasoning,dimension}}")
        ])
        return (prompt | self.llm.with_structured_output(EvaluationResult)).invoke({
            "question": question, "answer": answer
        })
    
    def full_evaluation(self, question, answer, context="", reference="") -> dict:
        results = {
            "correctness": self.evaluate_correctness(question, answer, reference),
            "relevance": self.evaluate_relevance(question, answer),
        }
        if context:
            results["faithfulness"] = self.evaluate_faithfulness(question, answer, context)
        results["overall"] = sum(r.score for r in results.values()) / len(results)
        return results
```

### 23.4.2 规则评估（快速低成本）

```python
import re


class RuleBasedEvaluator:
    def evaluate(self, question: str, answer: str, context: str = "") -> dict:
        results = {
            "length": self._check_length(answer),
            "format": self._check_format(answer),
            "safety": self._check_safety(answer),
            "empty": self._check_not_empty(question, answer),
        }
        if context:
            results["hallucination"] = self._check_hallucination(answer, context)
        results["overall"] = sum(r["score"] for r in results.values()) / len(results)
        return results
    
    def _check_length(self, answer):
        l = len(answer)
        if l < 20: return {"score": 0.3, "reason": "过短"}
        if l > 5000: return {"score": 0.7, "reason": "过长"}
        return {"score": 1.0, "reason": "合理"}
    
    def _check_format(self, answer):
        score = 1.0
        if "```" not in answer and any(k in answer for k in ["代码","函数","import"]):
            score -= 0.2
        if "\n" not in answer and len(answer) > 200:
            score -= 0.1
        return {"score": max(0, score), "reason": "格式"}
    
    def _check_safety(self, answer):
        for p in [r"rm\s+-rf", r"DROP\s+TABLE", r"format\s+[a-zA-Z]:"]:
            if re.search(p, answer, re.IGNORECASE):
                return {"score": 0.0, "reason": f"危险: {p}"}
        return {"score": 1.0, "reason": "安全"}
    
    def _check_not_empty(self, question, answer):
        if not answer or len(answer.strip()) < 10:
            return {"score": 0.3, "reason": "为空或过短"}
        return {"score": 1.0, "reason": "正常"}
    
    def _check_hallucination(self, answer, context):
        a_words = set(re.findall(r'\w+', answer.lower()))
        c_words = set(re.findall(r'\w+', context.lower()))
        overlap = len(a_words & c_words) / max(len(a_words), 1)
        return {"score": min(1.0, overlap * 1.5), "reason": f"重叠:{overlap:.2f}"}
```

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate


class SyntheticTestGenerator:
    """用LLM生成测试数据"""
    
    def __init__(self, llm=None):
        self.llm = llm or ChatOpenAI(model="gpt-4o", temperature=0.8)
    
    def generate(self, topic: str, count: int = 10) -> list:
        """生成测试用例"""
        prompt = ChatPromptTemplate.from_messages([
            ("system", "为AI文档助手生成测试用例。每个包含question、expected_points、difficulty。"),
            ("human", "主题：{topic}\n生成{count}个测试用例，JSON数组格式。")
        ])
        resp = (prompt | self.llm).invoke({"topic": topic, "count": count})
        import json
        try:
            content = resp.content
            if "```" in content:
                content = content.split("```")[1].replace("json", "")
            return json.loads(content)
        except:
            return []
    
    def generate_edge_cases(self, questions: list) -> list:
        """生成边界情况"""
        edge = []
        for q in questions[:5]:
            resp = self.llm.invoke(
                f"为'{q}'生成3个边界变体(模糊/错误信息/超范围)，JSON数组："
            )
            try:
                import json
                edge.extend(json.loads(resp.content))
            except:
                pass
        return edge
    
    def generate_adversarial(self, topic: str) -> list:
        """生成对抗性测试"""
        resp = self.llm.invoke(
            f"为'{topic}'生成5个对抗性测试(幻觉/注入/泄露)，JSON数组："
        )
        import json
        try:
            return json.loads(resp.content)
        except:
            return []

## 23.5.2 A/B测试框架

```python
from dataclasses import dataclass


@dataclass
class ABTestResult:
    a_name: str
    b_name: str
    a_scores: list
    b_scores: list
    
    @property
    def a_avg(self):
        return sum(self.a_scores) / max(len(self.a_scores), 1)
    
    @property
    def b_avg(self):
        return sum(self.b_scores) / max(len(self.b_scores), 1)
    
    @property
    def winner(self):
        if self.a_avg > self.b_avg * 1.05:
            return self.a_name
        elif self.b_avg > self.a_avg * 1.05:
            return self.b_name
        return "no_significant_difference"
    
    def report(self):
        return f"A/B测试: {self.a_name}(avg={self.a_avg:.3f}) vs {self.b_name}(avg={self.b_avg:.3f}) → {self.winner}"


class ABTestRunner:
    def __init__(self, evaluator=None):
        self.evaluator = evaluator or LLMJudge()
    
    def run(self, cases, variant_a, variant_b, name_a="A", name_b="B"):
        a_scores, b_scores = [], []
        for case in cases:
            q = case["input"]["question"]
            ref = case.get("expected", "")
            
            ans_a = variant_a(q)
            a_scores.append(self.evaluator.evaluate_correctness(q, ans_a, ref).score)
            
            ans_b = variant_b(q)
            b_scores.append(self.evaluator.evaluate_correctness(q, ans_b, ref).score)
        
        return ABTestResult(name_a, name_b, a_scores, b_scores)
```

## 23.6 实操：完整测试套件

```python
class DocMindEvaluator:
    def __init__(self, agent, llm_judge=None):
        self.agent = agent
        self.judge = llm_judge or LLMJudge()
        self.rule = RuleBasedEvaluator()
    
    def evaluate_single(self, question, reference="", context="") -> dict:
        answer = self.agent.ask(question) if hasattr(self.agent, 'ask') else "模拟"
        return {
            "question": question, "answer": answer,
            "rule": self.rule.evaluate(question, answer, context),
            "llm": self.judge.full_evaluation(question, answer, context, reference),
        }
    
    def evaluate_batch(self, cases: list) -> dict:
        results = [self.evaluate_single(c["input"]["question"], c.get("expected","")) for c in cases]
        scores = [r["llm"].get("overall", 0) for r in results]
        return {
            "total": len(results),
            "avg": sum(scores)/len(scores) if scores else 0,
            "pass_rate": sum(1 for s in scores if s >= 0.7)/len(scores) if scores else 0,
            "results": results
        }
    
    def generate_report(self, summary) -> str:
        return f"""# DocMind评估报告
- 总数: {summary['total']}
- 平均分: {summary['avg']:.2f}
- 通过率: {summary['pass_rate']*100:.1f}%
"""
```

## 23.7 本章小结

| 方法 | 优点 | 缺点 | 场景 |
| ------ | ------ | ------ | ------ |
| LLM-as-Judge | 灵活准确 | 成本高 | 重要场景 |
| 规则评估 | 快速低成本 | 无法评估语义 | CI/CD |
| 人工评估 | 最准确 | 最慢 | 关键发布 |
| LangSmith | 可视化自动化 | 依赖外部 | 日常监控 |
