## 7.2 Vite 配置与 Monorepo 的配合

编辑 `apps/web/vite.config.ts`：

```typescript
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';
import { fileURLToPath, URL } from 'node:url';

export default defineConfig({
  plugins: [vue()],
  resolve: {
    alias: {
      '@': fileURLToPath(new URL('./src', import.meta.url)),
    },
  },
  server: {
    port: 5173,
    proxy: {
      // 开发环境下，将 /api 请求代理到后端服务
      '/api': {
        target: 'http://localhost:3000',
        changeOrigin: true,
      },
    },
  },
  build: {
    // 将 @monorepo/shared 视为预构建依赖，提升首次加载速度
    commonjsOptions: {
      transformMixedEsModules: true,
    },
  },
  // 关键：告诉 Vite 不要对 workspace 包进行依赖预构建
  // Vite 会直接解析源码，这样修改 shared 后 HMR 能立即生效
  optimizeDeps: {
    include: ['@monorepo/shared'],
  },
});
```

**为什么 `optimizeDeps.include` 要包含 `@monorepo/shared`？**

Vite 的依赖预构建机制会将 `node_modules` 中的 CJS/UMD 模块转换为 ESM 并缓存。workspace 包通过符号链接引用，Vite 默认不会预构建它们。但 `@monorepo/shared` 输出的是构建产物（`dist/` 目录），显式 `include` 可以确保 Vite 正确处理其模块格式，同时享受预构建带来的性能提升。

## 7.4 路由配置（Vue Router）

创建 `apps/web/src/router/index.ts`：

```typescript
import { createRouter, createWebHistory } from 'vue-router';
import type { RouteRecordRaw } from 'vue-router';

// 路由懒加载 — 每个路由对应的组件在首次访问时才加载
const routes: RouteRecordRaw[] = [
  {
    path: '/',
    name: 'Home',
    component: () => import('@/views/HomeView.vue'),
    meta: { title: '首页' },
  },
  {
    path: '/users',
    name: 'UserList',
    component: () => import('@/views/UserListView.vue'),
    meta: { title: '用户列表' },
  },
  {
    path: '/users/:id',
    name: 'UserDetail',
    component: () => import('@/views/UserDetailView.vue'),
    meta: { title: '用户详情' },
    // 路由参数类型校验
    props: (route) => ({ id: String(route.params.id) }),
  },
  {
    path: '/login',
    name: 'Login',
    component: () => import('@/views/LoginView.vue'),
    meta: { title: '登录', requiresAuth: false },
  },
  {
    path: '/:pathMatch(.*)*',
    name: 'NotFound',
    component: () => import('@/views/NotFoundView.vue'),
    meta: { title: '404' },
  },
];

const router = createRouter({
  history: createWebHistory(),
  routes,
});

// 全局导航守卫 — 检查认证状态
router.beforeEach((to, _from, next) => {
  // 设置页面标题
  const title = to.meta.title as string | undefined;
  document.title = title
    ? `${title} - ${import.meta.env.VITE_APP_TITLE}`
    : import.meta.env.VITE_APP_TITLE;

  // 认证检查
  if (to.meta.requiresAuth !== false) {
    const token = localStorage.getItem('accessToken');
    if (!token) {
      next({ name: 'Login', query: { redirect: to.fullPath } });
      return;
    }
  }

  next();
});

export default router;
```

## 7.5 状态管理（Pinia）与共享类型

创建 `apps/web/src/stores/user.ts`：

```typescript
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { User, LoginDto, LoginResponse, ApiResponse } from '@monorepo/shared/types';
import { UserRole } from '@monorepo/shared/types';
import { API_ROUTES } from '@monorepo/shared/constants';
import { apiClient } from '@/utils/api';

export const useUserStore = defineStore('user', () => {
  // ====== State ======
  const currentUser = ref<User | null>(null);
  const isLoading = ref(false);
  const error = ref<string | null>(null);

  // ====== Getters ======
  const isLoggedIn = computed(() => currentUser.value !== null);
  const isAdmin = computed(() => currentUser.value?.role === UserRole.ADMIN);
  const displayName = computed(() => currentUser.value?.username ?? '游客');

  // ====== Actions ======
  async function login(credentials: LoginDto): Promise<boolean> {
    isLoading.value = true;
    error.value = null;

    try {
      const response = await apiClient.post<ApiResponse<LoginResponse>>(
        API_ROUTES.AUTH.LOGIN,
        credentials,
      );

      if (response.data.success && response.data.data) {
        const { user, accessToken, refreshToken } = response.data.data;
        currentUser.value = user;
        localStorage.setItem('accessToken', accessToken);
        localStorage.setItem('refreshToken', refreshToken);
        return true;
      }

      error.value = response.data.error?.message ?? '登录失败';
      return false;
    } catch (err) {
      error.value = err instanceof Error ? err.message : '网络请求失败';
      return false;
    } finally {
      isLoading.value = false;
    }
  }

  async function logout(): Promise<void> {
    try {
      await apiClient.post(API_ROUTES.AUTH.LOGOUT);
    } catch {
      // 即使请求失败也要清理本地状态
    } finally {
      currentUser.value = null;
      localStorage.removeItem('accessToken');
      localStorage.removeItem('refreshToken');
    }
  }

  async function fetchProfile(): Promise<void> {
    isLoading.value = true;
    try {
      const response = await apiClient.get<ApiResponse<User>>(
        API_ROUTES.USERS.PROFILE,
      );
      if (response.data.success && response.data.data) {
        currentUser.value = response.data.data;
      }
    } catch (err) {
      error.value = err instanceof Error ? err.message : '获取用户信息失败';
    } finally {
      isLoading.value = false;
    }
  }

  return {
    // State
    currentUser,
    isLoading,
    error,
    // Getters
    isLoggedIn,
    isAdmin,
    displayName,
    // Actions
    login,
    logout,
    fetchProfile,
  };
});
```

**为什么使用 Composition API 风格的 Pinia Store？**

Pinia 支持 Options API 和 Composition API 两种风格。Composition API 风格（使用 `setup` 函数 + `ref` / `computed`）与 Vue 3 的 `<script setup>` 保持一致，代码更灵活，类型推导也更好。特别是在使用共享类型时，`ref<User | null>` 的类型声明比 Options API 的 state 函数更直观。

## 7.6 API 请求封装（axios + 共享类型推导）

创建 `apps/web/src/utils/api.ts`：

```typescript
import axios from 'axios';
import type { AxiosInstance, InternalAxiosRequestConfig, AxiosResponse } from 'axios';
import type { ApiResponse } from '@monorepo/shared/types';
import { HTTP_STATUS } from '@monorepo/shared/constants';
import router from '@/router';

/**
 * 创建 axios 实例
 * 为什么不直接用全局 axios？因为全局配置会影响所有请求，
 * 包括第三方库发出的请求。独立实例让配置隔离。
 */
export const apiClient: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// ====== 请求拦截器：自动附加 JWT Token ======
apiClient.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = localStorage.getItem('accessToken');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error),
);

// ====== 响应拦截器：统一错误处理 ======
apiClient.interceptors.response.use(
  (response: AxiosResponse) => response,
  async (error) => {
    if (error.response) {
      const status = error.response.status;
      const data: ApiResponse = error.response.data;

      switch (status) {
        case HTTP_STATUS.UNAUTHORIZED:
          // Token 过期或无效，跳转到登录页
          localStorage.removeItem('accessToken');
          localStorage.removeItem('refreshToken');
          router.push({ name: 'Login' });
          break;

        case HTTP_STATUS.FORBIDDEN:
          // 权限不足，可以显示全局提示
          console.error('权限不足:', data.error?.message);
          break;

        case HTTP_STATUS.INTERNAL_SERVER_ERROR:
          console.error('服务器错误:', data.error?.message);
          break;
      }
    } else if (error.request) {
      // 请求已发出但没有收到响应 — 网络问题
      console.error('网络连接失败，请检查网络设置');
    }

    return Promise.reject(error);
  },
);
```

**类型推导的端到端体验：**

```typescript
// 在组件中使用时，类型推导是完整的：
const response = await apiClient.get<ApiResponse<User>>(API_ROUTES.USERS.BY_ID('123'));

// response.data 的类型是 ApiResponse<User>
// response.data.data 的类型是 User | undefined
// response.data.data.username 的类型是 string — 完整的类型推导链！
```

这就是"端到端类型安全"的威力：从后端数据库查出的 `User` 对象，到前端组件中渲染的用户名，整条链路上的类型都是关联的。如果后端的 `User` 接口增加了一个字段，前端使用到该字段的地方会自动获得类型提示；如果字段被删除或改名，TypeScript 会在编译时报错。

```json
  }
```

**为什么共享组件库直接导出源码（`./src/index.ts`）而不是构建产物？**

在开发阶段，直接引用源码有两个优势：

1. **即时 HMR**：修改组件代码后，Vite 能立即检测到变化并热更新，不需要先构建
2. **简化配置**：不需要 tsup/vite 的 library mode 构建步骤

当需要发布到 npm 时，再切换到构建产物导出。对于 monorepo 内部使用，源码导出是更高效的方案。

在 `apps/web/package.json` 中添加依赖：

```json

    "@monorepo/ui": "workspace:*"

}
```

创建登录页面 `apps/web/src/views/LoginView.vue`：

```vue
<script setup lang="ts">
import { ref, reactive } from 'vue';
import { useRouter, useRoute } from 'vue-router';
import { BaseButton, BaseInput } from '@monorepo/ui';
import type { LoginDto } from '@monorepo/shared/types';
import { isValidEmail } from '@monorepo/shared/utils';
import { useUserStore } from '@/stores/user';

const router = useRouter();
const route = useRoute();
const userStore = useUserStore();

const form = reactive<LoginDto>({
  email: '',
  password: '',
});

const errors = reactive({
  email: '',
  password: '',
});

function validate(): boolean {
  let valid = true;
  errors.email = '';
  errors.password = '';

  if (!form.email) {
    errors.email = '请输入邮箱';
    valid = false;
  } else if (!isValidEmail(form.email)) {
    errors.email = '邮箱格式不正确';
    valid = false;
  }

  if (!form.password) {
    errors.password = '请输入密码';
    valid = false;
  }

  return valid;
}

async function handleLogin() {
  if (!validate()) return;

  const success = await userStore.login(form);
  if (success) {
    const redirect = (route.query.redirect as string) || '/';
    router.push(redirect);
  }
}
</script>

<template>
  <div class="login-page">
    <div class="login-card">
      <h2>登录</h2>

      <p v-if="userStore.error" class="global-error">
        {{ userStore.error }}
      </p>

          v-model="form.email"
          type="email"
          label="邮箱"
          placeholder="your@email.com"
```
