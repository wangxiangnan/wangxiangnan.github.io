# 第17章：Cluster 高级——扩缩容与运维

## 引言：运维工程师的瑞士军刀

如果说第16章是教你怎么"建造"一座 Redis Cluster 大厦，那这一章就是教你怎么"维护"它——加楼层（扩容）、拆楼层（缩容）、修水管（故障排查）、装监控（运维监控）。

在生产环境中，搭建集群只是第一步。随着业务增长，你需要在线扩容；机器故障时，你需要排查和恢复；性能下降时，你需要调优。这些"脏活累活"才是运维的日常。

> **编程隐喻时间**：Cluster 运维就像维护一个微服务架构——你不能只是"部署了就完事"，你需要持续监控、动态调整、灰度发布、故障演练。好消息是，Redis Cluster 内置了大量运维工具，就像自带了一个运维工具箱。

## 17.1 在线扩容

### 17.1.1 添加新主节点并分配槽

```bash
# Step 1: 启动新的 Redis 节点（7007）
redis-server redis-7007.conf

# Step 2: 将新节点加入集群
redis-cli --cluster add-node 127.0.0.1:7007 127.0.0.1:7001

# Step 3: 重新分配槽（从现有节点匀一些槽给新节点）
redis-cli --cluster reshard 127.0.0.1:7001 \
  --cluster-from all \
  --cluster-to <new-node-id> \
  --cluster-slots 4096 \
  --cluster-yes
```

### 17.1.2 手动精细迁移槽

```js
const Redis = require('ioredis');

const source = new Redis({ host: '127.0.0.1', port: 7001 });
const target = new Redis({ host: '127.0.0.1', port: 7007 });

async function migrateSlot(sourceNode, targetNode, slot) {
  const sourceId = await sourceNode.cluster('myid');
  const targetId = await targetNode.cluster('myid');

  console.log(`开始迁移槽 ${slot} ...`);

  // 1. 目标节点设置 IMPORTING
  await targetNode.cluster('setslot', slot, 'importing', sourceId);

  // 2. 源节点设置 MIGRATING
  await sourceNode.cluster('setslot', slot, 'migrating', targetId);

  // 3. 迁移该槽的所有 key
  let migratedCount = 0;
  while (true) {
    const keys = await sourceNode.cluster('getkeysinslot', slot, 100);
    if (keys.length === 0) break;

    await sourceNode.migrate('127.0.0.1', 7007, '', 0, 5000, { keys });
    migratedCount += keys.length;
  }

  // 4. 更新槽归属
  await sourceNode.cluster('setslot', slot, 'node', targetId);
  await targetNode.cluster('setslot', slot, 'node', targetId);

  console.log(`槽 ${slot} 迁移完成，共迁移 ${migratedCount} 个 key`);
}

// 迁移槽 0-100
(async () => {
  for (let slot = 0; slot <= 100; slot++) {
    await migrateSlot(source, target, slot);
  }
  source.disconnect();
  target.disconnect();
})();
```

## 17.2 在线缩容

```bash
# Step 1: 将要移除的节点的槽迁移到其他节点
redis-cli --cluster reshard 127.0.0.1:7001 \
  --cluster-from <removing-node-id> \
  --cluster-to <remaining-node-id> \
  --cluster-slots <slot-count> \
  --cluster-yes

# Step 2: 确认该节点没有槽了
redis-cli -p 7003 CLUSTER NODES | grep 7003

# Step 3: 从集群中移除节点
redis-cli --cluster del-node 127.0.0.1:7001 <removing-node-id>

# Step 4: 关闭被移除的节点
redis-cli -p 7003 SHUTDOWN
```

## 17.3 集群运维

### 17.3.1 CLUSTER INFO 状态监控

```js
const Redis = require('ioredis');

async function clusterHealthCheck(seedNodes) {
  const cluster = new Redis.Cluster(seedNodes);

  try {
    // 获取集群基本信息
    const info = await cluster.cluster('info');
    console.log('=== 集群基本信息 ===');
    console.log(info);

    // 解析集群信息
    const lines = info.split('\r\n');
    const clusterInfo = {};
    for (const line of lines) {
      const [key, value] = line.split(':');
      if (key && value) clusterInfo[key] = value;
    }

    // 检查集群状态
    if (clusterInfo.cluster_state !== 'ok') {
      console.error('集群状态异常:', clusterInfo.cluster_state);
    }

    // 获取所有节点信息
    const nodes = await cluster.cluster('nodes');
    console.log('\n=== 节点详情 ===');

    // 解析并检查每个节点
    const nodeLines = nodes.split('\n').filter(l => l.trim());
    for (const line of nodeLines) {
      const parts = line.split(' ');
      const address = parts[1];
      const flags = parts[2];
      const linkState = parts[7];

      if (flags.includes('fail')) {
        console.error(`节点故障: ${address}`);
      }
      if (linkState === 'disconnected') {
        console.warn(`节点断连: ${address}`);
      }
    }

    // 获取槽分配信息
    const slots = await cluster.cluster('slots');
    console.log('\n=== 槽分配 ===');
    for (const slotInfo of slots) {
      console.log(`槽 ${slotInfo[0]}-${slotInfo[1]}: ${slotInfo[2][0]}:${slotInfo[2][1]}`);
    }

  } finally {
    cluster.disconnect();
  }
}

clusterHealthCheck([
  { host: '127.0.0.1', port: 7001 },
  { host: '127.0.0.1', port: 7002 },
]);
```

### 17.3.2 常见故障排查

**故障1：节点失联**

```bash
# 症状：CLUSTER INFO 显示 cluster_state:fail

# 1. 检查失联节点
redis-cli -p 7001 CLUSTER NODES | grep fail

# 2. 检查节点是否存活
redis-cli -p 7003 PING

# 3. 尝试手动修复
redis-cli -p 7003 CLUSTER MEET 127.0.0.1 7001
```

**故障2：槽分配不均**

```bash
# 排查
redis-cli --cluster check 127.0.0.1:7001

# 修复
redis-cli --cluster rebalance 127.0.0.1:7001
```

**故障3：迁移卡住**

```bash
# 排查
redis-cli -p 7001 CLUSTER NODES | grep "\->"

# 修复
redis-cli -p 7001 CLUSTER SETSLOT <slot> STABLE
```

### 17.3.3 集群修复命令

```bash
# Redis 提供了自动修复工具
redis-cli --cluster fix 127.0.0.1:7001
# 会修复迁移中的槽和不一致的槽分配
```

## 17.4 集群备份与恢复

```js
const Redis = require('ioredis');
const path = require('path');
const fs = require('fs');

class ClusterBackup {
  constructor(seedNodes, backupDir) {
    this.cluster = new Redis.Cluster(seedNodes);
    this.backupDir = backupDir;
  }

  async backup() {
    const nodes = await this.cluster.cluster('nodes');
    const masters = nodes.split('\n')
      .filter(line => line.includes('master') && !line.includes('fail'));

    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupPath = path.join(this.backupDir, `backup-${timestamp}`);
    fs.mkdirSync(backupPath, { recursive: true });

    for (const masterLine of masters) {
      const parts = masterLine.split(' ');
      const [host, port] = parts[1].split('@')[0].split(':');

      console.log(`备份节点 ${host}:${port}...`);

      const nodeRedis = new Redis({ host, port: parseInt(port) });
      await nodeRedis.bgsave();

      // 等待 BGSAVE 完成
      while (true) {
        const info = await nodeRedis.info('Persistence');
        if (info.includes('rdb_bgsave_in_progress:0')) break;
        await new Promise(r => setTimeout(r, 1000));
      }

      const config = await nodeRedis.config('get', 'dir');
      const rdbFile = path.join(config[1], 'dump.rdb');
      const destFile = path.join(backupPath, `node-${host}-${port}.rdb`);
      
      try {
        fs.copyFileSync(rdbFile, destFile);
        console.log(`备份完成: ${destFile}`);
      } catch (err) {
        console.error(`备份失败: ${err.message}`);
      }

      nodeRedis.disconnect();
    }

    console.log(`集群备份完成: ${backupPath}`);
    this.cluster.disconnect();
  }
}

const backup = new ClusterBackup(
  [{ host: '127.0.0.1', port: 7001 }],
  '/data/redis-backups'
);
backup.backup().catch(console.error);
```

## 17.6 监控方案

### 17.6.1 使用 INFO 命令监控

```js
const Redis = require('ioredis');

class ClusterMonitor {
  constructor(seedNodes) {
    this.cluster = new Redis.Cluster(seedNodes);
  }

  async collectMetrics() {
    const nodes = await this.cluster.cluster('nodes');
    const nodeLines = nodes.split('\n').filter(l => l.trim());
    const metrics = [];

    for (const line of nodeLines) {
      const parts = line.split(' ');
      const [host, port] = parts[1].split('@')[0].split(':');
      const flags = parts[2];

      const nodeRedis = new Redis({ host, port: parseInt(port) });

      try {
        const memoryInfo = await nodeRedis.info('memory');
        const statsInfo = await nodeRedis.info('stats');

        metrics.push({
          node: `${host}:${port}`,
          role: flags.includes('master') ? 'master' : 'replica',
          usedMemory: this._parse(memoryInfo, 'used_memory_human'),
          memFragmentation: this._parse(memoryInfo, 'mem_fragmentation_ratio'),
          connectedClients: this._parse(statsInfo, 'connected_clients'),
          opsPerSec: this._parse(statsInfo, 'instantaneous_ops_per_sec'),
          keyspaceHits: this._parse(statsInfo, 'keyspace_hits'),
          keyspaceMisses: this._parse(statsInfo, 'keyspace_misses'),
        });
      } finally {
        nodeRedis.disconnect();
      }
    }

    this.cluster.disconnect();
    return metrics;
  }

  _parse(infoString, key) {
    const match = infoString.match(new RegExp(`${key}:(.+?)\\r?\\n`));
    return match ? match[1] : null;
  }
}

(async () => {
  const monitor = new ClusterMonitor([{ host: '127.0.0.1', port: 7001 }]);
  const metrics = await monitor.collectMetrics();
  console.table(metrics);
})();
```

## 小结

Cluster 的高级运维是每个 Redis 管理员必备的技能：

- **扩容**：添加新节点 -> 分配槽 -> 迁移数据（全程在线，不影响服务）
- **缩容**：迁移槽 -> 移除节点 -> 关闭节点（安全有序）
- **监控**：INFO 命令 + Prometheus + Grafana 组成完整的监控体系
- **备份**：定期 BGSAVE + RDB 文件复制，确保数据安全

## 练习题

1. **基础题**：使用 redis-cli 搭建 3 主 3 从的 Cluster，然后在线添加一个 Master 节点并重新分配槽。
2. **进阶题**：编写一个 Node.js 的集群健康检查脚本，定期检查所有节点的状态、内存使用和命中率，异常时发送邮件告警。
3. **挑战题**：模拟在线扩容场景：在不停止服务的情况下，将一个 3 主集群扩展为 4 主集群，并使用 benchmark 工具验证扩容前后的吞吐量变化。
