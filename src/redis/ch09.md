# 第9章：Redis Stream——事件驱动的终极武器

## 引言：从广播到物流系统

上一章我们说 Pub/Sub 像广播电台——听过就忘了。那如果你需要一个"物流系统"呢？每个包裹（消息）都有追踪号，你可以查看历史轨迹，可以有多个快递员（消费者）分工配送，还可以确认"这个包裹已签收"。

这就是 **Redis Stream**——Redis 5.0 引入的消息系统，它是 Kafka 的轻量级表亲，但住在 Redis 的内存里，速度快得飞起。

## 9.1 Stream 是什么？为什么需要它？

Stream 是 Redis 内置的消息队列数据结构。设计灵感直接来自 Kafka，但完美融入了 Redis 的哲学：**简单、快速、内存友好**。

```
Redis Stream 消息结构：

  ID: 1609459200000-0  |  Data: { event: "user_signup", id: 42 }
  ID: 1609459200001-0  |  Data: { event: "order_placed", id: 100 }
  ID: 1609459200002-0  |  Data: { event: "payment_done", id: 100 }

特点：
  - 每条消息有唯一 ID（时间戳+序号）
  - 消息持久化在 Stream 中
  - 支持消费者组和消息确认
  - 可以回溯历史消息
```

## 9.2 Stream vs Pub/Sub vs List

| 特性 | List (LPUSH/BRPOP) | Pub/Sub | Stream |
|------|--------------------|---------|---------|
| 消息持久化 | 是 | 否 | 是 |
| 多消费者 | 否（竞争消费） | 是（广播） | 是（两者都支持） |
| 消费者组 | 否 | 否 | 是 |
| 消息确认 | 否 | 否 | 是 |
| 历史回溯 | 部分 | 否 | 是 |
| 消息ID | 否 | 否 | 是 |
| 消息上限 | 手动管理 | N/A | 是 (MAXLEN) |
| 适用场景 | 简单任务队列 | 实时通知 | 事件驱动、消息队列 |

## 9.3 基础操作

### 9.3.1 添加消息：XADD

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function streamBasics() {
  const id1 = await redis.xadd(
    'events:stream', '*',
    'type', 'user_signup', 'userId', '42', 'email', 'alice@example.com'
  );
  console.log('消息1 ID:', id1);

  const id2 = await redis.xadd(
    'events:stream', '*',
    'type', 'order_placed', 'orderId', '100', 'amount', '99.99'
  );
  console.log('消息2 ID:', id2);

  const id3 = await redis.xadd(
    'events:stream', '*',
    'type', 'payment_done', 'orderId', '100', 'method', 'credit_card'
  );
  console.log('消息3 ID:', id3);

  const length = await redis.xlen('events:stream');
  console.log(`Stream 消息总数: ${length}`);

  const allMessages = await redis.xrange('events:stream', '-', '+');
  console.log('\n所有消息:');
  allMessages.forEach(([id, fields]) => {
    console.log(`  [${id}]`, fields);
  });

  const firstTwo = await redis.xrange('events:stream', '-', '+', 'COUNT', 2);
  console.log('\n前2条消息:');
  firstTwo.forEach(([id, fields]) => {
    console.log(`  [${id}]`, fields);
  });

  const latest = await redis.xrevrange('events:stream', '+', '-', 'COUNT', 1);
  console.log('\n最新消息:', latest);

  await redis.xdel('events:stream', id1);
  console.log(`已删除消息: ${id1}`);

  await redis.xtrim('events:stream', 'MAXLEN', '~', 1000);
  console.log('已裁剪 Stream');

  await redis.quit();
}

streamBasics().catch(console.error);
```

### 9.3.2 读取消息：XREAD

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function readMessages() {
  const history = await redis.xread('COUNT', 10, 'STREAMS', 'events:stream', '0');
  console.log('历史消息:');
  if (history) {
    const [streamName, messages] = history[0];
    messages.forEach(([id, fields]) => {
      console.log(`  [${id}]`, fields);
    });
  }

  console.log('\n等待新消息（5秒超时）...');
  const newMsg = await redis.xread(
    'COUNT', 5, 'BLOCK', 5000, 'STREAMS', 'events:stream', '$'
  );

  if (newMsg) {
    console.log('收到新消息:', newMsg);
  } else {
    console.log('超时，没有新消息');
  }

  await redis.quit();
}

readMessages().catch(console.error);
```

## 9.4 消费者组（Consumer Groups）

这是 Stream 最强大的特性。想象一个快递分拣中心：所有包裹排成一排，多个快递员组成团队，每人领取不同的包裹去配送，配送完成后签收确认。

核心理念：**消息只被组内的一个消费者处理，而且需要确认**。

### 9.4.1 消费者组操作

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function consumerGroupsDemo() {
  for (let i = 1; i <= 5; i++) {
    await redis.xadd(
      'orders:stream', '*', 'orderId', `${i}`, 'item', `商品${i}`, 'qty', `${i * 2}`
    );
  }
  console.log('已添加 5 条订单消息\n');

  try {
    await redis.xgroup('CREATE', 'orders:stream', 'order_processors', '0');
    console.log('消费者组 "order_processors" 创建成功');
  } catch (err) {
    if (err.message.includes('BUSYGROUP')) {
      console.log('消费者组已存在，跳过创建');
    } else throw err;
  }

  console.log('\n--- 消费者1 读取消息 ---');
  const msgs1 = await redis.xreadgroup(
    'GROUP', 'order_processors', 'consumer_1',
    'COUNT', 3, 'STREAMS', 'orders:stream', '>'
  );

  if (msgs1) {
    const [, messages] = msgs1[0];
    messages.forEach(([id, fields]) => {
      console.log(`  消费者1 处理: [${id}]`, fields);
    });
    for (const [id] of messages) {
      await redis.xack('orders:stream', 'order_processors', id);
      console.log(`  已确认消息: ${id}`);
    }
  }

  console.log('\n--- 消费者2 读取消息 ---');
  const msgs2 = await redis.xreadgroup(
    'GROUP', 'order_processors', 'consumer_2',
    'COUNT', 3, 'STREAMS', 'orders:stream', '>'
  );
  if (msgs2) {
    const [, messages] = msgs2[0];
    messages.forEach(([id, fields]) => {
      console.log(`  消费者2 处理: [${id}]`, fields);
    });
  }

  const groupInfo = await redis.xinfo('GROUPS', 'orders:stream');
  console.log('\n消费者组信息:', groupInfo);

  await redis.quit();
}

consumerGroupsDemo().catch(console.error);
```

### 9.4.2 处理未确认消息（XPENDING）

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function handlePendingMessages() {
  const pending = await redis.xpending('orders:stream', 'order_processors');
  console.log('Pending 摘要:', pending);

  const pendingDetail = await redis.xpending(
    'orders:stream', 'order_processors', '-', '+', 10
  );
  console.log('\nPending 详情:');
  if (pendingDetail && pendingDetail.length > 0) {
    pendingDetail.forEach(([id, consumer, idleTime, deliveryCount]) => {
      console.log(`  [${id}] 消费者:${consumer}, 空闲:${idleTime}ms, 投递:${deliveryCount}`);
    });

    console.log('\n重新认领超时的消息...');
    const claimed = await redis.xclaim(
      'orders:stream', 'order_processors', 'consumer_recovery',
      10000, ...pendingDetail.map(([id]) => id)
    );
    console.log('认领到的消息:', claimed);
  }

  await redis.quit();
}

handlePendingMessages().catch(console.error);
```

## 9.5 完整的生产者-消费者模型

```js
const Redis = require('ioredis');

class EventProducer {
  constructor(streamName) {
    this.redis = new Redis({ host: '127.0.0.1', port: 6379 });
    this.streamName = streamName;
  }

  async emit(eventType, data) {
    const fields = ['type', eventType, 'timestamp', Date.now().toString()];
    for (const [key, value] of Object.entries(data)) {
      fields.push(key, typeof value === 'object' ? JSON.stringify(value) : String(value));
    }
    const id = await this.redis.xadd(this.streamName, '*', ...fields);
    console.log(`[Producer] 发布: ${eventType} (ID: ${id})`);
    return id;
  }

  async close() { await this.redis.quit(); }
}

class EventConsumer {
  constructor(streamName, groupName, consumerName, handlers = {}) {
    this.redis = new Redis({ host: '127.0.0.1', port: 6379 });
    this.streamName = streamName;
    this.groupName = groupName;
    this.consumerName = consumerName;
    this.handlers = handlers;
    this.running = false;
  }

  async init() {
    try {
      await this.redis.xgroup('CREATE', this.streamName, this.groupName, '0', 'MKSTREAM');
      console.log(`[${this.consumerName}] 消费者组就绪`);
    } catch (err) {
      if (!err.message.includes('BUSYGROUP')) throw err;
    }
  }

  async processMessage(id, fields) {
    const data = {};
    for (let i = 0; i < fields.length; i += 2) data[fields[i]] = fields[i + 1];

    const handler = this.handlers[data.type] || this.handlers['*'];
    if (handler) {
      try {
        await handler(id, data);
        await this.redis.xack(this.streamName, this.groupName, id);
        console.log(`[${this.consumerName}] 完成: ${data.type} (${id})`);
      } catch (err) {
        console.error(`[${this.consumerName}] 失败: ${data.type}`, err.message);
      }
    } else {
      await this.redis.xack(this.streamName, this.groupName, id);
    }
  }

  async start(batchSize = 10, blockMs = 2000) {
    this.running = true;

    // 先处理 pending 消息
    while (this.running) {
      const pending = await this.redis.xreadgroup(
        'GROUP', this.groupName, this.consumerName,
        'COUNT', batchSize, 'STREAMS', this.streamName, '0'
      );
      if (!pending || pending[0][1].length === 0) break;
      for (const [id, fields] of pending[0][1]) {
        if (!this.running) break;
        await this.processMessage(id, fields);
      }
    }

    // 持续读取新消息
    while (this.running) {
      const result = await this.redis.xreadgroup(
        'GROUP', this.groupName, this.consumerName,
        'COUNT', batchSize, 'BLOCK', blockMs,
        'STREAMS', this.streamName, '>'
      );
      if (result) {
        for (const [id, fields] of result[0][1]) {
          if (!this.running) break;
          await this.processMessage(id, fields);
        }
      }
    }
  }

  async stop() {
    this.running = false;
    await this.redis.quit();
  }
}

async function main() {
  const STREAM = 'app:events';
  const producer = new EventProducer(STREAM);
  const consumer = new EventConsumer(STREAM, 'event_handlers', 'worker_1', {
    'user_signup': async (id, data) => {
      console.log(`   处理注册: 用户 ${data.userId}`);
      await new Promise((r) => setTimeout(r, 100));
    },
    'order_placed': async (id, data) => {
      console.log(`   处理订单: ${data.orderId}`);
      await new Promise((r) => setTimeout(r, 200));
    },
    '*': async (id, data) => {
      console.log(`   通用处理: ${data.type}`);
    },
  });

  await consumer.init();
  const consumerPromise = consumer.start(5, 1000);
  await new Promise((r) => setTimeout(r, 500));

  await producer.emit('user_signup', { userId: '42', email: 'alice@example.com' });
  await new Promise((r) => setTimeout(r, 300));
  await producer.emit('order_placed', { orderId: '100', amount: '99.99' });
  await new Promise((r) => setTimeout(r, 2000));

  await consumer.stop();
  await producer.close();
  try { await consumerPromise; } catch (e) { /* ignore */ }
}

main().catch(console.error);
```

## 9.6 Stream 的持久化特性

```js
const Redis = require('ioredis');
const redis = new Redis({ host: '127.0.0.1', port: 6379 });

async function streamPersistence() {
  const info = await redis.xinfo('STREAM', 'app:events');
  console.log('Stream 信息:');
  for (let i = 0; i < info.length; i += 2) {
    console.log(`  ${info[i]}: ${info[i + 1]}`);
  }

  // 使用 MAXLEN 控制大小
  await redis.xadd('logs:stream', 'MAXLEN', '~', 10000, '*',
    'level', 'info', 'msg', 'Hello');

  // 时间范围查询
  const oneHourAgo = Date.now() - 3600000;
  const recent = await redis.xrange('app:events', `${oneHourAgo}-0`, '+');
  console.log(`\n最近1小时消息数: ${recent.length}`);

  await redis.quit();
}

streamPersistence().catch(console.error);
```

## 小结

- **Stream** 是 Redis 内置的消息队列，功能对标 Kafka 的核心特性
- 支持**消费者组**，实现消息的负载均衡和可靠投递
- 消息**持久化**在 Redis 中，支持回溯和历史查询
- **XPENDING + XCLAIM** 处理消费者故障后的消息重新投递
